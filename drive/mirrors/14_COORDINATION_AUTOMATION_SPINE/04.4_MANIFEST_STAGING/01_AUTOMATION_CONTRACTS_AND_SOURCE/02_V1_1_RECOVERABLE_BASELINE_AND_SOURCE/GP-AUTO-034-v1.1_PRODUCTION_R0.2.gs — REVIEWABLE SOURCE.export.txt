 * and Dylan Roy operator ratification dated 2026-07-21.
 *
 * DESIGN CONTRACT
 * 1. Source research artifacts are read-only.
 * 2. Drive traversal is direct-child only. RECURSIVE_SCAN must remain FALSE.

 *    the last completed successful refresh exceeds the configured threshold.

 *    alter configuration, scope, safety switches, authority, or eligibility.

/**
 * Main orchestration entry point.
 * The function builds every output in memory, writes hidden staging sheets,
 * validates them, publishes a staged manifest, commits live outputs with
 * rollback protection, updates navigation, and only then records SUCCESS.
 */
function refreshAllRegisters() {
  const lock = LockService.getScriptLock();
  const started = new Date();
  const runId = makeRunId_(started);
  let context = null;
  let stageState = null;
  let navBackup = null;
  let manifestPublication = null;


  if (!lock.tryLock(GP_AUTO.LOCK_WAIT_MS)) {
    const lockError = new Error('FAIL-CLOSED: another refresh holds the script lock.');
    appendFailureWithoutContext_(runId, started, lockError);
    throw lockError;
  }


  try {
    const ss = SpreadsheetApp.openById(GP_AUTO.REGISTRY_SPREADSHEET_ID);
    const config = readConfig_(ss);
    assertSafety_(config);
    assertAuthorization_(config);
    context = makeContext_(ss, config, runId, started);


    const closureState = readClosureState_(ss);
    const inventory = collectInventory_(context, closureState);
    const derived = deriveWorkspaceState_(context, inventory, closureState);
    const manifest = buildManifest_(context, inventory, derived, closureState);
    const manifestValidation = validateManifest_(manifest, context);
    if (!manifestValidation.ok) {
      throw new Error('Manifest validation failed: ' + manifestValidation.errors.join(' | '));
    }


    const tables = buildGeneratedTables_(context, inventory, derived, manifest);
    validateGeneratedTables_(tables);


    stageState = stageGeneratedSheets_(context, tables);
    validateStagedSheets_(context, stageState, tables);


    manifestPublication = publishManifestStaged_(context, manifest);
    navBackup = backupNavigationDoc_(context);


    commitGeneratedSheets_(context, stageState, tables);
    updateNavigationDoc_(context, derived, manifest, manifestPublication, false, '');
    setDashboardFreshness_(context, false, 'CURRENT — last validated refresh ' + iso_(new Date()));


    const finished = new Date();
    const receipt = makeRunReceipt_(context, inventory, tables, manifestPublication, finished, 'SUCCESS', []);
    appendRunLog_(context.ss, receipt);
    publishValidationReceipt_(context, receipt, manifestValidation);


    const props = PropertiesService.getScriptProperties();
    props.setProperty(GP_AUTO.PROPERTY_LAST_SUCCESS, finished.toISOString());
    props.setProperty(GP_AUTO.PROPERTY_LAST_RUN, runId);
    props.setProperty(GP_AUTO.PROPERTY_LAST_MANIFEST_ID, manifestPublication.canonicalFileId);
    props.setProperty(GP_AUTO.PROPERTY_LAST_MANIFEST_HASH, manifestPublication.sha256);
    props.deleteProperty(GP_AUTO.PROPERTY_LAST_STALE_TOKEN);


    cleanTransientStageSheets_(context, stageState);
    return receipt;
  } catch (error) {
    if (context && stageState && stageState.commitStarted) {
      try {
        rollbackGeneratedSheets_(context, stageState);
      } catch (rollbackError) {
        error.message += ' | ROLLBACK ERROR: ' + rollbackError.message;
      }
    }


    if (context && navBackup) {
      try {
        restoreNavigationDoc_(context, navBackup);
      } catch (navRestoreError) {
        error.message += ' | NAVIGATION ROLLBACK ERROR: ' + navRestoreError.message;
      }
    }


    if (context) {
      const warnings = ['Run failed closed; previous validated manifest remains authoritative.'];
      if (manifestPublication && manifestPublication.stagedFileId) {
        warnings.push('Staged manifest retained for audit: ' + manifestPublication.stagedFileId);
      }
      appendRunLogSafe_(context.ss, {
        run_id: runId,
        started_at: started.toISOString(),
        finished_at: new Date().toISOString(),
        mode: 'REFRESH',
        files_scanned: context.inventoryCount || 0,
        rows_written: 0,
        warnings: warnings.join(' | '),
        result: 'FAILED',
        details: truncate_(error.stack || error.message, 45000)
      });
      setStaleBannerSafe_(context, 'FAILED — ' + error.message);
    } else {
      appendFailureWithoutContext_(runId, started, error);
    }
    throw error;
  } finally {
    lock.releaseLock();
  }
}


/**
 * Installs one hourly refresh trigger and one hourly stale-guard trigger.
 * This function refuses installation unless both operator authorization and
 * the HOURLY_TRIGGER switch are TRUE. Existing matching triggers are removed
 * first; unrelated user triggers are untouched.
 */
function installHourlyTrigger() {
  const ss = SpreadsheetApp.openById(GP_AUTO.REGISTRY_SPREADSHEET_ID);
  const config = readConfig_(ss);
  assertSafety_(config);
  assertAuthorization_(config);
  requireTrue_(config, 'HOURLY_TRIGGER', 'Hourly trigger installation is disabled.');


  removeRefreshTriggers();
  ScriptApp.newTrigger(GP_AUTO.RUN_HANDLER).timeBased().everyHours(1).create();
  ScriptApp.newTrigger(GP_AUTO.STALE_HANDLER).timeBased().everyHours(1).create();


  appendRunLog_(ss, {
    run_id: makeRunId_(new Date()),
    started_at: new Date().toISOString(),
    finished_at: new Date().toISOString(),
    mode: 'TRIGGER_INSTALL',
    files_scanned: 0,
    rows_written: 0,
    warnings: '',
    result: 'SUCCESS',
    details: 'Installed hourly refresh and stale-guard triggers.'
  });
}


/** Removes only the two handlers owned by this package. */
function removeRefreshTriggers() {
  const handlers = {};
  handlers[GP_AUTO.RUN_HANDLER] = true;
  handlers[GP_AUTO.STALE_HANDLER] = true;
  ScriptApp.getProjectTriggers().forEach(function(trigger) {
    if (handlers[trigger.getHandlerFunction()]) {
      ScriptApp.deleteTrigger(trigger);
    }
  });
}


/**
 * Checks whether the last SUCCESS is older than STALE_THRESHOLD_MINUTES.
 * A stale condition writes one append-only FAILED receipt per stale token and
 * updates both Dashboard and the Navigation Companion with an explicit banner.
 */
function enforceStaleStatus() {
  const ss = SpreadsheetApp.openById(GP_AUTO.REGISTRY_SPREADSHEET_ID);
  const config = readConfig_(ss);
  assertSafety_(config);
  const thresholdMinutes = intConfig_(config, 'STALE_THRESHOLD_MINUTES', 120);
  if (thresholdMinutes !== 120) {
    throw new Error('FAIL-CLOSED: STALE_THRESHOLD_MINUTES must equal 120 under R0.2.');
  }


  const props = PropertiesService.getScriptProperties();
  const lastSuccessIso = props.getProperty(GP_AUTO.PROPERTY_LAST_SUCCESS);
  const lastSuccess = lastSuccessIso ? new Date(lastSuccessIso) : null;
  const ageMinutes = lastSuccess ? (Date.now() - lastSuccess.getTime()) / 60000 : Infinity;
  const stale = !lastSuccess || ageMinutes > thresholdMinutes;
  const context = makeContext_(ss, config, makeRunId_(new Date()), new Date());


  if (!stale) {
    setDashboardFreshness_(context, false,
      'CURRENT — last successful refresh ' + lastSuccess.toISOString());
    return {stale: false, age_minutes: ageMinutes};
  }


  const staleToken = lastSuccessIso || 'NO_SUCCESS_RECORDED';
  if (props.getProperty(GP_AUTO.PROPERTY_LAST_STALE_TOKEN) !== staleToken) {
    appendRunLog_(ss, {
      run_id: context.runId,
      started_at: context.started.toISOString(),
      finished_at: new Date().toISOString(),
      mode: 'STALE_GUARD',
      files_scanned: 0,
      rows_written: 0,
      warnings: 'No validated refresh completed inside the 120-minute window.',
      result: 'FAILED',
      details: 'Last success: ' + staleToken + '; age_minutes=' + ageMinutes
    });
    props.setProperty(GP_AUTO.PROPERTY_LAST_STALE_TOKEN, staleToken);
  }


  const message = 'STALE — no successful refresh within 120 minutes. ' +
    'Downstream models must not rely on generated registers or manifest.';
  setDashboardFreshness_(context, true, message);
  updateNavigationDoc_(context, null, null, null, true, message);
  return {stale: true, age_minutes: ageMinutes};
}


/** Runs non-destructive unit and integration-readiness checks T1–T16. */
function runAcceptanceTests() {
  const ss = SpreadsheetApp.openById(GP_AUTO.REGISTRY_SPREADSHEET_ID);
  const config = readConfig_(ss);
  const results = [];


  results.push(testResult_('T1', 'Hard safety switches fail closed', testHardSwitches_(config)));
  results.push(testResult_('T2', 'Out-of-scope writes rejected by target guard', testWriteTargetGuard_(config)));
  results.push(testResult_('T3', 'Last validated snapshot preserved by stage/backup design', testStageContract_()));
  results.push(testResult_('T4', 'Injected metadata cannot mutate configuration', testInstructionIsolation_()));
  results.push(testResult_('T5', 'Manifest publish is create-validate-rename, not blind retry', testManifestPublishContract_()));
  results.push(testResult_('T6', 'Unreadable and uncertain states remain representable', testReadStatusEnums_()));
  results.push(testResult_('T7', 'Same declared ID can retain distinct Drive IDs', testCollisionRetention_()));
  results.push(testResult_('T8', 'Object-card dimensions remain distinct', testObjectCardSeparation_()));
  results.push(testResult_('T9', 'Malformed cards are flagged', testMalformedCard_()));
  results.push(testResult_('T10', 'Closure contradiction cannot become REFEREE-READY', testClosureGate_()));
  results.push(testResult_('T11', 'Superseded dependency removes eligibility', testSupersededDependencyGate_()));
  results.push(testResult_('T12', 'Instrument cannot compile as proof prose', testInstrumentGate_()));
  results.push(testResult_('T13', 'Unscoped cited source is not extraction eligible', testScopeGate_()));
  results.push(testResult_('T14', 'Coverage gaps are explicit', testCoverageGapContract_()));
  results.push(testResult_('T15', 'Stale threshold is exactly 120 minutes', intConfig_(config, 'STALE_THRESHOLD_MINUTES', 0) === 120));
  results.push(testResult_('T16', 'Run receipt contains required audit fields', testReceiptSchema_()));


  const failed = results.filter(function(row) { return row.status !== 'PASS'; });
  const detail = results.map(function(row) {
    return row.id + ': ' + row.status + ' — ' + row.description;
  }).join('\n');


  appendRunLog_(ss, {
    run_id: makeRunId_(new Date()),
    started_at: new Date().toISOString(),
    finished_at: new Date().toISOString(),
    mode: 'ACCEPTANCE_TEST',
    files_scanned: 0,
    rows_written: results.length,
    warnings: failed.length ? failed.length + ' test(s) failed.' : '',
    result: failed.length ? 'FAILED' : 'SUCCESS',
    details: detail
  });


  if (failed.length) {
    throw new Error('Acceptance battery failed: ' + failed.map(function(x) { return x.id; }).join(', '));
  }
  return results;
}


function readConfig_(ss) {
  const sheet = requiredSheet_(ss, 'Config');
  const values = sheet.getDataRange().getDisplayValues();
  if (values.length < 2) throw new Error('Config sheet has no configuration rows.');


  const config = {};
  for (let r = 1; r < values.length; r++) {
    const key = String(values[r][0] || '').trim();
    if (!key) continue;
    const enabled = parseBoolean_(values[r][2]);
    if (enabled === false) continue;
    config[key] = String(values[r][1] == null ? '' : values[r][1]).trim();
  }


  if (config.REGISTRY_SPREADSHEET_ID &&
      config.REGISTRY_SPREADSHEET_ID !== GP_AUTO.REGISTRY_SPREADSHEET_ID) {
    throw new Error('FAIL-CLOSED: registry ID does not match the approved target.');
  }
  config.REGISTRY_SPREADSHEET_ID = GP_AUTO.REGISTRY_SPREADSHEET_ID;
  return Object.freeze(config);
}


function assertSafety_(config) {
  GP_AUTO.HARD_FALSE_SWITCHES.forEach(function(key) {
    if (parseBoolean_(config[key]) !== false) {
      throw new Error('FAIL-CLOSED: ' + key + ' must be explicitly FALSE.');
    }
  });
  if (intConfig_(config, 'STALE_THRESHOLD_MINUTES', 120) !== 120) {
    throw new Error('FAIL-CLOSED: stale threshold must be 120 minutes.');
  }
}


function assertAuthorization_(config) {
  requireTrue_(config, 'AUTOMATION_AUTHORIZED', 'Operator authorization is absent.');
}


function requireTrue_(config, key, message) {
  if (parseBoolean_(config[key]) !== true) {
    throw new Error('FAIL-CLOSED: ' + message + ' [' + key + ']');
  }
}


function makeContext_(ss, config, runId, started) {
  const sourceScopes = collectConfiguredScopes_(config);
  const manifestFolderId = config.MANIFEST_OUTPUT_FOLDER_ID ||
    config.TREE_04_4_FOLDER_ID || GP_AUTO.DEFAULT_MANIFEST_FOLDER_ID;
  const navigationDocId = config.NAVIGATION_DOC_ID || GP_AUTO.DEFAULT_NAVIGATION_DOC_ID;
  const allowedDriveWriteIds = {};
  allowedDriveWriteIds[manifestFolderId] = true;
  allowedDriveWriteIds[navigationDocId] = true;


  return {
    ss: ss,
    config: config,
    runId: runId,
    started: started,
    sourceScopes: sourceScopes,
    manifestFolderId: manifestFolderId,
    navigationDocId: navigationDocId,
    allowedDriveWriteIds: allowedDriveWriteIds,
    warnings: [],
    inventoryCount: 0
  };
}


function collectConfiguredScopes_(config) {
  const scopes = [];
  const seen = {};


  Object.keys(config).sort().forEach(function(key) {
    if (key === 'PRIMARY_OQ_FOLDER_ID' || /^SOURCE_FOLDER_ID_\d+$/.test(key)) {
      const id = config[key];
      if (id && !seen[id]) {
        scopes.push({scope_key: key, folder_id: id, root_direct: false});
        seen[id] = true;
      }
    }
  });


  if (parseBoolean_(config.SOURCE_ROOT_DIRECT) === true && !seen.root) {
    scopes.push({scope_key: 'SOURCE_ROOT_DIRECT', folder_id: 'root', root_direct: true});
  }


  if (!scopes.length) {
    throw new Error('FAIL-CLOSED: no explicitly configured source scopes.');
  }
  return scopes;
}


function collectInventory_(context, closureState) {
  const byDriveId = {};
  const coverageGaps = [];


  context.sourceScopes.forEach(function(scope) {
    const folder = scope.root_direct ? DriveApp.getRootFolder() : DriveApp.getFolderById(scope.folder_id);
    const files = folder.getFiles();
    while (files.hasNext()) {
      const file = files.next();
      if (byDriveId[file.getId()]) continue;
      const record = inspectFile_(file, scope, context, closureState);
      byDriveId[record.drive_id] = record;
    }


    const folders = folder.getFolders();
    while (folders.hasNext()) {
      const child = folders.next();
      coverageGaps.push({
        parent_scope: scope.scope_key,
        parent_folder_id: scope.folder_id,
        discovered_folder_id: child.getId(),
        discovered_folder_name: child.getName(),
        status: 'COVERAGE-GAP',
        reason: 'Direct-child-only policy: nested folder not traversed unless separately configured.'
      });
    }
  });


  const inventory = Object.keys(byDriveId).map(function(id) { return byDriveId[id]; });
  inventory.sort(function(a, b) {
    return a.created_at.localeCompare(b.created_at) || a.drive_id.localeCompare(b.drive_id);
  });
  context.inventoryCount = inventory.length;
  context.coverageGaps = coverageGaps;
  return inventory;
}


function inspectFile_(file, scope, context, closureState) {
  const mimeType = file.getMimeType();
  const title = file.getName();
  let readResult;
  try {
    readResult = readFileText_(file, mimeType);
  } catch (error) {
    readResult = {
      text: '',
      rawBytes: null,
      fetch_path: 'ERROR',
      sync_status: 'SYNC-UNCERTAIN',
      machine_readable: false,
      read_warning: error.message
    };
  }


  const metadata = parseMetadata_(readResult.text, title);
  const normalizedText = normalizeText_(readResult.text);
  const rawHash = readResult.rawBytes ? sha256Bytes_(readResult.rawBytes) :
    (readResult.text ? sha256Text_(readResult.text) : '');
  const normalizedHash = normalizedText ? sha256Text_(normalizedText) : '';
  const created = file.getDateCreated();
  const modified = file.getLastUpdated();
  const closure = lookupClosure_(closureState, metadata.artifact_id, title, file.getId());


  const record = {
    drive_id: file.getId(),
    title: title,
    declared_artifact_id: metadata.artifact_id || '',
    artifact_id: metadata.artifact_id || '',
    line: metadata.line || lineFromArtifactId_(metadata.artifact_id),
    organization: metadata.organization || 'UNDECLARED',
    class: metadata.class || classFromArtifactId_(metadata.artifact_id),
    version: metadata.version || versionFromTitle_(title),
    mime_type: mimeType,
    created_time: created.toISOString(),
    modified_time: modified.toISOString(),
    created_at: created.toISOString(),
    modified_at: modified.toISOString(),
    source_folder_id: scope.folder_id,
    source_scope_key: scope.scope_key,
    source_urls: [file.getUrl()],
    raw_sha256: rawHash,
    normalized_sha256: normalizedHash,
    content_hash: rawHash,
    normalized_text_hash: normalizedHash,
    hash_basis: readResult.rawBytes ? 'RAW_DRIVE_BYTES' :
      (readResult.text ? 'EXTRACTED_TEXT' : 'UNAVAILABLE'),
    fetch_path: readResult.fetch_path,
    sync_status: readResult.sync_status,
    machine_readable: readResult.machine_readable,
    read_warning: readResult.read_warning || '',
    topics: metadata.topics,
    object_tags: metadata.object_tags,
    priority: metadata.priority || 'UNDECLARED',
    status: metadata.status || 'UNDECLARED',
    authority: metadata.authority || 'none',
    canonical_impact: metadata.canonical_impact || 'NONE',
    dependencies: metadata.dependencies,
    dependency_hashes: metadata.dependency_hashes,
    supersedes: metadata.supersedes,
    superseded_by: [],
    lineage_of: metadata.lineage_of,
    measure: metadata.measure || 'UNDECLARED',
    model_layer: metadata.model_layer || 'UNDECLARED',
    frame: metadata.frame || 'UNDECLARED',
    closure_state: closure.state || metadata.closure_state || 'UNVERIFIED',
    extraction_class: metadata.extraction_class || 'UNCLASSIFIED',
    evidence_chain: metadata.evidence_chain,
    reviewed_by: GP_AUTO.RELEASE,
    reviewed_at: new Date().toISOString(),
    human_decision: metadata.human_decision,
    help_needed: metadata.help_needed || '',
    offered_capacity: metadata.offered_capacity || '',
    decisive_test: metadata.decisive_test || '',
    falsifiers: metadata.falsifiers,
    tree_path: TREE_PATHS.UNROUTED,
    validation_errors: [],
    text_excerpt: truncate_(normalizedText, 8000)
  };


  record.tree_path = routeTreePath_(record);
  applyExtractionEligibility_(record, closure);
  validateObjectCard_(record);
  return record;
}


function readFileText_(file, mimeType) {
  const id = file.getId();
  const result = {
    text: '', rawBytes: null, fetch_path: '', sync_status: 'COMPLETE',
    machine_readable: true, read_warning: ''
  };


  if (mimeType === MimeType.GOOGLE_DOCS) {
    result.text = DocumentApp.openById(id).getBody().getText();
    result.fetch_path = 'DOCUMENT_APP_TEXT';
  } else if (mimeType === MimeType.GOOGLE_SHEETS) {
    result.text = extractSpreadsheetText_(id);
    result.fetch_path = 'SPREADSHEET_APP_BOUNDED_VALUES';
  } else if (mimeType === MimeType.GOOGLE_SLIDES) {
    result.text = extractSlidesText_(id);
    result.fetch_path = 'SLIDES_APP_TEXT';
  } else if (isTextMime_(mimeType, file.getName())) {
    const blob = file.getBlob();
    result.rawBytes = blob.getBytes();
    result.text = blob.getDataAsString('UTF-8');
    result.fetch_path = 'DRIVE_BLOB_UTF8';
  } else {
    result.fetch_path = 'METADATA_ONLY';
    result.sync_status = 'UNREADABLE-MIME';
    result.machine_readable = false;
    result.read_warning = 'Unsupported semantic extraction path for MIME ' + mimeType;
  }


  if (result.text.length > GP_AUTO.MAX_FILE_TEXT_CHARS) {
    result.text = result.text.substring(0, GP_AUTO.MAX_FILE_TEXT_CHARS);
    result.sync_status = 'SYNC-UNCERTAIN';
    result.read_warning = 'Text truncated at configured extraction ceiling.';
  }
  return result;
}


function extractSpreadsheetText_(spreadsheetId) {
  const ss = SpreadsheetApp.openById(spreadsheetId);
  const sections = [];
  ss.getSheets().forEach(function(sheet) {
    const rows = Math.min(sheet.getLastRow(), GP_AUTO.MAX_SHEET_EXTRACT_ROWS);
    const cols = Math.min(sheet.getLastColumn(), GP_AUTO.MAX_SHEET_EXTRACT_COLUMNS);
    sections.push('### SHEET: ' + sheet.getName());
    if (!rows || !cols) return;
    const values = sheet.getRange(1, 1, rows, cols).getDisplayValues();
    values.forEach(function(row) { sections.push(row.join('\t')); });
  });
  return sections.join('\n');
}


function extractSlidesText_(presentationId) {
  const presentation = SlidesApp.openById(presentationId);
  const output = [];
  presentation.getSlides().forEach(function(slide, index) {
    output.push('### SLIDE ' + (index + 1));
    slide.getPageElements().forEach(function(element) {
      if (element.getPageElementType() === SlidesApp.PageElementType.SHAPE) {
        try {
          const text = element.asShape().getText().asString();
          if (text) output.push(text);
        } catch (ignored) {}
      } else if (element.getPageElementType() === SlidesApp.PageElementType.TABLE) {
        try {
          const table = element.asTable();
          for (let r = 0; r < table.getNumRows(); r++) {
            const row = [];
            for (let c = 0; c < table.getNumColumns(); c++) {
              row.push(table.getCell(r, c).getText().asString());
            }
            output.push(row.join('\t'));
          }
        } catch (ignoredTable) {}
      }
    });
  });
  return output.join('\n');
}


function isTextMime_(mimeType, fileName) {
  if (/^text\//i.test(mimeType)) return true;
  if (/json|javascript|xml|yaml|csv/i.test(mimeType)) return true;
  return /\.(md|txt|json|jsonl|csv|tsv|py|js|gs|ts|yaml|yml|xml|html|css|tex|r)$/i.test(fileName);
}


function parseMetadata_(text, title) {
  const head = String(text || '').substring(0, 30000);
  const map = {};
  head.split(/\r?\n/).forEach(function(line) {
    const match = line.match(/^\s*([A-Za-z][A-Za-z0-9 _\-/]{1,50})\s*:\s*(.*?)\s*$/);
    if (!match) return;
    const key = normalizeMetadataKey_(match[1]);
    if (!map[key]) map[key] = match[2];
  });


  const artifactId = firstNonEmpty_(map.ARTIFACT_ID, artifactIdFromTitle_(title));
  return {
    artifact_id: artifactId,
    line: firstNonEmpty_(map.LINE, map.MODEL_LINE),
    organization: firstNonEmpty_(map.ORGANIZATION, map.AUTHOR_ORGANIZATION),
    class: firstNonEmpty_(map.CLASS, classFromArtifactId_(artifactId)),
    version: firstNonEmpty_(map.VERSION, versionFromTitle_(title)),
    topics: parseList_(map.TOPICS),
    object_tags: parseList_(firstNonEmpty_(map.OBJECT_TAGS, map.OBJECTS)),
    priority: map.PRIORITY || '',
    status: map.STATUS || '',
    authority: map.AUTHORITY || '',
    canonical_impact: firstNonEmpty_(map.CANONICAL_IMPACT, map.CANONICAL_STATUS),
    dependencies: parseList_(map.DEPENDENCIES),
    dependency_hashes: parseList_(map.DEPENDENCY_HASHES),
    supersedes: parseList_(map.SUPERSEDES),
    lineage_of: parseList_(map.LINEAGE_OF),
    decisive_test: firstNonEmpty_(map.DECISIVE_TEST, map.FALSIFICATION_TEST),
    falsifiers: parseList_(map.FALSIFIERS),
    help_needed: firstNonEmpty_(map.HELP_NEEDED, map.BLOCKER),
    offered_capacity: firstNonEmpty_(map.OFFERED_CAPACITY, map.HELP_OFFER),
    measure: map.MEASURE || '',
    model_layer: map.MODEL_LAYER || '',
    frame: map.FRAME || '',
    closure_state: map.CLOSURE_STATE || '',
    extraction_class: map.EXTRACTION_CLASS || '',
    evidence_chain: parseList_(map.EVIDENCE_CHAIN),
    human_decision: parseNullableBoolean_(map.HUMAN_DECISION)
  };
}


function normalizeMetadataKey_(key) {
  return String(key).trim().toUpperCase().replace(/[\s\-/]+/g, '_');
}


function deriveWorkspaceState_(context, inventory, closureState) {
  const byArtifactId = {};
  const byNormalizedHash = {};
  const byTitle = {};
  const recentCutoff = Date.now() - intConfig_(context.config, 'RECENT_WINDOW_MINUTES', 120) * 60000;
  const recent = [];
  const humanDecisionCount = inventory.filter(function(x) { return x.human_decision === true; }).length;


  inventory.forEach(function(record) {
    if (record.artifact_id) pushMapArray_(byArtifactId, record.artifact_id, record);
    if (record.normalized_text_hash) pushMapArray_(byNormalizedHash, record.normalized_text_hash, record);
    pushMapArray_(byTitle, record.title.toLowerCase(), record);
    if (new Date(record.modified_at).getTime() >= recentCutoff) recent.push(record);
  });


  Object.keys(byArtifactId).forEach(function(id) {
    const group = byArtifactId[id];
    if (group.length > 1) {
      group.forEach(function(record) {
        record.validation_errors.push('DECLARED_ID_COLLISION:' + id);
      });
    }
  });


  const supersededIndex = {};
  inventory.forEach(function(record) {
    record.supersedes.forEach(function(oldId) {
      pushMapArray_(supersededIndex, oldId, record.artifact_id || record.drive_id);
    });
  });
  inventory.forEach(function(record) {
    record.superseded_by = supersededIndex[record.artifact_id] || [];
    if (record.superseded_by.length && record.extraction_class === 'REFEREE-READY') {
      record.extraction_class = 'ACTIVE';
      record.validation_errors.push('REFEREE_READY_REVOKED_BY_SUPERSESSION');
    }
  });


  return {
    byArtifactId: byArtifactId,
    byNormalizedHash: byNormalizedHash,
    byTitle: byTitle,
    recent: recent,
    humanDecisionCount: humanDecisionCount,
    closureState: closureState,
    coverageGaps: context.coverageGaps || []
  };
}


function buildManifest_(context, inventory, derived, closureState) {
  const scopeDescriptor = context.sourceScopes.map(function(scope) {
    return {
      scope_key: scope.scope_key,
      folder_id: scope.folder_id,
      traversal: 'DIRECT_CHILDREN_ONLY'
    };
  });


  return {
    manifest_name: GP_AUTO.MANIFEST_FILE_NAME,
    schema_version: 'WORKSPACE-MANIFEST-v1.1-R0.2',
    automation_release: GP_AUTO.RELEASE,
    config_version: context.config.CONFIG_VERSION || GP_AUTO.CONFIG_VERSION,
    run_id: context.runId,
    generated_at: new Date().toISOString(),
    stale_threshold_minutes: 120,
    recursive_scan: false,
    safety: {
      WRITE_SOURCE_FILES: false,
      DELETE_OR_TRASH: false,
      AUTO_PROMOTE: false,
      RECURSIVE_SCAN: false
    },
    source_scope: scopeDescriptor,
    source_count: inventory.length,
    human_decision_count: derived.humanDecisionCount,
    coverage_gaps: derived.coverageGaps,
    objects: inventory.map(function(record) {
      const copy = shallowClone_(record);
      delete copy.text_excerpt;
      return copy;
    })
  };
}


function validateManifest_(manifest, context) {
  const errors = [];
  if (!manifest || !Array.isArray(manifest.objects)) errors.push('objects must be an array');
  if (manifest.recursive_scan !== false) errors.push('recursive_scan must be false');
  GP_AUTO.HARD_FALSE_SWITCHES.forEach(function(key) {
    if (manifest.safety[key] !== false) errors.push('unsafe manifest switch ' + key);
  });


  const driveIds = {};
  (manifest.objects || []).forEach(function(object, index) {
    GP_AUTO.REQUIRED_MANIFEST_PROPERTIES.forEach(function(key) {
      if (!Object.prototype.hasOwnProperty.call(object, key)) {
        errors.push('object[' + index + '] missing property ' + key);
      }
    });
    if (!object.drive_id) errors.push('object[' + index + '] missing drive_id');
    if (driveIds[object.drive_id]) errors.push('duplicate drive_id ' + object.drive_id);
    driveIds[object.drive_id] = true;
    if (object.extraction_class === 'REFEREE-READY' && object.validation_errors.length) {
      errors.push('REFEREE-READY object has validation errors: ' + object.drive_id);
    }
    if (object.tree_path === TREE_PATHS.UNROUTED) {
      object.validation_errors.push('TREE_ROUTING_REVIEW_REQUIRED');
    }
  });


  return {ok: errors.length === 0, errors: errors};
}

function buildOpenQuestionRows_(context) {
  const header = ['Question ID', 'Title', 'Priority', 'Status', 'Owner', 'Blocker', 'Decisive test', 'Human decision', 'Source URL', 'Last parsed'];
  const id = context.config.OQ_REGISTER_FILE_ID;
  if (!id) return [header, ['CONFIG-MISSING', 'OQ_REGISTER_FILE_ID absent', '', 'FAILED', '', '', '', true, '', iso_(new Date())]];


  let text = '';
  let url = '';
  try {
    const file = DriveApp.getFileById(id);
    url = file.getUrl();
    text = readFileText_(file, file.getMimeType()).text;
  } catch (error) {
    return [header, ['READ-FAILED', error.message, '', 'FAILED', '', '', '', true, '', iso_(new Date())]];
  }


  const rows = [];
  const regex = /(?:^|\n)\s*(OQ-\d{3})\s*(?:—|-|:)\s*([^\n]+)([\s\S]*?)(?=\n\s*OQ-\d{3}\s*(?:—|-|:)|$)/g;
  let match;
  while ((match = regex.exec(text)) !== null) {
    const body = match[3] || '';
    rows.push([
      match[1], match[2].trim(),
      captureField_(body, 'PRIORITY'), captureField_(body, 'STATUS') || 'OPEN',
      captureField_(body, 'OWNER'), captureField_(body, 'BLOCKER'),
      captureField_(body, 'DECISIVE TEST'),
      /HUMAN[-_ ]DECISION|DYLAN/i.test(body), url, iso_(new Date())
    ]);
  }
  if (!rows.length) rows.push(['PARSE-WARNING', 'No OQ headings matched declared parser', '', 'SYNC-UNCERTAIN', '', '', '', true, url, iso_(new Date())]);
  return [header].concat(rows);
}

function validateGeneratedTables_(tables) {
  GP_AUTO.GENERATED_SHEETS.forEach(function(name) {
    const rows = tables[name];
    if (!rows || !rows.length || !rows[0].length) {
      throw new Error('Generated table missing or empty: ' + name);
    }
    const width = rows[0].length;
    rows.forEach(function(row, index) {
      if (row.length !== width) {
        throw new Error(name + ' row ' + index + ' has width ' + row.length + ', expected ' + width);
      }
    });
  });
}


function stageGeneratedSheets_(context, tables) {
  const state = {sheets: {}, commitStarted: false, committed: []};
  GP_AUTO.GENERATED_SHEETS.forEach(function(targetName) {
    const slug = sheetSlug_(targetName);
    const stageName = '_STAGE_' + slug;
    const backupName = '_BACKUP_' + slug;
    const stage = getOrCreateSheet_(context.ss, stageName);
    const backup = getOrCreateSheet_(context.ss, backupName);
    stage.clear({contentsOnly: false});
    backup.clear({contentsOnly: false});
    writeMatrix_(stage, tables[targetName]);
    stage.hideSheet();
    backup.hideSheet();
    state.sheets[targetName] = {stage: stage, backup: backup};
  });
  SpreadsheetApp.flush();
  return state;
}


function validateStagedSheets_(context, stageState, tables) {
  Object.keys(stageState.sheets).forEach(function(targetName) {
    const stage = stageState.sheets[targetName].stage;
    const expected = tables[targetName];
    if (stage.getLastRow() !== expected.length || stage.getLastColumn() !== expected[0].length) {
      throw new Error('Staged sheet dimension mismatch: ' + targetName);
    }
    const actualHeader = stage.getRange(1, 1, 1, expected[0].length).getDisplayValues()[0];
    if (JSON.stringify(actualHeader) !== JSON.stringify(expected[0].map(String))) {
      throw new Error('Staged header mismatch: ' + targetName);
    }
  });
}


function commitGeneratedSheets_(context, stageState, tables) {
  stageState.commitStarted = true;
  GP_AUTO.GENERATED_SHEETS.forEach(function(targetName) {
    const target = requiredSheet_(context.ss, targetName);
    const pair = stageState.sheets[targetName];
    copyWholeSheetData_(target, pair.backup);
    target.clear({contentsOnly: false});
    copyWholeSheetData_(pair.stage, target);
    formatGeneratedSheet_(target, targetName, tables[targetName]);
    stageState.committed.push(targetName);
  });
  SpreadsheetApp.flush();
}


function rollbackGeneratedSheets_(context, stageState) {
  stageState.committed.slice().reverse().forEach(function(targetName) {
    const target = requiredSheet_(context.ss, targetName);
    const backup = stageState.sheets[targetName].backup;
    target.clear({contentsOnly: false});
    copyWholeSheetData_(backup, target);
  });
  SpreadsheetApp.flush();
}


function cleanTransientStageSheets_(context, stageState) {
  // Deliberately retain hidden _STAGE_* and _BACKUP_* sheets as reusable
  // fail-closed infrastructure. They are overwritten only by future staging,
  // are not research sources, and provide immediate post-run inspection.
  Object.keys(stageState.sheets).forEach(function(name) {
    stageState.sheets[name].stage.hideSheet();
    stageState.sheets[name].backup.hideSheet();
  });
}


function copyWholeSheetData_(source, destination) {
  const rows = source.getLastRow();
  const cols = source.getLastColumn();
  if (!rows || !cols) return;
  ensureGridSize_(destination, rows, cols);
  source.getRange(1, 1, rows, cols).copyTo(destination.getRange(1, 1, rows, cols));
}


function writeMatrix_(sheet, rows) {
  ensureGridSize_(sheet, rows.length, rows[0].length);
  sheet.getRange(1, 1, rows.length, rows[0].length).setValues(rows);
}


function formatGeneratedSheet_(sheet, targetName, rows) {
  sheet.setFrozenRows(targetName === 'Dashboard' ? 4 : 1);
  const range = sheet.getRange(1, 1, rows.length, rows[0].length);
  range.setWrap(true).setVerticalAlignment('top');
  if (targetName === 'Dashboard') {
    sheet.getRange(1, 1, 1, Math.min(4, rows[0].length))
      .setFontWeight('bold').setFontSize(14).setBackground('#1f4e78').setFontColor('#ffffff');
    sheet.getRange(4, 1, 1, rows[3].length)
      .setFontWeight('bold').setBackground('#d9eaf7');
  } else {
    sheet.getRange(1, 1, 1, rows[0].length)
      .setFontWeight('bold').setBackground('#1f4e78').setFontColor('#ffffff');
  }
  const maxCols = Math.min(sheet.getLastColumn(), 30);
  for (let c = 1; c <= maxCols; c++) sheet.autoResizeColumn(c);
}


function publishManifestStaged_(context, manifest) {
  assertDriveWriteTarget_(context, context.manifestFolderId);
  const folder = DriveApp.getFolderById(context.manifestFolderId);
  const json = stableStringify_(manifest, 2) + '\n';
  const sha = sha256Text_(json);
  const stagedName = 'WORKSPACE_MASTER_MANIFEST.__STAGING__.' + context.runId + '.json';
  const staged = folder.createFile(Utilities.newBlob(json, 'application/json', stagedName));


  const readBack = staged.getBlob().getDataAsString('UTF-8');
  const parsed = JSON.parse(readBack);
  const readBackHash = sha256Text_(readBack);
  if (parsed.run_id !== context.runId || readBackHash !== sha) {
    throw new Error('Staged manifest read-back validation failed.');
  }


  const priorIds = [];
  const existing = folder.getFilesByName(GP_AUTO.MANIFEST_FILE_NAME);
  while (existing.hasNext()) {
    const prior = existing.next();
    priorIds.push(prior.getId());
    const archiveName = 'WORKSPACE_MASTER_MANIFEST.__VALIDATED__.' +
      formatCompactDate_(prior.getLastUpdated()) + '.' + prior.getId() + '.json';
    prior.setName(archiveName);
  }
  staged.setName(GP_AUTO.MANIFEST_FILE_NAME);


  const routingIndex = buildRoutingIndex_(manifest);
  const routingJson = stableStringify_(routingIndex, 2) + '\n';
  const routingFile = folder.createFile(Utilities.newBlob(
    routingJson, 'application/json', GP_AUTO.ROUTING_INDEX_PREFIX + '.' + context.runId + '.json'));


  return {
    canonicalFileId: staged.getId(),
    canonicalUrl: staged.getUrl(),
    stagedFileId: staged.getId(),
    sha256: sha,
    priorCanonicalIds: priorIds,
    routingIndexId: routingFile.getId(),
    objectCount: manifest.objects.length
  };
}


function buildRoutingIndex_(manifest) {
  const byPath = {};
  manifest.objects.forEach(function(object) {
    if (!byPath[object.tree_path]) byPath[object.tree_path] = [];
    byPath[object.tree_path].push({
      drive_id: object.drive_id,
      artifact_id: object.artifact_id,
      title: object.title,
      source_url: object.source_urls[0],
      extraction_class: object.extraction_class,
      status: object.status
    });
  });
  return {
    schema_version: 'TREE-ROUTING-INDEX-v1.0',
    run_id: manifest.run_id,
    generated_at: manifest.generated_at,

function readClosureState_(ss) {
  const sheet = ss.getSheetByName('Closure Log');
  if (!sheet || sheet.getLastRow() < 2) return {byArtifactId: {}, byTitle: {}, byDriveId: {}};
  const values = sheet.getDataRange().getDisplayValues();
  const header = values[0].map(function(x) { return normalizeMetadataKey_(x); });
  const byArtifactId = {};
  const byTitle = {};
  const byDriveId = {};


  for (let r = 1; r < values.length; r++) {
    const row = {};
    header.forEach(function(key, c) { row[key] = values[r][c]; });
    const state = firstNonEmpty_(row.TERMINAL_LABEL, row.STATUS, row.DECISION_STATE, row.CLOSURE_STATE);
    const entry = {state: state || 'UNVERIFIED', row: r + 1, raw: row};
    const artifactId = firstNonEmpty_(row.ARTIFACT_ID, row.ITEM, row.OBJECT_ID);
    const title = firstNonEmpty_(row.TITLE, row.OBJECT, row.ITEM);
    const driveId = firstNonEmpty_(row.DRIVE_ID, row.FILE_ID);
    if (artifactId) byArtifactId[artifactId] = entry;
    if (title) byTitle[String(title).toLowerCase()] = entry;
    if (driveId) byDriveId[driveId] = entry;
  }
  return {byArtifactId: byArtifactId, byTitle: byTitle, byDriveId: byDriveId};
}


function lookupClosure_(closureState, artifactId, title, driveId) {
  return closureState.byDriveId[driveId] ||
    (artifactId ? closureState.byArtifactId[artifactId] : null) ||
    closureState.byTitle[String(title).toLowerCase()] || {state: ''};
}


function applyExtractionEligibility_(record, closure) {
  const title = record.title.toLowerCase();
  const classText = String(record.class).toUpperCase();
  const statusText = String(record.status).toUpperCase();
  const closureText = String(record.closure_state).toUpperCase();


  if (/DATA|SCRIPT|SOURCE|CODE|PYTHON|EXECUTABLE|RECEIPT|FIXTURE/.test(classText + ' ' + title)) {
    record.extraction_class = 'INSTRUMENT';
  } else if (/KILLED|SUPERSEDED|DUPLICATE|HISTORICAL|PROVENANCE/.test(statusText + ' ' + closureText)) {
    record.extraction_class = 'PROVENANCE-ONLY';
  } else if (isTerminalClosure_(closureText) &&
             record.human_decision === true &&
             String(record.canonical_impact).toUpperCase() === 'APPROVED' &&
             record.authority && record.authority.toLowerCase() !== 'none' &&
             record.dependencies.length && record.evidence_chain.length) {
    record.extraction_class = 'REFEREE-READY';
  } else {
    record.extraction_class = 'ACTIVE';
  }


  if (!GP_AUTO.ALLOWED_EXTRACTION_CLASSES.includes(record.extraction_class)) {
    record.validation_errors.push('INVALID_EXTRACTION_CLASS');
    record.extraction_class = 'ACTIVE';
  }
}


function validateObjectCard_(record) {
  if (!record.artifact_id) record.validation_errors.push('MISSING_ARTIFACT_ID');
  if (!record.class) record.validation_errors.push('MISSING_CLASS');
  if (!record.version) record.validation_errors.push('MISSING_VERSION');
  if (record.organization === 'UNDECLARED') record.validation_errors.push('MISSING_ORGANIZATION');
  if (record.measure === 'UNDECLARED') record.validation_errors.push('MISSING_MEASURE');
  if (record.model_layer === 'UNDECLARED') record.validation_errors.push('MISSING_MODEL_LAYER');
  if (record.frame === 'UNDECLARED') record.validation_errors.push('MISSING_FRAME');
  if (record.human_decision === null) record.validation_errors.push('MISSING_HUMAN_DECISION');
  if (record.sync_status !== 'COMPLETE') record.validation_errors.push(record.sync_status);
  if (record.extraction_class === 'REFEREE-READY' && record.validation_errors.length) {
    record.extraction_class = 'ACTIVE';
    record.validation_errors.push('REFEREE_READY_FAIL_CLOSED');
  }
}


function routeTreePath_(record) {
  const haystack = [record.title, record.class, record.topics.join(' '), record.object_tags.join(' '),
    record.artifact_id].join(' ').toLowerCase();
  if (/governance|protocol|operator rule|ballot|consensus/.test(haystack)) return TREE_PATHS.GOVERNANCE;
  if (/q0_master|q0 ledger|canonical master|consolidated spine/.test(haystack)) return TREE_PATHS.CANONICAL;
  if (/saddle.exit|sink.capture|endpoint block|isolating block/.test(haystack)) return TREE_PATHS.SADDLE;
  if (/invariant.strip|flow comparison|transit|dynamics/.test(haystack)) return TREE_PATHS.STRIP;
  if (/palm|normalizer|remainder|gaussian regularity|degree.five/.test(haystack)) return TREE_PATHS.PALM;
  if (/p0\.2.*validation|validation certificate|independent audit/.test(haystack)) return TREE_PATHS.P02_CERT;
  if (/law.specific q|q machine|machine snapshot|root schema/.test(haystack)) return TREE_PATHS.MACHINE;
  if (/negative test|falsif|mutation battery|fail.closed verifier/.test(haystack)) return TREE_PATHS.NEGATIVE;
  if (/validation receipt|schema sign.off/.test(haystack)) return TREE_PATHS.MACHINE_RECEIPT;
  if (/exact pair|elder.rule|critical set|pair topology/.test(haystack)) return TREE_PATHS.EXACT_PAIR;
  if (/power|sample.size|cluster.bootstrap|simulation plan/.test(haystack)) return TREE_PATHS.POWER;
  if (/connectivity|four.neighbor|4.connectivity|rebase/.test(haystack)) return TREE_PATHS.CONNECTIVITY;
  if (/live register|spreadsheet sync|gp-reg-032/.test(haystack)) return TREE_PATHS.REGISTER;
  if (/navigation|whole.drive map|read first|dashboard/.test(haystack)) return TREE_PATHS.NAVIGATION;
  if (/cross.model|ledger|review ledger/.test(haystack)) return TREE_PATHS.LEDGER;
  if (/manifest|automation|apps script|gp-auto-034/.test(haystack)) return TREE_PATHS.MANIFEST;
  return TREE_PATHS.UNROUTED;
}

function parseList_(value) {
  if (!value) return [];
  return String(value).split(/[;,|]/).map(function(x) { return x.trim(); }).filter(Boolean);
}


function parseBoolean_(value) {
  const x = String(value == null ? '' : value).trim().toUpperCase();
  if (x === 'TRUE' || x === 'YES' || x === '1') return true;
  if (x === 'FALSE' || x === 'NO' || x === '0') return false;
  return null;
}


function parseNullableBoolean_(value) {
  return parseBoolean_(value);
}

function normalizeText_(text) {
  return String(text || '')
    .replace(/\r\n?/g, '\n')
    .replace(/[ \t]+$/gm, '')
    .replace(/\n+$/g, '') + (text ? '\n' : '');
}


function sha256Text_(text) {
  return sha256Bytes_(Utilities.newBlob(String(text), 'text/plain').getBytes());
}


function sha256Bytes_(bytes) {
  const digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, bytes);
  return digest.map(function(byte) {
    const value = byte < 0 ? byte + 256 : byte;
    return ('0' + value.toString(16)).slice(-2);
  }).join('');
}


function stableStringify_(value, space) {
  return JSON.stringify(sortRecursively_(value), null, space || 0);
}


function sortRecursively_(value) {
  if (Array.isArray(value)) return value.map(sortRecursively_);
  if (value && typeof value === 'object') {
    const out = {};
    Object.keys(value).sort().forEach(function(key) { out[key] = sortRecursively_(value[key]); });
    return out;
  }
  return value;
}

function firstNonEmpty_() {
  for (let i = 0; i < arguments.length; i++) {
    if (arguments[i] !== null && arguments[i] !== undefined && String(arguments[i]).trim() !== '') {
      return String(arguments[i]).trim();
    }
  }
  return '';
}


function captureField_(body, label) {
  const escaped = label.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  const regex = new RegExp('(?:^|\\n)\\s*' + escaped + '\\s*:\\s*([^\\n]+)', 'i');
  const match = String(body).match(regex);
  return match ? match[1].trim() : '';
}
