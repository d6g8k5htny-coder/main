def det_fraction(A: List[List[F]]) -> F:
    A = [list(map(F, row)) for row in A]
    n = len(A)
    det = F(1)
    for i in range(n):
        pivot = next((k for k in range(i,n) if A[k][i] != 0), None)
        if pivot is None:
            return F(0)
        if pivot != i:
            A[i], A[pivot] = A[pivot], A[i]
            det = -det
        p = A[i][i]
        det *= p
        for k in range(i+1,n):
            f = A[k][i]/p
            for j in range(i+1,n):
                A[k][j] -= f*A[i][j]
    return det
