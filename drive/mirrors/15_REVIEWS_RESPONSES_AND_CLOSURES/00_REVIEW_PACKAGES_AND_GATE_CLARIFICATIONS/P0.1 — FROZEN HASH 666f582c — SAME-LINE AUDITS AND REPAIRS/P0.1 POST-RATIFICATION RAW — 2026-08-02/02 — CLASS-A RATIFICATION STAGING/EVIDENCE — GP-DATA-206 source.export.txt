def odd_double_factorial(n: int) -> int:
    """Return n!! for odd n >= -1, with (-1)!! = 1."""
    if n in (-1, 0, 1):
        return 1
    out = 1
    for k in range(n, 0, -2):
        out *= k
    return out




def planar_derivative_variance(alpha: tuple[int, int]) -> int:
    """Planar BF variance of partial^alpha f at one point."""
    ax, ay = alpha
    return odd_double_factorial(2 * ax - 1) * odd_double_factorial(2 * ay - 1)
