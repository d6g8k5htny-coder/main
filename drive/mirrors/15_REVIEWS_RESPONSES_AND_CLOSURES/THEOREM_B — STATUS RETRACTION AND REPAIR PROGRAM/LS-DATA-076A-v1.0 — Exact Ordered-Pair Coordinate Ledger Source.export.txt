# Exact ordered-pair coordinate and power-count ledger.


def det_fraction(matrix):
    a = [row[:] for row in matrix]
    n = len(a)
    det = F(1)
    for col in range(n):
        pivot = next(i for i in range(col, n) if a[i][col] != 0)
        if pivot != col:
            a[col], a[pivot] = a[pivot], a[col]
            det = -det
        p = a[col][col]
        det *= p
        for j in range(col, n):
            a[col][j] /= p
        for i in range(col + 1, n):
            q = a[i][col]
            if q:
                for j in range(col, n):
                    a[i][j] -= q * a[col][j]
    return det


for d in range(1, 9):

        A[d + i][i] = 1
        A[d + i][d + i] = F(1, 2)
    assert det_fraction(A) == 1
