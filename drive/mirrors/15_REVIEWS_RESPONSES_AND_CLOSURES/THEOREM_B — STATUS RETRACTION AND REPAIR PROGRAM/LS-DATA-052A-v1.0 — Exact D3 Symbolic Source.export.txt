import sympy as sp


y, z, u, v = sp.symbols('y z u v', positive=True)


# Projective triangle integrand from LS-DER-052 Eq. (24).
num = z * (3*y**2-z**2) * (1-y)**2 * ((2+y)**2-3*z**2)**2
den = (1+y**2+z**2)**6


# Integrate z exactly using u=z^2, so z dz=du/2.
A = 1+y**2
num_u = sp.expand(sp.Rational(1,2)*(3*y**2-u)*(1-y)**2*((2+y)**2-3*u)**2)
num_v = sp.Poly(sp.expand(num_u.subs(u, v-A)), v)
primitive = sum(
    -num_v.coeff_monomial(v**k) * v**(k-5) / (5-k)
    for k in range(4)
)
inner = sp.factor(primitive.subs(v, 1+4*y**2)-primitive.subs(v, 1+y**2))


# Exact outer integral.
outer_primitive = sp.integrate(inner, y, risch=True)
triangle_raw = sp.simplify(outer_primitive.subs(y, 1)-sp.limit(outer_primitive, y, 0, dir='+'))
triangle_expected = sp.Rational(-57,640) + sp.Rational(5,64)*(sp.atan(sp.Rational(2,3))+sp.atan(sp.Rational(11,3)))
assert sp.simplify(triangle_raw-triangle_expected) == 0


# atan(2/3)+atan(11/3)=pi-atan(3); exact quadrant follows because the sum lies in (pi/2,pi).
angle_sum = sp.atan(sp.Rational(2,3))+sp.atan(sp.Rational(11,3))
assert abs(sp.N(angle_sum-(sp.pi-sp.atan(3)), 80)) < sp.Float('1e-70')
triangle = sp.simplify(sp.Rational(-57,640)+sp.Rational(5,64)*(sp.pi-sp.atan(3)))


J3 = sp.simplify(sp.Rational(40960,9)*triangle)

assert sp.simplify(D3-D3_expected) == 0
