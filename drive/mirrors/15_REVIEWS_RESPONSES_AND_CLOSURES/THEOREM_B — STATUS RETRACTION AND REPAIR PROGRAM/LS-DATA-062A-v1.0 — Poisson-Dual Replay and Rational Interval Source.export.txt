def image_moments(L, cutoff=2):
    Z = mp.mpf("0")
    sums = {0: mp.mpf("0"), 2: mp.mpf("0"), 4: mp.mpf("0"), 6: mp.mpf("0")}
    for n in range(-cutoff, cutoff + 1):
        a = n * L
        e = mp.exp(-a * a / 2)
        Z += e
        vals = {
            0: mp.mpf(1),
            2: a*a - 1,
            4: a**4 - 6*a*a + 3,
            6: a**6 - 15*a**4 + 45*a*a - 15,
        }
        for k in sums:
            sums[k] += vals[k] * e
    return {k: sums[k] / Z for k in sums}


def poisson_moments(L, cutoff=170):
    alpha = 2 * mp.pi**2 / L**2
    scale = 2 * mp.pi / L
    Z = mp.mpf("0")
    sums = {0: mp.mpf("0"), 2: mp.mpf("0"), 4: mp.mpf("0"), 6: mp.mpf("0")}
    for m in range(-cutoff, cutoff + 1):
        e = mp.exp(-alpha * m * m)
        Z += e
        for k in sums:
            sums[k] += mp.mpf(m)**k * e
    return {
        0: sums[0] / Z,
        2: -scale**2 * sums[2] / Z,
        4: scale**4 * sums[4] / Z,
        6: -scale**6 * sums[6] / Z,
    }


def conv_dict(A, B):
    out = {}
    for a, ca in A.items():
        for b, cb in B.items():
            g = tuple(a[i] + b[i] for i in range(3))
            out[g] = out.get(g, mp.mpf("0")) + ca * cb
    return out


def directional(v):
    return {(1,0,0): v[0], (0,1,0): v[1], (0,0,1): v[2]}


def op_from_vectors(vecs):
    d = {(0,0,0): mp.mpf(1)}
    for v in vecs:
        d = conv_dict(d, directional(v))
    return {k: v for k, v in d.items() if v != 0}


def Kder(gamma, moments):
    val = mp.mpf(1)
    for n in gamma:
        if n % 2:
            return mp.mpf(0)
        val *= moments[n]
    return val


def op_order(A):
    return sum(next(iter(A.keys())))


def cov_ops(A, B, moments):
    sign = (-1)**op_order(B)
    total = mp.mpf(0)
    for a, ca in A.items():
        for b, cb in B.items():
            g = tuple(a[i] + b[i] for i in range(3))
            total += ca * cb * sign * Kder(g, moments)
    return total


def cov_matrix(A, B, moments):
    return mp.matrix([[cov_ops(a, b, moments) for b in B] for a in A])


def frame(z, phi):
    r = mp.sqrt(1 - z*z)
    cp, sp = mp.cos(phi), mp.sin(phi)
    return (r*cp, r*sp, z), (-sp, cp, mp.mpf(0)), (-z*cp, -z*sp, r)


def contact_blocks(z, phi, moments):
    t, n1, n2 = frame(z, phi)
    u = {k: -v/12 for k, v in op_from_vectors([t,t,t]).items()}
    g = [op_from_vectors([t]), op_from_vectors([n1]), op_from_vectors([n2])]
    h = [op_from_vectors([t,t]), op_from_vectors([t,n1]), op_from_vectors([t,n2])]
    q = [op_from_vectors([n1,n1]), op_from_vectors([n1,n2]), op_from_vectors([n2,n2])]
    Sg = cov_matrix(g, g, moments)
    su = cov_ops(u, u, moments)
    cug = cov_matrix([u], g, moments)
    sigma2 = su - (cug * (Sg**-1) * cug.T)[0]
    Sh = cov_matrix(h, h, moments)
    Sq = cov_matrix(q, q, moments)
    Cqh = cov_matrix(q, h, moments)
    Gamma = Sq - Cqh * (Sh**-1) * Cqh.T
    return Sg, Sh, sigma2, Gamma


GL_CACHE = {}
def gauss_legendre(n, a, b):
    key = (n, mp.nstr(a, 40), mp.nstr(b, 40))
    if key not in GL_CACHE:
        x, w = mp.gauss_quadrature(n, "legendre")
        GL_CACHE[key] = (
            [(b-a)*xx/2 + (a+b)/2 for xx in x],
            [(b-a)*ww/2 for ww in w],
        )
    return GL_CACHE[key]

def D_spectral(Gamma, inner_n):
    C = mp.cholesky(Gamma)
    B = C.T * H * C
    vals, _ = mp.eigsy(B)
    vals = sorted([vals[i] for i in range(3)])
    nu, mu, lam = -vals[0], -vals[1], vals[2]
    nodes, weights = gauss_legendre(inner_n, mp.mpf(0), mp.pi/2)
    integral = mp.mpf(0)
    for phi, weight in zip(nodes, weights):
        hh = mu*mp.cos(phi)**2 + nu*mp.sin(phi)**2
        integral += weight * hh**(mp.mpf(5)/2) / mp.sqrt(lam + hh)
    poly = (mp.mpf("1.5")*lam**2 - lam*(mu+nu)
            + mp.mpf("0.5")*(3*mu**2 + 2*mu*nu + 3*nu**2))
    return poly - (8/mp.pi)*integral


def D_direct_triple(lam, mu, nu):
    # Independent cap integration: 3-point Gauss-Legendre is exact in x
    # because the cap integrand is a quartic polynomial.
    phis, pweights = gauss_legendre(96, mp.mpf(0), 2*mp.pi)
    total = mp.mpf(0)
    for phi, pw in zip(phis, pweights):
        h = mu*mp.cos(phi)**2 + nu*mp.sin(phi)**2
        a = mp.sqrt(h/(lam+h))
        xs, xweights = gauss_legendre(3, a, mp.mpf(1))
        cap = 2*sum(w*((lam+h)*x*x-h)**2 for x, w in zip(xs, xweights))
        total += pw*cap
    return (15/(4*mp.pi)*total)/2


def D_reduced_triple(lam, mu, nu):
    phis, pweights = gauss_legendre(96, mp.mpf(0), 2*mp.pi)
    total = mp.mpf(0)
    for phi, pw in zip(phis, pweights):
        h = mu*mp.cos(phi)**2 + nu*mp.sin(phi)**2
        total += pw*(3*lam**2 - 4*lam*h + 8*h**2
                     - 8*h**(mp.mpf(5)/2)/mp.sqrt(lam+h))
    return total/(4*mp.pi)
