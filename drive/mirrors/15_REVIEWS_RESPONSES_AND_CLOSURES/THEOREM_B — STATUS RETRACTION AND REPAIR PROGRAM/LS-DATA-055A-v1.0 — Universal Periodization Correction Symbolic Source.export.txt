import sympy as sp


# Exact first-image periodization correction for the d-dimensional
# normalized cubic Bargmann-Fock field.
d, L = sp.symbols('d L', positive=True)


# Leading radial-moment changes divided by q_L=exp(-L^2/2).
dM2 = -2*d*L**2
dM4 = 2*d*(L**4 - 2*(d+2)*L**2)
dM6 = 2*d*(-L**6 + 3*(d+4)*L**4 - 3*(d+2)*(d+4)*L**2)


# Isotropic directional moments a,m4,chi.
da = sp.simplify(dM2/d)
dm4 = sp.simplify(3*dM4/(d*(d+2)))
dchi = sp.simplify(15*dM6/(d*(d+2)*(d+4)))


# At the planar BF point: a=1, m4=3, chi=15, Omega=6.
dOmega = sp.simplify(15*da + dchi - 6*dm4)


# LS-DER-050: c_d is proportional to
# Omega^(2/3) m4^((d-2)/2) a^(-(3d+4)/6).
dlogc = sp.factor(sp.simplify(
    dOmega/9 + (d-2)*dm4/6 - (3*d+4)*da/6
))


expected = sp.factor(
    -d*L**2 + (d+4)*L**4/(d+2)
    - sp.Rational(10,3)*L**6/((d+2)*(d+4))
)
assert sp.simplify(dlogc-expected) == 0


# Recovery of LS-DER-045 in d=2.
p2 = sp.factor(dlogc.subs(d,2))
expected2 = -L**2*(5*L**4-54*L**2+72)/36
assert sp.simplify(p2-expected2) == 0


# New d=3 correction polynomial.
p3 = sp.factor(dlogc.subs(d,3))
expected3 = -sp.Rational(2,21)*L**6 + sp.Rational(7,5)*L**4 - 3*L**2
assert sp.simplify(p3-expected3) == 0


q24 = sp.exp(-288)
rel24_d3 = sp.simplify(p3.subs(L,24)*q24)


c3_bf = (
    2**sp.Rational(1,6)*3**sp.Rational(1,3)
    *(29*sp.sqrt(6)-36)*sp.gamma(sp.Rational(1,6))
    /(432*sp.pi**sp.Rational(5,2))
)
abs24_d3 = sp.simplify(c3_bf*rel24_d3)


print('dM2/q =', sp.factor(dM2))
print('dM4/q =', sp.factor(dM4))
print('dM6/q =', sp.factor(dM6))
print('da/q =', sp.factor(da))
print('dm4/q =', sp.factor(dm4))
print('dchi/q =', sp.factor(dchi))
print('dOmega/q =', sp.factor(dOmega))
print('general_relative_polynomial =', dlogc)

print('d3_polynomial =', p3)

print('d3_L24_absolute_leading =', sp.N(abs24_d3, 70))
