#!/usr/bin/env python3
"""Exact symbolic checks for LS-DER-047.


Derives the d=1,2,3 planar Bargmann-Fock fold-contact coefficients from the
GOE representation in LS-DER-047.  No Monte Carlo is used.
"""
from __future__ import annotations


import sympy as sp




def main() -> None:
    pi = sp.pi
    gamma = sp.gamma


    # Universal kappa integral.
    kappa_integral = sp.Rational(1, 2) * 3 ** sp.Rational(7, 6) * gamma(sp.Rational(7, 6))
    assert sp.simplify(kappa_integral - 3 ** sp.Rational(7, 6) * gamma(sp.Rational(1, 6)) / 12) == 0


    # I_0: transverse dimension n=0.
    I0 = 2 * sp.sqrt(pi / 3)


    # I_1: G_1 ~ N(0,2).  Exact b/q Gaussian integral.
    b, q = sp.symbols("b q", real=True)
    I1 = sp.integrate(
        sp.exp(-sp.Rational(3, 4) * b**2)
        * sp.integrate(
            (q - b) ** 2 * sp.exp(-q**2 / 4) / sp.sqrt(4 * pi),
            (q, -sp.oo, b),
        ),
        (b, -sp.oo, sp.oo),
    )
    I1 = sp.simplify(I1)
    assert sp.simplify(I1 - 8 * sp.sqrt(3 * pi) / 9) == 0


    # I_2: diagonalize a 2x2 GOE matrix as eigenvalues m +/- s,
    # m~N(0,1), s Rayleigh with density s exp(-s^2/2).
    u, s = sp.symbols("u s", positive=True)
    rayleigh_inner = sp.integrate(
        s * sp.exp(-s**2 / 2) * (u**2 - s**2) ** 2,
        (s, 0, u),
    )
    rayleigh_inner = sp.simplify(rayleigh_inner)
    assert sp.simplify(rayleigh_inner - (u**4 - 4 * u**2 + 8 - 8 * sp.exp(-u**2 / 2))) == 0


    I2 = sp.sqrt(sp.Rational(2, 5)) * sp.integrate(
        sp.exp(-sp.Rational(3, 10) * u**2) * rayleigh_inner,
        (u, 0, sp.oo),
    )
    I2 = sp.simplify(I2)
    I2_closed = sp.sqrt(pi) * (29 * sp.sqrt(3) - 18 * sp.sqrt(2)) / 9
    assert sp.simplify(I2 - I2_closed) == 0


    def sphere_area(d: int) -> sp.Expr:
        return 2 * pi ** sp.Rational(d, 2) / gamma(sp.Rational(d, 2))


    def coefficient(d: int, I: sp.Expr) -> sp.Expr:
        return sp.simplify(
            sphere_area(d)
            * gamma(sp.Rational(1, 6))
            * I
            / (2 ** sp.Rational(4, 3) * 3 ** sp.Rational(2, 3) * (2 * pi) ** (d + 1))
        )


    c1 = coefficient(1, I0)
    c2 = coefficient(2, I1)
    c3 = coefficient(3, I2)


    c1_closed = 2 ** sp.Rational(2, 3) * 3 ** sp.Rational(5, 6) * gamma(sp.Rational(1, 6)) / (36 * pi ** sp.Rational(3, 2))
    c2_closed = 2 ** sp.Rational(2, 3) * 3 ** sp.Rational(5, 6) * gamma(sp.Rational(1, 6)) / (54 * pi ** sp.Rational(3, 2))
    c3_closed = (
        2 ** sp.Rational(1, 6)
        * 3 ** sp.Rational(1, 3)
        * (29 * sp.sqrt(6) - 36)
        * gamma(sp.Rational(1, 6))
        / (432 * pi ** sp.Rational(5, 2))
    )


    assert sp.simplify(c1 - c1_closed) == 0
    assert sp.simplify(c2 - c2_closed) == 0
    assert sp.simplify(c3 - c3_closed) == 0


    print("kappa_integral =", kappa_integral)
    print("I0 =", I0)
    print("I1 =", I1)
    print("I2 =", I2)
    print("c1 =", c1_closed)
    print("c1_decimal =", sp.N(c1_closed, 60))
    print("c2 =", c2_closed)
    print("c2_decimal =", sp.N(c2_closed, 60))
    print("c3 =", c3_closed)
    print("c3_decimal =", sp.N(c3_closed, 60))
    print("c1/c2 =", sp.simplify(c1_closed / c2_closed))
    print("all_symbolic_assertions = PASS")




if __name__ == "__main__":
    main()
