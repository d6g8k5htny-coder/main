# LS-DATA-071A-v1.0
# Combined exact pre-review ledger for LS-DER-071 Gate A and the
# degree-nine correction in Gate B. Pure integer/Fraction arithmetic.


DIM = 3
JET_CONTRACTION = 729
ENTRY_UPPER = 16
OP_UPPER = DIM * ENTRY_UPPER
INV_UPPER = 2
L1 = DIM * JET_CONTRACTION
INV_D1 = INV_UPPER**2 * L1
INV_D2 = 2 * INV_UPPER**3 * L1**2
C = OP_UPPER


SCHUR_D1 = (
    JET_CONTRACTION
    + 2 * L1 * INV_UPPER * C
    + C**2 * INV_D1
)
SCHUR_D2 = (
    2 * L1**2 * INV_UPPER
    + 4 * L1 * INV_D1 * C
    + C**2 * INV_D2
)
S1 = 30_000_000
S2 = 200_000_000_000
assert SCHUR_D1 < S1
assert SCHUR_D2 < S2


# f=(det Sigma_g det Sigma_h)^(-1/2).

LOGF_D1 = 3 * INV_UPPER * L1
LOGF_D2 = 3 * INV_UPPER**2 * L1**2
F1 = F0 * LOGF_D1
F2 = F0 * (LOGF_D1**2 + LOGF_D2)
assert F1 < 200_000
assert F2 < 3_000_000_000

assert D1_GAMMA < 20_000
assert D2_GAMMA < 2_000_000

assert DJ2 < 3_000_000_000_000_000_000_000
