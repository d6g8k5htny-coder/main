import sympy as sp


pi = sp.pi
d = sp.symbols('d', integer=True, positive=True)
n = d-1
m = d*(d-1)/2


# Universal outer constant after Kac-Rice, lifetime pushforward,
# exact kappa integration, and the two d-dimensional zero densities.
B_d = (
    3*2**(sp.Rational(5,3)-d)
    *sp.gamma(sp.Rational(7,6))
    /pi**(d+sp.Rational(1,2))
)


# Radial factor for a degree-2n homogeneous cone weight in R^m.
R_d = (
    2**(d-2)
    *sp.gamma((d-1)*(d+4)/4)
    /pi**(d*(d-1)/4)
)


C_nested = sp.simplify(B_d*R_d)
C_nested_expected = (
    3*2**(-sp.Rational(1,3))
    *sp.gamma(sp.Rational(7,6))
    *sp.gamma((d-1)*(d+4)/4)
    /pi**((d+1)*(d+2)/4)
)
assert sp.simplify(C_nested-C_nested_expected) == 0


# Planar Bargmann-Fock recovery in d=2 and d=3.
D1 = sp.Rational(4,3)
D2 = sp.Rational(29,6)-sp.sqrt(6)


B2 = sp.simplify(B_d.subs(d,2))
B3 = sp.simplify(B_d.subs(d,3))


c2 = sp.simplify(
    B2*(2*pi)*sp.Rational(1,24)**sp.Rational(2,3)
    *D1/sp.sqrt(3)
)
c3 = sp.simplify(
    B3*(4*pi)*sp.Rational(1,24)**sp.Rational(2,3)
    *D2/sp.sqrt(3)
)


c2_ref = (
    2**sp.Rational(2,3)*3**sp.Rational(5,6)
    *sp.gamma(sp.Rational(1,6))/(54*pi**sp.Rational(3,2))
)
c3_ref = (
    2**sp.Rational(1,6)*3**sp.Rational(1,3)
    *(29*sp.sqrt(6)-36)*sp.gamma(sp.Rational(1,6))
    /(432*pi**sp.Rational(5,2))
)
assert sp.simplify(c2-c2_ref) == 0
assert sp.simplify(c3-c3_ref) == 0


# Low-dimensional cone radial checks.
assert sp.simplify(R_d.subs(d,2)-sp.Rational(1,2)) == 0
assert sp.simplify(R_d.subs(d,3)-sp.Rational(15,4)/pi) == 0


print('B_d =', B_d)
print('R_d =', R_d)
print('nested_prefactor =', C_nested_expected)
print('d2_recovery =', c2)
print('d3_recovery =', c3)
print('ALL_ASSERTIONS_PASS')
