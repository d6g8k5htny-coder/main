"""
DQ-050 exact symbolic verifier for TB-G3B endpoint determinants and density-frame algebra.
Uses SymPy exact arithmetic only. No numerical sampling enters any proof claim.
"""
import json
import sympy as sp


X,Y,c,s,eta,alpha,kappa,K = sp.symbols(
    "X Y c s eta alpha kappa K", nonzero=True
)
a,Q,w,z = sp.symbols("a Q w z")
P = (
    kappa*(X**3/sp.Integer(3)-eta**2*X/sp.Integer(4)-eta**3/sp.Integer(12))
    + a/sp.Integer(2)*(X**2-eta**2/sp.Integer(4))*Y
    + Q/sp.Integer(2)*Y**2
    + w/sp.Integer(2)*X*Y**2
    + z/sp.Integer(6)*Y**3
)
PX, PY = sp.diff(P,X), sp.diff(P,Y)
H = sp.hessian(P,(X,Y))
eqs = [
    sp.Eq(P.subs({X:c,Y:s}), -alpha*kappa*eta**3/sp.Integer(6)),
    sp.Eq(PX.subs({X:c,Y:s}), 0),
    sp.Eq(PY.subs({X:c,Y:s}), 0),
]
sol = sp.solve(eqs,[a,Q,z],dict=True,simplify=True)[0]
Wexpr = (K+4*c**2+4*c*eta+eta**2)/(2*s**2)
def simp(expr):
    return sp.factor(sp.together(expr.subs(sol).subs({w:kappa*Wexpr})))


detM = simp(H.det().subs({X:-eta/sp.Integer(2),Y:0}))
detS = simp(H.det().subs({X:+eta/sp.Integer(2),Y:0}))
detX = simp(H.det().subs({X:c,Y:s}))


BM = 64*alpha*c**2*eta**2-K**2
BS = 64*(1-alpha)*c**2*eta**2-(K+8*c*eta)**2
BX = (K+8*alpha*c*eta)**2+64*alpha*(1-alpha)*c**2*eta**2


assert sp.simplify(detM-kappa**2*eta**2*BM/(64*c**2*s**2)) == 0
assert sp.simplify(detS-kappa**2*eta**2*BS/(64*c**2*s**2)) == 0
assert sp.simplify(detX+kappa**2*eta**2*BX/(64*c**2*s**2)) == 0
prod = sp.factor(detM*detS*detX)
prod_expected = -kappa**6*eta**6*BM*BS*BX/(64**3*c**6*s**6)
assert sp.simplify(prod-prod_expected) == 0


for eq in eqs:
    assert sp.simplify((eq.lhs-eq.rhs).subs(sol)) == 0


Abar=sp.symbols("Abar")
transverse={}
for sign in (1,-1):
    tsol=sp.solve([
        sp.Eq(P.subs({X:0,Y:sign}),-alpha*kappa*eta**3/sp.Integer(6)),
        sp.Eq(PX.subs({X:0,Y:sign}),0),
        sp.Eq(PY.subs({X:0,Y:sign}),0),
    ],[w,Q,z],dict=True,simplify=True)[0]
    local={a:sign*kappa*Abar}
    dM=sp.factor(H.det().subs(tsol).subs(local).subs({X:-eta/2,Y:0}))
    dS=sp.factor(H.det().subs(tsol).subs(local).subs({X:+eta/2,Y:0}))
    dX=sp.factor(H.det().subs(tsol).subs(local).subs({X:0,Y:sign}))
    eM=kappa**2*eta**2*(4*alpha*eta**2-(Abar+eta)**2)/4
    eS=kappa**2*eta**2*(4*(1-alpha)*eta**2-(Abar-eta)**2)/4
    eX=-kappa**2*eta**2*((Abar+(1-2*alpha)*eta)**2
                          +4*alpha*(1-alpha)*eta**2)/4
    assert sp.simplify(dM-eM)==0
    assert sp.simplify(dS-eS)==0
    assert sp.simplify(dX-eX)==0
    transverse[str(sign)]={"detM":str(eM),"detS":str(eS),"detX":str(eX)}


cov11=s**2*(3*c**4+3*c**2*s**2+s**4)/24
cov12=-c*s**2*(2*c**2+s**2)/4
cov22=s**2*(4*c**2+s**2)/2
cov33=2*s**2
det2=sp.factor(cov11*cov22-cov12**2)
det3=sp.factor(det2*cov33)
cond1=sp.factor(det2/cov22)
assert sp.simplify(det2-s**6*(c**2+s**2)*(3*c**2+s**2)/48)==0
assert sp.simplify(det3-s**8*(c**2+s**2)*(3*c**2+s**2)/24)==0
assert sp.simplify(
    cond1-s**4*(c**2+s**2)*(3*c**2+s**2)/(24*(4*c**2+s**2))

    "replace_alpha_by_1_minus_alpha_at_M":
        detM-kappa**2*eta**2*(64*(1-alpha)*c**2*eta**2-K**2)/(64*c**2*s**2),
    "drop_kappa6_product": prod+eta**6*BM*BS*BX/(64**3*c**6*s**6),
    "eta4_product": prod+ kappa**6*eta**4*BM*BS*BX/(64**3*c**6*s**6),
