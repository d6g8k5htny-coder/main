#!/usr/bin/env python3
"""Exact symbolic reduction for the general isotropic 2D fold coefficient."""
from __future__ import annotations


import sympy as sp




def main() -> None:
    a, m4, chi = sp.symbols("a m4 chi", positive=True)
    pi = sp.pi
    Delta = m4 - a**2
    Omega = a * chi - m4**2


    # Even contact block for (f, f_11).
    Sigma_even = sp.Matrix([[1, -a], [-a, m4]])
    assert sp.simplify(Sigma_even.det() - Delta) == 0


    # Odd axial block for (-f_111/12, f_1).
    Sigma_odd = sp.Matrix([[chi / 144, m4 / 12], [m4 / 12, a]])
    assert sp.simplify(Sigma_odd.det() - Omega / 144) == 0


    # Transverse curvature q=f_22 conditioned on f=b, f_11=0.
    v = sp.Matrix([[-a, m4 / 3]])
    e_b = sp.Matrix([1, 0])
    mu_per_b = sp.simplify((v * Sigma_even.inv() * e_b)[0])
    sigma2 = sp.simplify(m4 - (v * Sigma_even.inv() * v.T)[0])
    assert sp.simplify(mu_per_b + 2 * a * m4 / (3 * Delta)) == 0
    assert sp.simplify(sigma2 - 4 * m4 * (2 * m4 - 3 * a**2) / (9 * Delta)) == 0


    # Under the normalized b-weight exp[-m4 b^2/(2 Delta)], b has variance Delta/m4.
    q_marginal_variance = sp.simplify(mu_per_b**2 * Delta / m4 + sigma2)
    assert sp.simplify(q_marginal_variance - 8 * m4 / 9) == 0


    b_integral = sp.simplify(
        sp.sqrt(2 * pi * Delta / m4) * q_marginal_variance / 2
    )


    # Pin-density constant, kappa integral, lifetime Jacobian, and angular factor.
    D, O = sp.symbols("D O", positive=True)
    pin_constant = 12 / ((2 * pi) ** 3 * sp.sqrt(D * O * a * (m4 / 3)))
    kappa_integral = sp.gamma(sp.Rational(1, 6)) / 12 * (O / (2 * a)) ** sp.Rational(7, 6)
    lifetime_factor = 6 ** sp.Rational(2, 3) / 3
    angular_factor = 2 * pi
    coefficient = sp.simplify(
        lifetime_factor * angular_factor * sp.Rational(1, 6)
        * pin_constant * kappa_integral * b_integral.subs(Delta, D)
    )
    closed_DO = (
        3 ** sp.Rational(1, 6)
        * O ** sp.Rational(2, 3)
        * sp.gamma(sp.Rational(1, 6))
        / (54 * pi ** sp.Rational(3, 2) * a ** sp.Rational(5, 3))
    )
    assert sp.simplify(coefficient - closed_DO) == 0
    closed = closed_DO.subs({D: Delta, O: Omega})


    # Squared-exponential/Bargmann-Fock check with length scale lambda.
    lam = sp.symbols("lambda", positive=True)
    bf = sp.simplify(closed.subs({a: lam**-2, m4: 3 * lam**-4, chi: 15 * lam**-6}))
    bf_expected = 2 ** sp.Rational(2, 3) * 3 ** sp.Rational(5, 6) * sp.gamma(sp.Rational(1, 6)) / (54 * pi ** sp.Rational(3, 2) * lam**2)
    assert sp.simplify(bf - bf_expected) == 0


    print("Delta =", Delta)
    print("Omega =", Omega)
    print("mu_per_b =", mu_per_b)
    print("sigma2 =", sigma2)
    print("q_marginal_variance =", q_marginal_variance)
    print("b_integral =", b_integral)
    print("coefficient =", closed)
    print("BF_length_scale_check =", bf)
    print("all_symbolic_assertions = PASS")
