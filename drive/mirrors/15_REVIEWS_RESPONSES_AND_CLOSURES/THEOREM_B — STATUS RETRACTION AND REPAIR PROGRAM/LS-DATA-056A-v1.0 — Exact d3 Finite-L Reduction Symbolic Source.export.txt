import sympy as sp


# Exact constants in the d=3 finite-L contact-coefficient reduction.
pi = sp.pi


# Kac-Rice/lifetime factor, kappa Gaussian integral, and the two
# three-dimensional zero-density factors p_g(0), p_H(0).
C_raw = (
    6**sp.Rational(2,3)/18
    * 72**sp.Rational(7,6)/(2*sp.sqrt(2*pi))
    * (2*pi)**(-3)
)
C_exact = sp.simplify(C_raw)
C_expected = 3/(2**sp.Rational(4,3)*pi**sp.Rational(7,2))
assert sp.simplify(C_exact-C_expected) == 0


# Radial integral for a homogeneous degree-four cone weight in R^3.
r = sp.symbols('r', positive=True)
radial = sp.integrate(r**6*sp.exp(-r**2/2),(r,0,sp.oo))/(2*pi)**sp.Rational(3,2)
assert sp.simplify(radial-sp.Rational(15,4)/pi) == 0


# Planar BF recovery data in ambient dimension three.
D2 = sp.Rational(29,6)-sp.sqrt(6)
sigma2 = sp.Rational(1,24)
det_Sigma_g = 1
det_Sigma_H = 3
sphere_area = 4*pi


c3_from_finiteL_formula = sp.simplify(
    C_expected*sp.gamma(sp.Rational(7,6))*sphere_area
    *sigma2**sp.Rational(2,3)*D2
    /sp.sqrt(det_Sigma_g*det_Sigma_H)
)


c3_reference = sp.simplify(
    2**sp.Rational(1,6)*3**sp.Rational(1,3)
    *(29*sp.sqrt(6)-36)*sp.gamma(sp.Rational(1,6))
    /(432*pi**sp.Rational(5,2))
)
assert sp.simplify(c3_from_finiteL_formula-c3_reference) == 0


print('outer_prefactor_without_Gamma =', C_exact)
print('cone_radial_factor =', radial)
print('D2 =', D2)
print('c3_recovered =', c3_from_finiteL_formula)
print('c3_decimal =', sp.N(c3_from_finiteL_formula,60))
print('ALL_ASSERTIONS_PASS')
