from fractions import Fraction
from itertools import permutations

F = Fraction

# ---------------------------------------------------------------------------
# Laurent polynomials in r:  dict {power: Fraction coefficient}
# ---------------------------------------------------------------------------
def lp(c=0, p=0):
    return {p: F(c)} if c != 0 else {}

def lp_mul(a, b):
    out = {}
    for pa, ca in a.items():
        for pb, cb in b.items():
            k = pa + pb
            out[k] = out.get(k, F(0)) + ca * cb
    return {k: v for k, v in out.items() if v != 0}

def lp_add(a, b):
    out = dict(a)
    for p, c in b.items():
        out[p] = out.get(p, F(0)) + c
    return {k: v for k, v in out.items() if v != 0}

def lp_scale(a, s):
    return {p: c * s for p, c in a.items() if c * s != 0}

def lp_str(a):
    if not a:
        return "0"
    terms = []
    for p in sorted(a, reverse=True):
        c = a[p]
        terms.append(f"({c})*r^({p})" if p else f"({c})")
    return " + ".join(terms)

def det6(M):
    """Exact determinant of a 6x6 Laurent-polynomial matrix (720 terms)."""
    n = 6
    total = {}
    for perm in permutations(range(n)):
        # permutation sign
        sign, seen = 1, [False] * n
        for i in range(n):
            if seen[i]:
                continue
            j, clen = i, 0
            while not seen[j]:
                seen[j] = True
                j = perm[j]
                clen += 1
            if clen % 2 == 0:
                sign = -sign
        term = lp(sign, 0)
        for i in range(n):
            term = lp_mul(term, M[i][perm[i]])
            if not term:
                break
        total = lp_add(total, term)
    return total

def build_T(vminus_denom_power=3, gradient_correction=True):
    """Corrected transform T_r with Y_r = T_r X_r.  The two keyword arguments
    exist only to implement planted mutations; defaults are the true frame."""
    Z = lambda: [{} for _ in range(6)]
    T = [Z() for _ in range(6)]
    d = -vminus_denom_power                    # V- denominator power (true: r^-3)
    # row 0: V+
    T[0][0] = lp(F(1, 2)); T[0][1] = lp(F(1, 2))
    # row 1: V-_corr = f(S)/r^3 - f(M)/r^3 - (f_t(M)+f_t(S)) / (2 r^2)
    T[1][0] = lp(-1, d); T[1][1] = lp(1, d)
    if gradient_correction:
        T[1][2] = lp(F(-1, 2), d + 1); T[1][4] = lp(F(-1, 2), d + 1)
    # row 2: G_t^+
    T[2][2] = lp(F(1, 2)); T[2][4] = lp(F(1, 2))
    # row 3: G_t^-
    T[3][2] = lp(-1, -1); T[3][4] = lp(1, -1)
    # row 4: G_s^+
    T[4][3] = lp(F(1, 2)); T[4][5] = lp(F(1, 2))
    # row 5: G_s^-
    T[5][3] = lp(-1, -1); T[5][5] = lp(1, -1)
    return T
