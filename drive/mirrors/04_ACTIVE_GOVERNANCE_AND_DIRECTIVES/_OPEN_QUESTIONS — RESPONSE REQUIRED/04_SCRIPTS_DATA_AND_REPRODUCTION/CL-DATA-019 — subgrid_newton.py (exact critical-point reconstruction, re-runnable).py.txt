import numpy as np

TORUS = 24.0; NBASE = 128; AMP_CUT = 1e-13
line = lambda c="-": print(c*74)


class Field:
    """Band-limited BF field held as an explicit mode list."""
    def __init__(self, n, rng, side=TORUS):
        kx = 2.0*np.pi*np.fft.fftfreq(n, d=side/n)
        KX, KY = np.meshgrid(kx, kx, indexing="ij")
        amp = np.exp(-(KX**2+KY**2)/4.0)
        C = np.fft.fft2(rng.normal(size=(n, n)))*amp
        keep = amp > AMP_CUT
        self.kx = KX[keep]; self.ky = KY[keep]; self.c = C[keep]/(n*n)
        self.side = side
        self.c = self.c/self.grid(n).std()

    def grid(self, N):
        x = np.arange(N)*(self.side/N)
        PH = np.exp(1j*(self.kx[:, None]*x[None, :]))
        PHy = np.exp(1j*(self.ky[:, None]*x[None, :]))
        return np.real(np.einsum('m,mi,mj->ij', self.c, PH, PHy))

    def fgh(self, p):
        """f, grad, Hessian at p. EXACT (closed form)."""
        e = self.c*np.exp(1j*(self.kx*p[0]+self.ky*p[1]))
        return (np.real(e.sum()),
                np.array([np.real((1j*self.kx*e).sum()),
                          np.real((1j*self.ky*e).sum())]),
                np.array([[np.real((-self.kx**2*e).sum()),
                           np.real((-self.kx*self.ky*e).sum())],
                          [np.real((-self.kx*self.ky*e).sum()),
                           np.real((-self.ky**2*e).sum())]]))

    def newton(self, p0, iters=60, tol=1e-13):
        p = np.array(p0, dtype=float)
        for _ in range(iters):
            f, g, H = self.fgh(p)
            if abs(np.linalg.det(H)) < 1e-14: return None
            p = p - np.linalg.solve(H, g)
            if np.linalg.norm(g) < tol: break
        f, g, H = self.fgh(p)
        if np.linalg.norm(g) > 1e-7: return None
        return p, f, np.linalg.det(H), H[0, 0]+H[1, 1]


def h0_pairs_pix(field, side=TORUS):
    n = field.shape[0]; N = n*n; h = side/n
    flat = field.ravel(); order = np.argsort(-flat, kind="stable")
    parent = np.empty(N, dtype=np.int64); rb = np.zeros(N)
    bp = np.zeros(N, dtype=np.int64); added = np.zeros(N, dtype=bool)
    idx = np.arange(N); row, col = idx//n, idx % n
    nb = np.stack([((row-1) % n)*n+col, ((row+1) % n)*n+col,
                   row*n+(col-1) % n, row*n+(col+1) % n], axis=1)

    def find(x):
        r = x
        while parent[r] != r: r = parent[r]
        while parent[x] != r: parent[x], x = r, parent[x]
        return r

    out = []
    for p in order:
        v = flat[p]; parent[p] = p; rb[p] = v; bp[p] = p; added[p] = True
        for q in nb[p]:
            if not added[q]: continue
            rp, rq = find(p), find(q)
            if rp == rq: continue
            if rb[rp] >= rb[rq]: eld, yng = rp, rq
            else: eld, yng = rq, rp
            ell = rb[yng]-v
            if ell > 0:
                b = bp[yng]
                out.append((((b//n)*h, (b % n)*h), ((p//n)*h, (p % n)*h), ell))
            parent[yng] = eld
            if rb[eld] < rb[yng]: rb[eld] = rb[yng]; bp[eld] = bp[yng]
    return out
