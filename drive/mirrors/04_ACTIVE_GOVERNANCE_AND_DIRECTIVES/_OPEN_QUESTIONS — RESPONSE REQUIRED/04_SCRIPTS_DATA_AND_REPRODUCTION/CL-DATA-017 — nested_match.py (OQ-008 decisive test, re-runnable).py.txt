"""
CL-DATA-017 — nested_match.py — OQ-008 DECISIVE TEST
Nested matching of the SAME bars across resolutions.
Author: Claude (Anthropic), CL-* instance. 2026-07-20. Seeds 20260720+7000+k.
Uploaded to Drive in response to GP-AUD-010 sec 2 (reproducibility block).

DESIGN. The Bargmann-Fock amplitude exp(-k^2/4) is negligible beyond |k| ~ 9,
i.e. mode index ~33 on a torus of side 24. A field synthesised with modes capped
at the n=128 Nyquist is therefore the SAME CONTINUUM FUNCTION when sampled at
256, 512, 1024 via Fourier zero-padding. No ensemble variation; only the
sampling rate changes. Each bar is tracked as an individual object by birth
location and compared ONLY TO ITSELF -- so no global fold coefficient is used
and no lifetime-to-separation conversion is made. This sidesteps the kappa
objection in GP-AUD-010 sec 3 entirely.

PRE-COMMITTED READINGS (frozen before running):
  M1 matched short bars converge under refinement + high match rate -> REAL
  M2 lifetimes wander or match rate collapses at small ell -> DISCRETIZATION
  M3 mixed or unreadable -> INCONCLUSIVE
Reported regardless of outcome. See CL-DER-018 for the result and for why the
M1/M2 dichotomy turned out to be mis-specified.
"""
import numpy as np

TORUS = 24.0
NBASE = 128
line = lambda c="-": print(c * 74)


def base_spectrum(n, rng, side=TORUS):
    kx = 2.0*np.pi*np.fft.fftfreq(n, d=side/n)
    KX, KY = np.meshgrid(kx, kx, indexing="ij")
    return np.fft.fft2(rng.normal(size=(n, n))) * np.exp(-(KX**2+KY**2)/4.0)


def sample_at(F, N):
    """Zero-pad base spectrum to N x N and invert: SAME continuum field."""
    n = F.shape[0]
    if N == n:
        G = F.copy()
    else:
        h_ = n//2
        G = np.zeros((N, N), dtype=complex)
        G[:h_, :h_] = F[:h_, :h_]
        G[:h_, N-h_:] = F[:h_, n-h_:]
        G[N-h_:, :h_] = F[n-h_:, :h_]
        G[N-h_:, N-h_:] = F[n-h_:, n-h_:]
        G *= (N/n)**2
    return np.fft.ifft2(G).real


def h0_bars(field, side=TORUS):
    """Return (birth_x, birth_y, ell) for each finite H0 superlevel bar."""
    n = field.shape[0]; N = n*n; h = side/n
    flat = field.ravel()
    order = np.argsort(-flat, kind="stable")
    parent = np.empty(N, dtype=np.int64)
    rb = np.zeros(N); bp = np.zeros(N, dtype=np.int64)
    added = np.zeros(N, dtype=bool)
    idx = np.arange(N); row, col = idx//n, idx % n
    nb = np.stack([((row-1) % n)*n+col, ((row+1) % n)*n+col,
                   row*n+(col-1) % n, row*n+(col+1) % n], axis=1)

    def find(x):
        r = x
        while parent[r] != r: r = parent[r]
        while parent[x] != r: parent[x], x = r, parent[x]
        return r

    out = []
    for p in order:
        v = flat[p]; parent[p] = p; rb[p] = v; bp[p] = p; added[p] = True
        for q in nb[p]:
            if not added[q]: continue
            rp, rq = find(p), find(q)
            if rp == rq: continue
            if rb[rp] >= rb[rq]: eld, yng = rp, rq
            else: eld, yng = rq, rp
            ell = rb[yng]-v
            if ell > 0:
                b = bp[yng]
                out.append(((b//n)*h, (b % n)*h, ell))
            parent[yng] = eld
            if rb[eld] < rb[yng]: rb[eld] = rb[yng]; bp[eld] = bp[yng]
    return np.array(out)


def match(ref, other, tol, side=TORUS):
    out = []
    for bx, by, be in ref:
        dx = np.abs(other[:, 0]-bx); dx = np.minimum(dx, side-dx)
        dy = np.abs(other[:, 1]-by); dy = np.minimum(dy, side-dy)
        d = np.hypot(dx, dy); j = int(np.argmin(d))
        out.append((be, other[j, 2], d[j]) if d[j] <= tol else (be, np.nan, d[j]))
    return np.array(out)


print("OQ-008 DECISIVE TEST — nested matching, identical continuum field")
line("=")
RES = [128, 256, 512, 1024]
FIELDS = 8
rows = []
for k in range(FIELDS):
    F = base_spectrum(NBASE, np.random.default_rng(20260720+7000+k))
    bars = {}
    for N in RES:
        f = sample_at(F, N); f = f/f.std()
        bars[N] = h0_bars(f)
    ref = bars[RES[-1]]
    tol = TORUS/NBASE
    for N in RES[:-1]:
        for fine_ell, coarse_ell, d in match(ref, bars[N], tol):
            rows.append((k, N, fine_ell, coarse_ell, d))
    if k == 0:
        print("bar counts, field 0: " + ", ".join(f"n={N}:{len(bars[N])}" for N in RES))
R = np.array(rows)
print(f"fields={FIELDS}, reference n={RES[-1]}, match tolerance={TORUS/NBASE:.4f}")

line("=")
print("MATCH RATE by lifetime decade (does the bar exist at coarser n?)")
line()
bands = [(0, 1e-4), (1e-4, 1e-3), (1e-3, 1e-2), (1e-2, 1e-1), (1e-1, 1e9)]
print(f"{'ell band':>18}" + "".join(f"{f'n={N}':>12}" for N in RES[:-1]) + f"{'#ref bars':>11}")
for lo, hi in bands:
    sel = (R[:, 2] >= lo) & (R[:, 2] < hi)
    if sel.sum() == 0: continue
    cells = [100*np.isfinite(R[sel & (R[:, 1] == N), 3]).mean() for N in RES[:-1]]
    print(f"[{lo:.0e},{hi:.0e})".rjust(18) + "".join(f"{c:>11.1f}%" for c in cells)
          + f"{int(sel.sum()/3):>11}")

line("=")
print("LIFETIME CONVERGENCE: median |ell(n) - ell(1024)| / ell(1024)")
line()
print(f"{'ell band (at n=1024)':>22}" + "".join(f"{f'n={N}':>14}" for N in RES[:-1]))
for lo, hi in bands:
    sel = (R[:, 2] >= lo) & (R[:, 2] < hi) & np.isfinite(R[:, 3])
    if sel.sum() < 5: continue
    cells = []
    for N in RES[:-1]:
        s2 = sel & (R[:, 1] == N)
        if s2.sum() < 3: cells.append(np.nan); continue
        cells.append(np.median(np.abs(R[s2, 3]-R[s2, 2])/np.maximum(R[s2, 2], 1e-15)))
    print(f"[{lo:.0e},{hi:.0e})".rjust(22)
          + "".join("          n/a " if np.isnan(c) else f"{c:>14.3f}" for c in cells))

line("=")
print("ADJUDICATION")
line()
short = (R[:, 2] < 1e-3); long_ = (R[:, 2] >= 1e-2)
mr_s = 100*np.isfinite(R[short, 3]).mean()
mr_l = 100*np.isfinite(R[long_, 3]).mean()
print(f"  match rate, short bars (ell<1e-3):  {mr_s:.1f}%   n={int(short.sum())}")
print(f"  match rate, long  bars (ell>=1e-2): {mr_l:.1f}%   n={int(long_.sum())}")
s512 = short & (R[:, 1] == 512) & np.isfinite(R[:, 3])
rel = np.abs(R[s512, 3]-R[s512, 2])/np.maximum(R[s512, 2], 1e-15)
print(f"  short-bar median relative ell error, n=512 vs n=1024: {np.median(rel):.3f}")
print()
if mr_s < 60:
    print("  -> M2 SUPPORTED: short bars fail to match. Discretization.")
elif mr_s > 85 and np.median(rel) < 0.25:
    print("  -> M1 SUPPORTED: short bars persist and converge. Real features.")
else:
    print("  -> M3: INCONCLUSIVE as pre-specified. See CL-DER-018: the M1/M2")
    print("     dichotomy is mis-specified -- EXISTENCE and VALUE separate.")
