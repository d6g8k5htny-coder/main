#!/usr/bin/env python3
# OP-PROT-017-REF-v1.0 - reference implementation of the line-anchored
# frozen-body extraction standard. Published byte-exact with Google Docs
# conversion disabled, so that these bytes are the authored bytes.
#
# Normative steps N1-N8 of OP-PROT-017-v1.0.
# Verified 10/10 against the declared Theorem B chain identities on 2026-07-27.

import hashlib
import sys


class ExtractionAmbiguity(Exception):
    """Raised instead of guessing when marker resolution is not unique."""


def frozen_identity(raw, begin, end):
    """Return (byte_length, sha256_hex) of the frozen body in `raw`.

    raw   : exact bytes of a native text/plain export (step N1)
    begin : begin marker token, without terminator
    end   : end marker token, without terminator
    """
    text = raw.decode("utf-8-sig")                            # N2 drop BOM
    text = text.replace("\r\n", "\n").replace("\r", "\n")     # N3 normalize
    lines = text.split("\n")                                  # N4 resolve
    b = [i for i, L in enumerate(lines) if L.rstrip("\r") == begin]
    e = [i for i, L in enumerate(lines) if L.rstrip("\r") == end]
    if len(b) != 1 or len(e) != 1 or e[0] <= b[0]:
        raise ExtractionAmbiguity(
            "begin=%d end=%d ordered=%s : fail closed"
            % (len(b), len(e), bool(b and e and e[0] > b[0]))
        )
    seg = "\n".join(lines[b[0] + 1:e[0]])                     # N5, N6 preserve
    seg = seg.rstrip("\n") + "\n"                             # N7 terminate
    body = seg.encode("utf-8")                                # N8 hash
    return len(body), hashlib.sha256(body).hexdigest()


def verify(raw, begin, end, declared_bytes, declared_sha256):
    """True only if both the byte length and the digest match."""
    n, h = frozen_identity(raw, begin, end)
    return n == int(declared_bytes) and h == declared_sha256


if __name__ == "__main__":
    path, begin, end = sys.argv[1], sys.argv[2], sys.argv[3]
    n, h = frozen_identity(open(path, "rb").read(), begin, end)
    print("BODY_BYTES: %d" % n)
    print("BODY_SHA256: %s" % h)
    if len(sys.argv) > 5:
        ok = (n == int(sys.argv[4]) and h == sys.argv[5])
        print("VERDICT: %s" % ("PASS" if ok else "FAIL"))
        sys.exit(0 if ok else 1)
