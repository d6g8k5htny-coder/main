#!/usr/bin/env python3
import copy, json, re, sys
from pathlib import Path


OPEN = {"OPEN", "KILLED", "SUPERSEDED", "RETIRED", "BLOCKED"}


def load(path):
    return json.loads(Path(path).read_text())


def is_q_claim(claim):
    return claim.get("law_object", "").startswith("Q_") or "Q_" in claim.get("statement", "")


def verify(machine):
    errors=[]
    objects=machine.get("objects",{})
    claims=machine.get("claims",{})
    roots=machine.get("proposed_roots",[])
    if machine.get("schema")!="q0-law-specific/1.0": errors.append("SCHEMA")
    if not machine.get("active_machine_untouched"): errors.append("ACTIVE_MACHINE_OVERWRITE")
    for cid,c in claims.items():
        for field in ["law_object","measure_tag","status","claim_grade","statement","dependency_ids","bridge_ids","required_marks","source_files","falsifiers"]:
            if field not in c: errors.append(f"{cid}:MISSING:{field}")
        obj=c.get("law_object")
        if obj not in objects: errors.append(f"{cid}:UNKNOWN_OBJECT:{obj}")
        if obj in {"Q","q","Q0",""} and is_q_claim(c): errors.append(f"{cid}:BARE_Q")
        if re.search(r"(?<![A-Za-z0-9_])q\s*\(", c.get("statement","")):
            errors.append(f"{cid}:BARE_Q_STATEMENT")
        for dep in c.get("dependency_ids",[]):
            if dep not in claims and dep not in objects: errors.append(f"{cid}:UNKNOWN_DEP:{dep}")
    for cid,c in claims.items():
        if c.get("measure_tag")=="P_ADJ":
            if "ADJ_CONDITIONING_TRANSFER" not in c.get("bridge_ids",[]):
                errors.append(f"{cid}:MISSING_ADJ_BRIDGE")
            if "GRADIENT_ADJACENCY_MS" not in c.get("required_marks",[]):
                errors.append(f"{cid}:MISSING_ADJ_MARK")
    typed=claims.get("Q_TYPED_MS_RATE",{})
    if "ADJACENCY_TO_ONE_CUBIC" not in typed.get("dependency_ids",[]):
        errors.append("Q_TYPED_MS_RATE:MISSING_ADJ_TO_ONE")
    adj=claims.get("Q_ADJ_CONFIGURED_RATE",{})
    if "ADJ_POSITIVITY_UNIFORM" not in adj.get("dependency_ids",[]):
        errors.append("Q_ADJ_CONFIGURED_RATE:MISSING_ADJ_POSITIVITY")
    if "ADJ_CONDITIONING_TRANSFER" not in adj.get("dependency_ids",[]):
        errors.append("Q_ADJ_CONFIGURED_RATE:MISSING_TRANSFER_DEP")
    tb=claims.get("THEOREM_B_ADJ_CONTACT_FACTOR",{})
    for dep in ["J_TYPED_CONTACT_LIMIT","ADJ_POSITIVITY_UNIFORM"]:
        if dep not in tb.get("dependency_ids",[]): errors.append(f"THEOREM_B:MISSING:{dep}")
    root_reports={}
    def closure(cid,seen=None):
        seen=set() if seen is None else set(seen)
        if cid in seen: return {"cycle"}
        seen.add(cid)
        c=claims[cid]
        blockers=set()
        if c.get("status") in OPEN: blockers.add(cid)
        for dep in c.get("dependency_ids",[]):
            if dep in claims: blockers |= closure(dep,seen)
        return blockers
    for root in roots:
        if root not in claims:
            errors.append(f"UNKNOWN_ROOT:{root}")
            continue
        blockers=sorted(closure(root))
        root_reports[root]={"blockers":blockers,"unconditional_promotable":not blockers}
    return {"valid":not errors,"errors":errors,"root_reports":root_reports}


def negative_tests(base):
    tests=[]
    def run(name,mutator,needle):
        m=copy.deepcopy(base);mutator(m);r=verify(m)
        passed=(not r["valid"] and any(needle in e for e in r["errors"]))
        tests.append({"name":name,"passed":passed,"errors":r["errors"]})
    run("bare_q",lambda m:m["claims"]["Q_TYPED_MS_RATE"].update(law_object="q"),"BARE_Q")
    run("adj_without_bridge",lambda m:m["claims"]["Q_ADJ_CONFIGURED_RATE"].update(bridge_ids=[]),"MISSING_ADJ_BRIDGE")
    run("typed_without_adj_to_one",lambda m:m["claims"]["Q_TYPED_MS_RATE"].update(dependency_ids=["Q_ADJ_CONFIGURED_RATE"]),"MISSING_ADJ_TO_ONE")
    run("adj_without_positivity",lambda m:m["claims"]["Q_ADJ_CONFIGURED_RATE"].update(dependency_ids=["TYPED_REGIONAL_DEFECT_CUBIC","ADJ_CONDITIONING_TRANSFER"]),"MISSING_ADJ_POSITIVITY")
    run("theorem_b_drops_a",lambda m:m["claims"]["THEOREM_B_ADJ_CONTACT_FACTOR"].update(dependency_ids=["J_TYPED_CONTACT_LIMIT"]),"THEOREM_B:MISSING:ADJ_POSITIVITY_UNIFORM")
    run("adj_mark_missing",lambda m:m["claims"]["Q_ADJ_CONFIGURED_RATE"].update(required_marks=["MAXIMUM_M","SADDLE_S"]),"MISSING_ADJ_MARK")
    run("unknown_dependency",lambda m:m["claims"]["Q_TYPED_MS_RATE"]["dependency_ids"].append("NONEXISTENT"),"UNKNOWN_DEP")
    run("active_overwrite",lambda m:m.update(active_machine_untouched=False),"ACTIVE_MACHINE_OVERWRITE")
    return tests

    path=sys.argv[1] if len(sys.argv)>1 else "q0_law_specific_successor_v1.json"
