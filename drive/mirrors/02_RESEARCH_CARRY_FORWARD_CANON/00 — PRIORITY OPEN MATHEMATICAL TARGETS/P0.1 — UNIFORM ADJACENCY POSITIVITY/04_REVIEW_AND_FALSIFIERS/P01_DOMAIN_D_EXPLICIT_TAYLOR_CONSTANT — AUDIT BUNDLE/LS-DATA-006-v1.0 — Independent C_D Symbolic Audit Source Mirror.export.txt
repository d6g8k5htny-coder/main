T = sp.symbols("t", real=True)
I = sp.Interval(sp.Rational(-1, 4), sp.Rational(5, 4))


H00 = 2*T**3 - 3*T**2 + 1
H10 = T**3 - 2*T**2 + T
H01 = -2*T**3 + 3*T**2
H11 = T**3 - T**2
B0 = 1 - T

def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()




def exact_sup_abs(poly: sp.Expr) -> sp.Expr:
    """Exact sup |poly| on I by endpoints and real critical points."""
    derivative = sp.diff(poly, T)
    candidates = [I.start, I.end]
    for root in sp.solve(sp.Eq(derivative, 0), T):
        if root.is_real and bool(I.contains(root)):
            candidates.append(root)
    values = [sp.Abs(sp.simplify(poly.subs(T, x))) for x in candidates]
    return sp.simplify(sp.Max(*values))




def exact_sup_sum_abs(polys: list[sp.Expr]) -> sp.Expr:
    """Exact sup of sum |p_i| via sign breakpoints and critical points per cell."""
    points = {I.start, I.end}
    for p in polys:
        for root in sp.solve(sp.Eq(p, 0), T):
            if root.is_real and bool(I.contains(root)):
                points.add(root)
    points_sorted = sorted(points, key=lambda x: float(x))
    candidates = set(points_sorted)
    for a, b in zip(points_sorted[:-1], points_sorted[1:]):
        mid = sp.Rational(1, 2) * (a + b)
        signed = sp.Integer(0)
        for p in polys:
            val = sp.N(p.subs(T, mid))
            sign = 1 if val >= 0 else -1
            signed += sign * p
        for root in sp.solve(sp.Eq(sp.diff(signed, T), 0), T):
            if root.is_real and float(a) <= float(root) <= float(b):
                candidates.add(root)
    vals = []
    for x in candidates:
        vals.append(sp.simplify(sum(sp.Abs(p.subs(T, x)) for p in polys)))
    return sp.simplify(max(vals, key=lambda x: float(x)))




def dec(x: sp.Expr, n: int = 70) -> str:
    return str(sp.N(x, n))
