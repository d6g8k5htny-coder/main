from decimal import Decimal, localcontext, ROUND_FLOOR, ROUND_CEILING, getcontext
import json


getcontext().prec = 80
D = Decimal
PREC = 80


def dec(x):
    return x if isinstance(x, Decimal) else Decimal(str(x))


def calc(fn, rounding):
    with localcontext() as ctx:
        ctx.prec = PREC
        ctx.rounding = rounding
        return fn()


class I:
    def __init__(self, lo, hi=None):
        self.lo = dec(lo)
        self.hi = dec(lo if hi is None else hi)
        if self.lo > self.hi:
            raise ValueError("invalid interval")


    def __add__(self, other):
        other = as_interval(other)
        return I(
            calc(lambda: self.lo + other.lo, ROUND_FLOOR),
            calc(lambda: self.hi + other.hi, ROUND_CEILING),
        )
    __radd__ = __add__


    def __neg__(self):
        return I(-self.hi, -self.lo)


    def __sub__(self, other):
        return self + (-as_interval(other))


    def __rsub__(self, other):
        return as_interval(other) - self


    def __mul__(self, other):
        other = as_interval(other)
        lower_products = []
        upper_products = []
        for a in (self.lo, self.hi):
            for b in (other.lo, other.hi):
                lower_products.append(calc(lambda a=a, b=b: a * b, ROUND_FLOOR))
                upper_products.append(calc(lambda a=a, b=b: a * b, ROUND_CEILING))
        return I(min(lower_products), max(upper_products))
    __rmul__ = __mul__


    def reciprocal(self):
        if self.lo <= 0 <= self.hi:
            raise ZeroDivisionError("interval contains zero")
        lower_values = [
            calc(lambda: D(1) / self.lo, ROUND_FLOOR),
            calc(lambda: D(1) / self.hi, ROUND_FLOOR),
        ]
        upper_values = [
            calc(lambda: D(1) / self.lo, ROUND_CEILING),
            calc(lambda: D(1) / self.hi, ROUND_CEILING),
        ]
        return I(min(lower_values), max(upper_values))


    def __truediv__(self, other):
        return self * as_interval(other).reciprocal()


    def __rtruediv__(self, other):
        return as_interval(other) / self


    def __pow__(self, exponent):
        if not isinstance(exponent, int) or exponent < 0:
            raise ValueError("only nonnegative integer powers")
        result = I(1)
        for _ in range(exponent):
            result = result * self
        return result


    def absolute(self):
        if self.lo >= 0:
            return self
        if self.hi <= 0:
            return -self
        return I(0, max(-self.lo, self.hi))


    def sqrt(self):
        if self.lo < 0:
            raise ValueError("sqrt of negative interval")
        return I(
            calc(lambda: self.lo.sqrt(), ROUND_FLOOR),
            calc(lambda: self.hi.sqrt(), ROUND_CEILING),
        )


    def as_list(self):
        return [str(self.lo), str(self.hi)]


def as_interval(x):
    return x if isinstance(x, I) else I(x)


def supabs(x):
    x = as_interval(x)
    return max(abs(x.lo), abs(x.hi))


R = I(D(1) / D(20))
SIGMA = I(D(1) / D(10))
SQRT2 = I(2).sqrt()
MU = I(1) / (I(40) * SQRT2)


def primary_box():
    return {
        "a": I("0.0024", "0.0026"),
        "w": I("-0.001", "0.001"),
        "z": I("0.99", "1.01"),
        "s": I("-1.02", "-0.98"),
        "c40": I("-0.01", "0.01"),
        "c31": I("-0.01", "0.01"),
        "c22": I("-0.01", "0.01"),
        "c13": I("-0.01", "0.01"),
        "c04": I("-0.01", "0.01"),

def altered_box(base, key, center, radius):
    result = {name: I(value.lo, value.hi) for name, value in base.items()}
    result[key] = I(dec(center) - dec(radius), dec(center) + dec(radius))
    return result


def flow_x(X, Y, p):
    a, w, c40, c31, c22, c13 = [p[k] for k in ("a","w","c40","c31","c22","c13")]
    return (
        X**2 - I(1)/4 + a*X*Y + (w/2)*(Y**2)
        + R * (
            c40*(X**3-X/4)/6
            + c31*(3*(X**2)-I(1)/4)*Y/6
            + c22*X*(Y**2)/2
            + c13*(Y**3)/6
        )
    )


def flow_y(X, Y, p):
    a, w, z, s, c31, c22, c13, c04 = [
        p[k] for k in ("a","w","z","s","c31","c22","c13","c04")
    ]
    return (
        a*(X**2-I(1)/4)/2 + s*Y + w*X*Y + (z/2)*(Y**2)
        + R * (
            c31*(X**3-X/4)/6
            + c22*(X**2)*Y/2
            + c13*X*(Y**2)/2
            + c04*(Y**3)/6
        )
    )


def jac11(X, Y, p):
    a, c40, c31, c22 = [p[k] for k in ("a","c40","c31","c22")]
    return 2*X + a*Y + R*(
        c40*(3*(X**2)-I(1)/4)/6 + c31*X*Y + c22*(Y**2)/2
    )


def jac12(X, Y, p):
    a, w, c31, c22, c13 = [p[k] for k in ("a","w","c31","c22","c13")]
    return a*X + w*Y + R*(
        c31*(3*(X**2)-I(1)/4)/6 + c22*X*Y + c13*(Y**2)/2
    )


def jac22(X, Y, p):
    s, w, z, c22, c13, c04 = [p[k] for k in ("s","w","z","c22","c13","c04")]
    return s + w*X + z*Y + R*(
        c22*(X**2)/2 + c13*X*Y + c04*(Y**2)/2
    )


def compute(p, kappa_value="0.01"):
    kappa = I(kappa_value)
    a, w, z, s, c40, c31, c22, c13, c04 = [
        p[k] for k in ("a","w","z","s","c40","c31","c22","c13","c04")
    ]


    witness_f = a.absolute()/8 + R*c31.absolute()/48
    witness_a = (z.absolute() + R*c13.absolute()/2) * witness_f
    witness_b = R*c04.absolute()*(witness_f**2)/2
    threshold_c = (
        w.absolute()/2 + MU + R*c22.absolute()/8
        + witness_a/MU + witness_b/(MU**2)
    )
    width = witness_f/MU


    margins = {}
    margins["m_threshold"] = (-s-threshold_c).lo


    a_s = I(1) + R*c40/12
    b_s = a/2 + R*c31/12
    d_s = s + w/2 + R*c22/8
    det_s = a_s*d_s - b_s**2


    a_m = -I(1) + R*c40/12
    b_m = -a/2 + R*c31/12
    d_m = s - w/2 + R*c22/8
    det_m = a_m*d_m - b_m**2


    gap = a_s-d_s
    if gap.lo <= 0:
        tau_upper = D("Infinity")
    else:
        tau_upper = supabs(b_s)/gap.lo


    margins["m_saddle_det"] = -det_s.hi
    margins["m_max_A"] = -a_m.hi
    margins["m_max_D"] = -d_m.hi
    margins["m_max_det"] = det_m.lo
    margins["m_axial"] = gap.lo
    margins["m_cone_slope"] = kappa.lo - 2*tau_upper


    xi = I(0, SIGMA.hi)


    def qx(xi, t):
        return (
            12*a*t*xi - 6*a*t
            - 2*c13*R*(t**3)*(xi**2)
            + 6*c22*R*(t**2)*(xi**2)
            - 3*c22*R*(t**2)*xi
            - 6*c31*R*t*(xi**2)
            + 6*c31*R*t*xi
            - c31*R*t
            + 2*c40*R*(xi**2)
            - 3*c40*R*xi
            + c40*R
            - 6*(t**2)*w*xi
            - 12*xi + 12
        ) / 12


    def qy(xi, t):
        return (
            12*a*xi - 12*a
            + 4*c04*R*(t**3)*(xi**2)
            - 12*c13*R*(t**2)*(xi**2)
            + 6*c13*R*(t**2)*xi
            + 12*c22*R*t*(xi**2)
            - 12*c22*R*t*xi
            + 3*c22*R*t
            - 4*c31*R*(xi**2)
            + 6*c31*R*xi
            - 2*c31*R
            + 24*s*t
            + 12*(t**2)*xi*z
            - 24*t*w*xi
            + 12*t*w
        ) / 24


    t_full = I(-kappa.hi, kappa.hi)
    t_plus = I(kappa.lo)
    t_minus = I(-kappa.lo)
    margins["m_cone_axis"] = qx(xi, t_full).lo
    margins["m_cone_upper"] = (kappa*qx(xi,t_plus)-qy(xi,t_plus)).lo
    margins["m_cone_lower"] = (kappa*qx(xi,t_minus)+qy(xi,t_minus)).lo
    margins["m_handoff"] = width.lo/2-kappa.lo*SIGMA.lo


    x_central = I(-D(1)/2+SIGMA.lo, D(1)/2-SIGMA.lo)
    y_top = width
    y_bottom = -width
    y_strip = I(-width.hi, width.hi)


    margins["m_strip_top"] = (-flow_y(x_central,y_top,p)).lo
    margins["m_strip_bottom"] = flow_y(x_central,y_bottom,p).lo
    margins["m_central_drift"] = (-flow_x(x_central,y_strip,p)).lo
    margins["m_transverse_contraction"] = -jac22(x_central,y_strip,p).hi


    rho = 2*((SIGMA**2+width**2).sqrt())
    rho_upper = rho.hi
    x_capture = I(-D(1)/2-rho_upper, -D(1)/2+rho_upper)
    y_capture = I(-rho_upper, rho_upper)


    j11 = jac11(x_capture,y_capture,p)
    j12 = jac12(x_capture,y_capture,p)
    j22 = jac22(x_capture,y_capture,p)
    offdiag = supabs(j12)


    margins["m_chart_X"] = D(3)/4-(D(1)/2+rho_upper)
    margins["m_chart_Y"] = D(1)-rho_upper
    margins["m_gersh_1"] = (-j11).lo-offdiag
    margins["m_gersh_2"] = (-j22).lo-offdiag


    extras = {
        "F": witness_f.as_list(),
        "A": witness_a.as_list(),
        "B": witness_b.as_list(),
        "C": threshold_c.as_list(),
        "W": width.as_list(),
        "rho": rho.as_list(),
        "tau_upper": str(tau_upper),
    }
    return margins, extras
