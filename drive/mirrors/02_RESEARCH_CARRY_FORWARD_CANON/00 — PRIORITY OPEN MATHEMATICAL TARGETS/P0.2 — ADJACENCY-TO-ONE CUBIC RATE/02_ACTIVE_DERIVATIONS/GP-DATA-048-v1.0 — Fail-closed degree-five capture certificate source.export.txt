K = 4096.0
EPSILON_0 = 1.0 / (1024.0 * K)
REQUIRED_KEYS = ("r", "q", "a", "w", "z", "c40", "c31", "c22", "c13", "c04")




def _finite_number(value: Any) -> bool:
    try:
        return math.isfinite(float(value))
    except (TypeError, ValueError, OverflowError):
        return False




def _strict_positive(value: float) -> bool:
    return math.isfinite(value) and value > 0.0




def evaluate_certificate(payload: dict[str, Any]) -> dict[str, Any]:

            "pass": False,

            "derived": {},
            "slacks": {},
            "scope": "No exact-field adjudication.",

            "pass": False,
            "failure_class": "INVALID_INPUT",
            "failures": ["all inputs must be finite real numbers"],

            "slacks": {},
            "scope": "No exact-field adjudication.",

        float(payload[key]) for key in REQUIRED_KEYS

            "pass": False,
            "failure_class": "INVALID_INPUT",
            "failures": ["r must be strictly positive"],
            "derived": {},
            "slacks": {},
            "scope": "No exact-field adjudication.",

    threshold_margin = -s - C_cap
