def ffloat(x: Any) -> Fraction:
    """Exact rational value of the input IEEE-754 float."""
    return Fraction.from_float(float(x))




def load_frozen_cl_source(path: Path):
    raw = path.read_bytes()
    digest = hashlib.sha256(raw).hexdigest()
    if digest != EXPECTED_CL_SHA256:

            f"CL source hash mismatch: expected {EXPECTED_CL_SHA256}, got {digest}"

    if spec is None:
        raise RuntimeError("Could not build import specification for CL source")
    module = importlib.util.module_from_spec(spec)
    loader.exec_module(module)
    return module, digest




def exact_certificate(U_float, lam: Fraction, sf: Fraction) -> dict[str, Fraction]:
    u = [ffloat(x) for x in U_float]
    a, w, z, c40, c31, c22, c13, c04 = u
    R = Fraction(1, 1) + sum(abs(x) for x in u)
    r = lam * EPS0 / R**5


    F = abs(a) / 8 + r * abs(c31) / 48
    F_star = 1 + F
    mu = K * R**5
    W = F_star / mu
    sigma = 4 * R * W
    rho = 8 * R * W
    C_cap = 2 * K * R**5 + 2 * R
    s = -sf * C_cap


    # T-B: exact scalar upper bound for partial_Y Ydot on the whole strip.
    non_s = (
        abs(w) / 2
        + r * abs(c22) / 8
        + (abs(z) + r * abs(c13) / 2) * W
        + r * abs(c04) * W**2 / 2
    )
    tb_derivative_margin = -mu - (s + non_s)


    # Once partial_Y Ydot <= -mu, |Ydot(X,0)| <= F and mu*W=1+F
    # certify Ydot(X,+W)<=-1 and Ydot(X,-W)>=+1 continuously.

    # upper bound 1/120 used in GP-DER-047.

        abs(a) * W / 2
        + abs(w) * W**2 / 2
        + r
        * (
            abs(c40) / 120
            + abs(c31) * W / 12
            + abs(c22) * W**2 / 4
            + abs(c13) * W**3 / 6
        )
    )
    tc_ratio_margin = Fraction(501, 1000) * R * W - E_X_upper
    tc_transit_margin = sigma * (1 - sigma) - E_X_upper - 3 * R * W


    # T-E: global quadratic-form lower bound from Young's inequality.
    A0 = 1 - r * c40 / 12
    B_M = -a / 2 + r * c31 / 12

        raise AssertionError("A0 is nonpositive on a purported REG_r sample")

    qy = T - 2 * B_M**2 / A0

    te_section_margin = rho**2 - sigma**2 - W**2

    # variation bound epsilon=96 R^2 W; no angular sampling is needed.
    epsilon = 96 * R**2 * W
    te_eps_margin = Fraction(3, 64) - epsilon
    te_inward_margin = lmin_lower - epsilon


    return {
        "R": R,
        "r": r,
        "s": s,
        "W": W,
        "sigma": sigma,
        "rho": rho,
        "TB_derivative_margin": tb_derivative_margin,
        "TB_boundary_identity": tb_boundary_identity,
        "TC_ratio_margin": tc_ratio_margin,
        "TC_transit_margin": tc_transit_margin,
        "TE_lmin_margin": te_lmin_margin,

        "TE_eps_margin": te_eps_margin,
        "TE_inward_margin": te_inward_margin,
