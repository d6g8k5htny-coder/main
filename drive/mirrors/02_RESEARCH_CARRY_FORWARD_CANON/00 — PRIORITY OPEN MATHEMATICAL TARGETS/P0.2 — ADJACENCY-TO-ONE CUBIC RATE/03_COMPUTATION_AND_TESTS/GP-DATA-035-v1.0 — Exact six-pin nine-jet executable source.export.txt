Conditioning pins at separation r:


    M=(-r/2,0): (f, f_x, f_y) = (b,0,0),

    q=f_yy, a=f_xxy, w=f_xyy, z=f_yyy,

    c13=f_xyyy, c04=f_yyyy.


The program evaluates the exact finite-r conditional Gaussian mean and

JET_NAMES = ["q", "a", "w", "z", "c40", "c31", "c22", "c13", "c04"]
JET_DERIVATIVES = [
    (0, 2), (2, 1), (1, 2), (0, 3),
    (4, 0), (3, 1), (2, 2), (1, 3), (0, 4),
]
PIN_DERIVATIVES = [(0, 0), (1, 0), (0, 1)]
COALESCED_CONSTRAINTS = [(0, 0), (1, 0), (2, 0), (0, 1), (1, 1), (3, 0)]




@dataclass(frozen=True)
class LawResult:
    r: mp.mpf
    mean: mp.matrix
    covariance: mp.matrix
    pin_gram: mp.matrix




def hermite_probabilists(n: int, x: mp.mpf) -> mp.mpf:
    """Probabilists' Hermite polynomial He_n(x), recurrence-evaluated."""
    if n < 0:
        raise ValueError("Hermite order must be nonnegative")
    if n == 0:
        return mp.mpf(1)
    if n == 1:
        return x
    h_nm2 = mp.mpf(1)
    h_nm1 = x
    for k in range(1, n):
        h_n = x * h_nm1 - k * h_nm2
        h_nm2, h_nm1 = h_nm1, h_n
    return h_nm1




def theta_derivative(t: mp.mpf, L: mp.mpf, order: int, shells: int) -> mp.mpf:
    """Derivative of sum_n exp(-(t+nL)^2/2)."""
    total = mp.mpf(0)
    for n in range(-shells, shells + 1):
        z = t + n * L
        total += ((-1) ** order) * hermite_probabilists(order, z) * mp.exp(-z * z / 2)
    return total




def kernel_1d(t: mp.mpf, L: mp.mpf, order: int = 0, shells: int = 6) -> mp.mpf:
    return theta_derivative(t, L, order, shells) / theta_derivative(mp.mpf(0), L, 0, shells)




def covariance(
    p: tuple[mp.mpf, mp.mpf],
    alpha: tuple[int, int],
    q: tuple[mp.mpf, mp.mpf],
    beta: tuple[int, int],
    L: mp.mpf,
    shells: int,
) -> mp.mpf:
    """Cov(partial^alpha f(p), partial^beta f(q))."""
    dx = p[0] - q[0]
    dy = p[1] - q[1]
    return (
        ((-1) ** (beta[0] + beta[1]))
        * kernel_1d(dx, L, alpha[0] + beta[0], shells)
        * kernel_1d(dy, L, alpha[1] + beta[1], shells)
    )




def finite_r_law(
    r: mp.mpf,
    *,
    L: mp.mpf,
    b: mp.mpf,
    shells: int,
) -> LawResult:
    if r <= 0:
        raise ValueError("r must be positive")
    ell = r ** 3 / 6
    M = (-r / 2, mp.mpf(0))
    S = (r / 2, mp.mpf(0))
    O = (mp.mpf(0), mp.mpf(0))


    pins: list[tuple[tuple[mp.mpf, mp.mpf], tuple[int, int]]] = []
    for point in (M, S):
        for derivative in PIN_DERIVATIVES:
            pins.append((point, derivative))


    pin_values = mp.matrix([b, 0, 0, b - ell, 0, 0])
    G = mp.matrix(6, 6)
    C_tp = mp.matrix(len(JET_DERIVATIVES), 6)
    C_tt = mp.matrix(len(JET_DERIVATIVES), len(JET_DERIVATIVES))


    for i, (p, alpha) in enumerate(pins):
        for j, (q, beta) in enumerate(pins):
            G[i, j] = covariance(p, alpha, q, beta, L, shells)


    for i, derivative in enumerate(JET_DERIVATIVES):
        for j, (p, alpha) in enumerate(pins):
            C_tp[i, j] = covariance(O, derivative, p, alpha, L, shells)
        for j, other_derivative in enumerate(JET_DERIVATIVES):
            C_tt[i, j] = covariance(O, derivative, O, other_derivative, L, shells)


    G_inv = mp.inverse(G)
    mean = C_tp * G_inv * pin_values
    conditional_covariance = C_tt - C_tp * G_inv * C_tp.T
    conditional_covariance = (conditional_covariance + conditional_covariance.T) / 2
    return LawResult(r=r, mean=mean, covariance=conditional_covariance, pin_gram=G)




def planar_same_point_covariance(alpha: tuple[int, int], beta: tuple[int, int]) -> mp.mpf:
    """Planar BF derivative covariance at one point."""
    nx = alpha[0] + beta[0]
    ny = alpha[1] + beta[1]
    return (
        ((-1) ** (beta[0] + beta[1]))
        * ((-1) ** nx) * hermite_probabilists(nx, mp.mpf(0))
        * ((-1) ** ny) * hermite_probabilists(ny, mp.mpf(0))
    )




def coalesced_limiting_law(b: mp.mpf) -> tuple[mp.matrix, mp.matrix]:
    """Independent limiting Schur complement.


    Coalesced six-pin constraints:
        (f, f_x, f_xx, f_y, f_xy, f_xxx)=(b,0,0,0,0,2).
    """
    values = mp.matrix([b, 0, 0, 0, 0, 2])
    G = mp.matrix([
        [planar_same_point_covariance(a, c) for c in COALESCED_CONSTRAINTS]
        for a in COALESCED_CONSTRAINTS
    ])
    C_tp = mp.matrix([
        [planar_same_point_covariance(d, c) for c in COALESCED_CONSTRAINTS]
        for d in JET_DERIVATIVES
    ])
    C_tt = mp.matrix([
        [planar_same_point_covariance(d, e) for e in JET_DERIVATIVES]
        for d in JET_DERIVATIVES
    ])
    G_inv = mp.inverse(G)
    mean = C_tp * G_inv * values
    cov = C_tt - C_tp * G_inv * C_tp.T
    cov = (cov + cov.T) / 2
    return mean, cov




def matrix_to_strings(M: mp.matrix, digits: int = 30) -> list[list[str]]:
    return [[mp.nstr(M[i, j], digits) for j in range(M.cols)] for i in range(M.rows)]




def vector_to_strings(v: mp.matrix, digits: int = 30) -> list[str]:
    return [mp.nstr(v[i], digits) for i in range(v.rows)]




def max_abs_matrix(M: mp.matrix) -> mp.mpf:
    return max(abs(M[i, j]) for i in range(M.rows) for j in range(M.cols))




def eigvals_symmetric(M: mp.matrix) -> list[mp.mpf]:
    values, _ = mp.eigsy(M)
    return [values[i] for i in range(values.rows)]




def sha256_file(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
