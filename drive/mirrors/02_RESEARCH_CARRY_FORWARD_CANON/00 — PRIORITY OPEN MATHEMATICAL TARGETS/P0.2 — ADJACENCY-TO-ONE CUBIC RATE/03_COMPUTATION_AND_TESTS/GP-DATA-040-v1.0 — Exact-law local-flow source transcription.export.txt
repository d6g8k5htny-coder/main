@njit(cache=True)
def quartic_derivatives(x, y, c40, c31, c22, c13, c04):
    p4x = ((c40 / 6.0) * x * (x * x - 0.25)
           + (c31 / 6.0) * (3.0 * x * x - 0.25) * y
           + (c22 / 2.0) * x * y * y
           + (c13 / 6.0) * y * y * y)
    p4y = ((c31 / 6.0) * x * (x * x - 0.25)
           + (c22 / 2.0) * x * x * y
           + (c13 / 2.0) * x * y * y
           + (c04 / 6.0) * y * y * y)
    return p4x, p4y




@njit(cache=True)
def nonlinear_rhs(x, y, r, params, q4_on):
    q, a, w, z, c40, c31, c22, c13, c04 = params
    fx = x * x - 0.25 + a * x * y + 0.5 * w * y * y
    fy = 0.5 * a * (x * x - 0.25) + w * x * y + 0.5 * z * y * y
    if q4_on:
        p4x, p4y = quartic_derivatives(x, y, c40, c31, c22, c13, c04)
        fx += r * p4x
        fy += r * p4y
    return fx, fy




@njit(cache=True)
def full_rhs(x, y, r, params, q4_on):
    fx, fy = nonlinear_rhs(x, y, r, params, q4_on)
    fy += (params[0] / r) * y
    return fx, fy




@njit(cache=True)
def unstable_vector_at_s(r, params, q4_on):
    q, a, w, z, c40, c31, c22, c13, c04 = params
    A = 1.0
    B = 0.5 * a
    D = q / r + 0.5 * w
    if q4_on:
        A += r * c40 / 12.0
        B += r * c31 / 12.0
        D += r * c22 / 8.0
    theta = 0.5 * math.atan2(2.0 * B, A - D)
    vx = math.cos(theta)
    vy = math.sin(theta)
    if vx > 0.0:
        vx = -vx
        vy = -vy
    norm = math.sqrt(vx * vx + vy * vy)
    return vx / norm, vy / norm




@njit(cache=True)
def rk4_nonlinear_step(x, y, dt, r, params, q4_on):
    k1x, k1y = nonlinear_rhs(x, y, r, params, q4_on)
    k2x, k2y = nonlinear_rhs(x + 0.5 * dt * k1x, y + 0.5 * dt * k1y, r, params, q4_on)
    k3x, k3y = nonlinear_rhs(x + 0.5 * dt * k2x, y + 0.5 * dt * k2y, r, params, q4_on)
    k4x, k4y = nonlinear_rhs(x + dt * k3x, y + dt * k3y, r, params, q4_on)
    return (x + (dt / 6.0) * (k1x + 2.0 * k2x + 2.0 * k3x + k4x),
            y + (dt / 6.0) * (k1y + 2.0 * k2y + 2.0 * k3y + k4y))




@njit(cache=True)
def integrate_one(r, params, q4_on, dt, tau_max):
    vx, vy = unstable_vector_at_s(r, params, q4_on)
    x = 0.5 + EPS * vx
    y = EPS * vy
    lam = params[0] / r
    linear_half = math.exp(lam * dt * 0.5)
    steps = int(math.ceil(tau_max / dt))
    cap2 = CAPTURE_RADIUS * CAPTURE_RADIUS
    other_dist2 = OTHER_MIN_DIST * OTHER_MIN_DIST
    for step in range(steps):
        y *= linear_half
        x, y = rk4_nonlinear_step(x, y, dt, r, params, q4_on)
        y *= linear_half
        tau = (step + 1) * dt


        dxm = x + 0.5
        if dxm * dxm + y * y <= cap2:
            return CAPTURE, x, y, tau
        if abs(x) >= CHART_RADIUS or abs(y) >= CHART_RADIUS or not math.isfinite(x) or not math.isfinite(y):
            return EXIT, x, y, tau
        if tau >= OTHER_MIN_TAU and (step % 20 == 0):
            f1, f2 = full_rhs(x, y, r, params, q4_on)
            dsm = (x - 0.5) * (x - 0.5) + y * y
            dmm = (x + 0.5) * (x + 0.5) + y * y
            if f1 * f1 + f2 * f2 <= OTHER_GRAD_TOL * OTHER_GRAD_TOL and dsm > other_dist2 and dmm > cap2:
                return OTHER, x, y, tau
    return STALL, x, y, tau_max




@njit(parallel=True, cache=True)
def integrate_batch(r, samples, q4_on, dt, tau_max):
    n = samples.shape[0]
    status = np.empty(n, dtype=np.int8)
    xend = np.empty(n, dtype=np.float64)
    yend = np.empty(n, dtype=np.float64)
    tend = np.empty(n, dtype=np.float64)
    for i in prange(n):
        st, x, y, t = integrate_one(r, samples[i], q4_on, dt, tau_max)
        status[i] = st
        xend[i] = x
        yend[i] = y
        tend[i] = t
    return status, xend, yend, tend




def type_and_weight(samples: np.ndarray, r: float, q4_on: bool):
    q, a, w = samples[:, 0], samples[:, 1], samples[:, 2]
    if q4_on:
        c40, c31, c22 = samples[:, 4], samples[:, 5], samples[:, 6]
        hm11 = -r + r * r * c40 / 12.0
        hm12 = -a * r / 2.0 + r * r * c31 / 12.0
        hm22 = q - w * r / 2.0 + r * r * c22 / 8.0
        hs11 = r + r * r * c40 / 12.0
        hs12 = a * r / 2.0 + r * r * c31 / 12.0
        hs22 = q + w * r / 2.0 + r * r * c22 / 8.0
    else:
        hm11 = np.full_like(q, -r)
        hm12 = -a * r / 2.0
        hm22 = q - w * r / 2.0
        hs11 = np.full_like(q, r)
        hs12 = a * r / 2.0
        hs22 = q + w * r / 2.0
    detm = hm11 * hm22 - hm12 * hm12
    dets = hs11 * hs22 - hs12 * hs12
    typed = (hm11 < 0.0) & (detm > 0.0) & (dets < 0.0)
    weights = np.zeros(samples.shape[0], dtype=np.float64)
    weights[typed] = np.abs(detm[typed] * dets[typed])
    return typed, weights, hs22, detm, dets




def weighted_ess(x: np.ndarray) -> float:
    s = float(np.sum(x))
    ss = float(np.dot(x, x))
    return 0.0 if ss == 0.0 else s * s / ss




def estimate_and_if(weights: np.ndarray, indicator: np.ndarray):
    n = weights.size
    sw = float(weights.sum())
    if sw <= 0.0:
        return 0.0, float("nan"), np.zeros(n)
    est = float(np.dot(weights, indicator) / sw)
    meanw = sw / n
    psi = weights * (indicator - est) / meanw
    se = float(np.std(psi, ddof=1) / math.sqrt(n))
    return est, se, psi




def selected_quantiles(values: np.ndarray, weights: np.ndarray, probs=(0.5, 0.9, 0.99)):
    if values.size == 0 or weights.sum() <= 0:
        return {str(p): None for p in probs}
    order = np.argsort(values)
    v = values[order]
    w = weights[order]
    c = np.cumsum(w)
    c /= c[-1]
    return {str(p): float(v[min(np.searchsorted(c, p), len(v)-1)]) for p in probs}




def full_cov_from_result(result):
    return np.array([[float(result.covariance[i, j]) for j in range(9)] for i in range(9)], dtype=np.float64)




def full_mean_from_result(result):
    return np.array([float(result.mean[i]) for i in range(9)], dtype=np.float64)




def stable_cholesky(cov):
    cov = (cov + cov.T) / 2.0
    vals = np.linalg.eigvalsh(cov)
    if vals[0] <= 0:
        cov = cov + np.eye(cov.shape[0]) * (-vals[0] + 1e-14)
    return np.linalg.cholesky(cov)




def summarize_model(model_name, model_index, r, samples, q4_on, rung_index):

    typed_idx = np.flatnonzero(typed)
    screen = typed & (m22s > -2.0 * r)
    screen_idx = np.flatnonzero(screen)
    outside_idx = np.flatnonzero(typed & ~screen)

    if outside_idx.size <= OUTSIDE_AUDIT_N:
        audit_idx = outside_idx
    else:
        audit_idx = rng_audit.choice(outside_idx, size=OUTSIDE_AUDIT_N, replace=False)

    integrated_samples = samples[integrate_idx]
    start = time.time()
    st, xe, ye, te = integrate_batch(r, integrated_samples, q4_on, DT, TAU_MAX)
    runtime = time.time() - start
    pos = {int(idx): j for j, idx in enumerate(integrate_idx)}


    screen_status = np.array([st[pos[int(i)]] for i in screen_idx], dtype=np.int8)
    audit_status = np.array([st[pos[int(i)]] for i in audit_idx], dtype=np.int8)
    screen_failure = np.isin(screen_status, np.array([EXIT, OTHER], dtype=np.int8))
    screen_stall = screen_status == STALL
    audit_failure = np.isin(audit_status, np.array([EXIT, OTHER], dtype=np.int8))


    failure_indicator = np.zeros(samples.shape[0], dtype=np.float64)
    stall_indicator = np.zeros(samples.shape[0], dtype=np.float64)

    est, se, psi = estimate_and_if(weights, failure_indicator)
    stall_est, stall_se, stall_psi = estimate_and_if(weights, stall_indicator)
    failure_weights = weights * failure_indicator


    stab_idx = integrate_idx[: min(STABILITY_N, integrate_idx.size)]
    st1 = np.array([st[pos[int(i)]] for i in stab_idx], dtype=np.int8)
    st2, _, _, _ = integrate_batch(r, samples[stab_idx], q4_on, DT / 2.0, TAU_MAX)
    nonstall = (st1 != STALL) & (st2 != STALL)
    disagreements = int(np.sum(nonstall & (st1 != st2)))
    denom = int(np.sum(nonstall))
    disagreement_rate = 0.0 if denom == 0 else disagreements / denom


    fail_idx = screen_idx[screen_failure]
    fail_w = weights[fail_idx]
    q_over_r = samples[fail_idx, 0] / r if fail_idx.size else np.array([])
    m22_over_r = m22s[fail_idx] / r if fail_idx.size else np.array([])


    other_local_positions = np.flatnonzero(st == OTHER)
    others = []

            "x": float(xe[j]),
            "y": float(ye[j]),
            "tau": float(te[j]),
            "weight": float(weights[global_idx]),
        })


    counts = {
        "capture": int(np.sum(screen_status == CAPTURE)),
        "chart_exit": int(np.sum(screen_status == EXIT)),
        "other_stationary": int(np.sum(screen_status == OTHER)),
        "stall": int(np.sum(screen_status == STALL)),
    }
    audit_counts = {
        "capture": int(np.sum(audit_status == CAPTURE)),
        "chart_exit": int(np.sum(audit_status == EXIT)),
        "other_stationary": int(np.sum(audit_status == OTHER)),
        "stall": int(np.sum(audit_status == STALL)),
    }
    return {

        "screen_n": int(screen_idx.size),
        "outside_typed_n": int(outside_idx.size),
        "screen_counts": counts,
        "outside_audit_n": int(audit_idx.size),
        "outside_audit_counts": audit_counts,
        "outside_screen_failures": int(audit_failure.sum()),
        "screened_weighted_failure": est,
        "screened_weighted_failure_se": se,
        "screened_weighted_stall": stall_est,
        "screened_weighted_stall_se": stall_se,

        "stability_audit_n": int(stab_idx.size),

def paired_comparison(a, b, label):
    est_a = a["screened_weighted_failure"]
    est_b = b["screened_weighted_failure"]
    psi_diff = b["_psi"] - a["_psi"]
    n = psi_diff.size
    diff = est_b - est_a
    se = float(np.std(psi_diff, ddof=1) / math.sqrt(n))
    lo, hi = diff - 1.96 * se, diff + 1.96 * se
    ratio = None
    log_ratio_se = None
    ratio_ci = None
    admissible = (
        a["raw_screen_failures"] >= 20
        and b["raw_screen_failures"] >= 20
        and a["failure_weight_ess"] >= 10
        and b["failure_weight_ess"] >= 10
        and est_a > 0 and est_b > 0
    )
    if admissible:
        psi_log = b["_psi"] / est_b - a["_psi"] / est_a
        log_ratio_se = float(np.std(psi_log, ddof=1) / math.sqrt(n))
        lr = math.log(est_b / est_a)
        ratio = est_b / est_a
        ratio_ci = [math.exp(lr - 1.96 * log_ratio_se), math.exp(lr + 1.96 * log_ratio_se)]
    return {
        "label": label,
        "difference": diff,
        "difference_se": se,
        "difference_95ci": [lo, hi],
        "ratio_admissible": admissible,
        "ratio": ratio,
        "log_ratio_se": log_ratio_se,
        "ratio_95ci": ratio_ci,
    }
