def planar_cov_exact(alpha: tuple[int, int], beta: tuple[int, int]) -> sp.Expr:
    x = sp.Symbol("x")
    nx = alpha[0] + beta[0]
    ny = alpha[1] + beta[1]
    hx = sp.hermite_prob(nx, x).subs(x, 0)
    hy = sp.hermite_prob(ny, x).subs(x, 0)
    return (
        (-1) ** (beta[0] + beta[1])
        * (-1) ** nx * hx
        * (-1) ** ny * hy
    )




def limiting_blocks() -> tuple[sp.Matrix, sp.Matrix, sp.Matrix, sp.Matrix]:
    g0 = sp.Matrix([
        [planar_cov_exact(a, b) for b in CONFLUENT_CONSTRAINTS]
        for a in CONFLUENT_CONSTRAINTS

        [planar_cov_exact(d, e) for e in JET_DERIVATIVES]
        for d in JET_DERIVATIVES

    sigma0 = sp.simplify(s0 - c0 * g0.inv() * c0.T)

def k_derivative(n: int, t: sp.Expr) -> sp.Expr:
    x = sp.Symbol("x")
    return sp.diff(sp.exp(-x**2 / 2), x, n).subs(x, t)




def planar_cov_expr(
    px: sp.Expr,
    alpha: tuple[int, int],
    qx: sp.Expr,
    beta: tuple[int, int],
) -> sp.Expr:

    return (
        (-1) ** (beta[0] + beta[1])
        * k_derivative(alpha[0] + beta[0], dx)
        * k_derivative(alpha[1] + beta[1], sp.Integer(0))
    )

def series_coefficients(matrix: sp.Matrix, r: sp.Symbol, max_power: int) -> list[sp.Matrix]:
    coeffs = [sp.zeros(matrix.rows, matrix.cols) for _ in range(max_power + 1)]
    for i in range(matrix.rows):
        for j in range(matrix.cols):
            series = sp.series(matrix[i, j], r, 0, max_power + 1).removeO().expand()
            for p in range(max_power + 1):
                coeffs[p][i, j] = sp.simplify(series.coeff(r, p))
    return coeffs




def exponential_term(term: sp.Expr, r: sp.Symbol):
    exps = list(term.atoms(sp.exp))
    if not exps:
        return None
    if len(exps) != 1:
        raise RuntimeError(f"unexpected exponential term: {term}")
    ex = exps[0]
    a = sp.simplify(-ex.args[0] / r**2)
    monomial = sp.simplify(term / ex)
    c, p = monomial.as_coeff_exponent(r)
    if sp.simplify(monomial - c * r**p) != 0:
        raise RuntimeError(f"nonmonomial prefactor: {term}")
    return sp.Rational(c), int(p), sp.Rational(a)




def tail_bound(expr: sp.Expr, r: sp.Symbol, cutoff_power: int, radius: mp.mpf) -> mp.mpf:
    total = mp.mpf("0")
    for term in sp.expand(expr).as_ordered_terms():
        info = exponential_term(term, r)
        if info is None:
            continue
        c, p, a = info
        j = max(0, (cutoff_power - p) // 2 + 1)
        while p + 2 * j <= cutoff_power:
            j += 1
        x = mp.mpf(str(a)) * radius**2
        first = x**j / mp.factorial(j)
        ratio = x / (j + 1)
        remainder = first / (1 - ratio)
        total += abs(mp.mpf(str(c))) * radius**p * remainder
    return total




def max_entry_deviation(
    matrix: sp.Matrix,
    coeffs: list[sp.Matrix],
    r: sp.Symbol,
    radius: mp.mpf,
    cutoff_power: int = 6,
) -> tuple[mp.mpf, mp.mpf]:
    largest = mp.mpf("0")
    largest_tail = mp.mpf("0")
    for i in range(matrix.rows):
        for j in range(matrix.cols):
            bound = mp.mpf("0")
            for p in range(1, cutoff_power + 1):
                if coeffs[p][i, j] != 0:
                    bound += abs(mp.mpf(str(coeffs[p][i, j]))) * radius**p
            tail = tail_bound(matrix[i, j], r, cutoff_power, radius)
            bound += tail
            largest = max(largest, bound)
            largest_tail = max(largest_tail, tail)
    return largest, largest_tail
