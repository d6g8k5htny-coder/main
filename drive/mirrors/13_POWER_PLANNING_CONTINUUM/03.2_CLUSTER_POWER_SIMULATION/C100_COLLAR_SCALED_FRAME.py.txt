#!/usr/bin/env python3
"""
C100 exact collar scaled-frame and collision certificate.

Object
------
Count any additional critical point y in

    C_r = B(M,2r) union B(S,2r)

under the typed six-pin maximum-saddle pair-Palm law.

Use scaled midpoint coordinates

    y = r (X,Y),   M=(-1/2,0),   S=(1/2,0).

Generic third-gradient law
--------------------------
Under the coalesced six-pin Gaussian law, the free leading jets are

    q=f_yy, a=f_xxy, w=f_xyy, z=f_yyy,

with means (-b,0,0,0) and covariance diag(2,2,2,6).

The scaled third gradient is

    G1 = f_x(rX,rY)/r^2
       -> X^2-1/4 + a X Y + (w/2)Y^2,

    G2 = f_y(rX,rY)/r
       -> q Y.

Hence

    det Cov(G1,G2)=Y^4(4X^2+Y^2),

and the raw gradient density has the exact r^-3 scale.

Triple determinant law
----------------------
Condition G1=G2=0. Writing q=r Q+..., the leading cubic field is

    P = X^3/3-X/4-1/12
        +(a/2)(X^2-1/4)Y
        +(Q/2)Y^2
        +(w/2)XY^2
        +(z/6)Y^3.

Solving P_X=P_Y=0 at the third station gives a,Q in terms of X,Y,w,z.
All three physical Hessians equal r times their cubic Hessians plus O(r^2),
so their determinant product has the r^6 scale.

Near either conditioned point, with

    (X,Y)=M+rho(c,s) or S+rho(c,s),

the exact cubic determinant product has a factor rho^2. This cancels the
rho^-2 collision singularity of the third-gradient density.

Axis charts
-----------
Away from the pair pins, Y->0 has Gaussian suppression

    exp[-(X^2-1/4)^2/(Y^2(4X^2+Y^2))].

At X=0 the stronger transverse suppression is exp[-1/(16Y^4)].
Near M or S the collision-scaled density has exp[-c^2/s^2]-class angular
suppression. The exact pair axis itself contains no additional cubic station.

Outputs
-------
C100_COLLAR_SCALED_FRAME.json
"""

from __future__ import annotations

import json
from pathlib import Path
import sympy as sp

BASE = Path(__file__).resolve().parent
OUTPUT = BASE / "C100_COLLAR_SCALED_FRAME.json"


def generic_gradient_frame() -> dict:
    X, Y, b = sp.symbols("X Y b", real=True)
    q, a, w = sp.symbols("q a w", real=True)

    mean = sp.Matrix([
        X**2 - sp.Rational(1, 4),
        -b * Y,
    ])
    covariance = sp.diag(
        Y**2 * (4 * X**2 + Y**2) / 2,
        2 * Y**2,
    )
    determinant = sp.factor(covariance.det())
    exponent = sp.factor(
        sp.Rational(1, 2)
        * (mean.T * covariance.inv() * mean)[0]
    )

    expected_determinant = Y**4 * (4 * X**2 + Y**2)
    expected_exponent = (
        (X**2 - sp.Rational(1, 4)) ** 2
        / (Y**2 * (4 * X**2 + Y**2))
        + b**2 / 4
    )

    checks = {
        "determinant": (
            sp.simplify(determinant - expected_determinant) == 0
        ),
        "exponent": sp.simplify(exponent - expected_exponent) == 0,
    }
    if not all(checks.values()):
        raise AssertionError(checks)

    rho, c, s = sp.symbols(
        "rho c s", positive=True, real=True
    )
    collision_M = {
        X: -sp.Rational(1, 2) + rho * c,
        Y: rho * s,
    }
    collision_S = {
        X: sp.Rational(1, 2) + rho * c,
        Y: rho * s,
    }

    collision_density_scale_M = sp.factor(
        determinant.subs(collision_M) / rho**4
    )
    collision_density_scale_S = sp.factor(
        determinant.subs(collision_S) / rho**4
    )
    collision_exponent_M = sp.factor(
        expected_exponent.subs(collision_M)
    )
    collision_exponent_S = sp.factor(
        expected_exponent.subs(collision_S)
    )

    transverse_exponent = sp.factor(
        expected_exponent.subs(X, 0)
    )
    transverse_determinant = sp.factor(
        expected_determinant.subs(X, 0)
    )

    return {
        "mean": [sp.sstr(item) for item in mean],
        "covariance": [
            [sp.sstr(covariance[i, j]) for j in range(2)]
            for i in range(2)
        ],
        "determinant": sp.sstr(determinant),
        "half_quadratic_exponent": sp.sstr(exponent),
        "raw_density": (
            "r^-3 [2*pi*|Y|^2*sqrt(4X^2+Y^2)]^-1 "
            "exp[-(X^2-1/4)^2/(Y^2(4X^2+Y^2))-b^2/4]"
        ),
        "collision_M": {
            "det_over_rho4": sp.sstr(collision_density_scale_M),
            "exponent": sp.sstr(collision_exponent_M),
        },
        "collision_S": {
            "det_over_rho4": sp.sstr(collision_density_scale_S),
            "exponent": sp.sstr(collision_exponent_S),
        },
        "transverse_X0": {
            "determinant": sp.sstr(transverse_determinant),
            "exponent": sp.sstr(transverse_exponent),
            "suppression": "exp[-1/(16 Y^4)] times exp[-b^2/4]",
        },
        "checks": checks,
    }


def cubic_triple_determinants() -> dict:
    x, y = sp.symbols("x y")
    X, Y = sp.symbols("X Y", real=True, nonzero=True)
    a, Q, w, z = sp.symbols("a Q w z", real=True)

    P = (
        x**3 / 3
        - x / 4
        - sp.Rational(1, 12)
        + a * (x**2 - sp.Rational(1, 4)) * y / 2
        + Q * y**2 / 2
        + w * x * y**2 / 2
        + z * y**3 / 6
    )

    station_equations = [
        sp.diff(P, x).subs({x: X, y: Y}),
        sp.diff(P, y).subs({x: X, y: Y}),
    ]
    solution = sp.solve(
        station_equations,
        (a, Q),
        dict=True,
        simplify=True,
    )[0]
    a_solution = sp.factor(solution[a])
    Q_solution = sp.factor(solution[Q])

    hessian = sp.hessian(P, (x, y)).subs(solution)

    def determinant_at(point_x, point_y):
        return sp.factor(
            hessian.subs({x: point_x, y: point_y}).det()
        )

    det_M = determinant_at(-sp.Rational(1, 2), 0)
    det_S = determinant_at(sp.Rational(1, 2), 0)
    det_X = determinant_at(X, Y)
    product = sp.factor(det_M * det_S * det_X)

    rho, c, s = sp.symbols(
        "rho c s", real=True
    )
    near_M = sp.factor(
        product.subs({
            X: -sp.Rational(1, 2) + rho * c,
            Y: rho * s,
        })
    )
    near_S = sp.factor(
        product.subs({
            X: sp.Rational(1, 2) + rho * c,
            Y: rho * s,
        })
    )

    near_M_reduced = sp.factor(near_M / rho**2)
    near_S_reduced = sp.factor(near_S / rho**2)

    checks = {
        "station_x": (
            sp.simplify(station_equations[0].subs(solution)) == 0
        ),
        "station_y": (
            sp.simplify(station_equations[1].subs(solution)) == 0
        ),
        "near_M_has_rho2": (
            sp.simplify(near_M - rho**2 * near_M_reduced) == 0
        ),
        "near_S_has_rho2": (
            sp.simplify(near_S - rho**2 * near_S_reduced) == 0
        ),
        "near_M_reduced_finite_generic": (
            not near_M_reduced.has(sp.zoo)
        ),
        "near_S_reduced_finite_generic": (
            not near_S_reduced.has(sp.zoo)
        ),
    }
    if not all(checks.values()):
        raise AssertionError(checks)

    # Leading collision factors at rho=0. These are rational in the angular
    # chart and polynomial in the remaining Gaussian cubic jets.
    lead_M = sp.factor(near_M_reduced.subs(rho, 0))
    lead_S = sp.factor(near_S_reduced.subs(rho, 0))

    return {
        "normal_form": sp.sstr(P),
        "third_station_solution": {
            "a": sp.sstr(a_solution),
            "Q": sp.sstr(Q_solution),
        },
        "determinants": {
            "M": sp.sstr(det_M),
            "S": sp.sstr(det_S),
            "third": sp.sstr(det_X),
        },
        "triple_product": sp.sstr(product),
        "physical_scaling": (
            "det H_M det H_S det H_y = "
            "r^6 [det D2P(M) det D2P(S) det D2P(X,Y)] + O(r^7)"
        ),
        "collision_factorization": {
            "near_M": "product = rho^2 * R_M(rho,c,s,w,z)",
            "near_S": "product = rho^2 * R_S(rho,c,s,w,z)",
            "R_M_at_rho0": sp.sstr(lead_M),
            "R_S_at_rho0": sp.sstr(lead_S),
        },
        "checks": checks,
    }


def transverse_chart() -> dict:
    Y, a, Q, w, z = sp.symbols(
        "Y a Q w z", real=True, nonzero=True
    )
    # At X=0, P_X=0 gives w=1/(2Y^2).
    w_solution = 1 / (2 * Y**2)
    # P_Y=0 gives Q=a/(8Y)-zY/2.
    Q_solution = a / (8 * Y) - z * Y / 2

    return {
        "station_solution": {
            "w": sp.sstr(w_solution),
            "Q": sp.sstr(Q_solution),
        },
        "Gaussian_cost": (
            "Since w is a centered variance-2 jet, the station constraint "
            "w=1/(2Y^2) gives exp[-1/(16Y^4)] suppression."
        ),
        "conclusion": (
            "Every polynomial singularity of the determinant formula at X=0 "
            "is integrable after the transverse Gaussian cost."
        ),
    }


def axis_chart() -> dict:
    X = sp.symbols("X", real=True)
    derivative = X**2 - sp.Rational(1, 4)

    return {
        "leading_pair_axis_gradient": sp.sstr(derivative),
        "zeros": ["X=-1/2", "X=1/2"],
        "conclusion": (
            "The leading cubic has no additional pair-axis critical point. "
            "Away from the two pins, the generic density has "
            "exp[-(X^2-1/4)^2/(Y^2(4X^2+Y^2))] suppression as Y->0. "
            "At the pins, the collision factor rho^2 closes the chart."
        ),
    }


def main() -> None:
    gradient = generic_gradient_frame()
    determinants = cubic_triple_determinants()
    transverse = transverse_chart()
    axis = axis_chart()
