#!/usr/bin/env python3
"""
C098 Gamma maximum-window count benchmark.

Purpose
-------
The exact deterministic reduction is

    Gamma_r subset {N_max((b-ell,b)) >= 1}
    P(Gamma_r) <= E_pair-Palm N_max((b-ell,b)).

This instrument evaluates the exact determinant-weighted six-pin Kac--Rice
integrand at a frozen station atlas. It is a diagnostic, not a continuum
certificate.

Law
---
1. Condition the exact periodized Bargmann--Fock field on
       f(M)=b, grad f(M)=0,
       f(S)=b-ell, grad f(S)=0.
2. Reweight by
       W_MS = |det H_M det H_S|
              1_{H_M negative definite}
              1_{H_S saddle}.
3. At a third point x, the maximum-window intensity is

   integral_{b-ell}^b p_{(f,grad f)(x)|six pins}(u,0)
       E[W_MS |det H_x| 1_{H_x negative definite}
         | six pins, f(x)=u, grad f(x)=0]
       / E[W_MS | six pins] du.

The u-integral uses three-node Gauss--Legendre quadrature. Gaussian covariance
matrices are assembled at 80 decimal digits. Eigenpairs for simulation are
computed with mpmath before conversion to float64. Monte Carlo uses frozen
seeds and common random numbers across the three u nodes.

Outputs
-------
C098_GAMMA_MAXCOUNT_BENCHMARK.json
"""

from __future__ import annotations

import importlib.util
import json
import math
from pathlib import Path
import sys
from typing import Iterable, Mapping, Sequence

import mpmath as mp
import numpy as np

BASE = Path(__file__).resolve().parent
COVARIANCE_MODULE = BASE / "C095_SIX_PIN_COVARIANCE_FACTORIZATION.py"
OUTPUT = BASE / "C098_GAMMA_MAXCOUNT_BENCHMARK.json"

mp.mp.dps = 80

L = mp.mpf("24")
B = mp.mpf("1.2")
RHO_MAX_B = mp.mpf("0.043685")
N_DEN = 500_000
N_NUM = 80_000
SEED_DEN = 2026071801
SEED_NUM_BASE = 2026071810

Multi = tuple[int, int]
Point = tuple[mp.mpf, mp.mpf]
Functional = tuple[Point, Multi]


def load_covariance_module():
    spec = importlib.util.spec_from_file_location(
        "c098_covariance_target", COVARIANCE_MODULE
    )
    if spec is None or spec.loader is None:
        raise RuntimeError("cannot import six-pin covariance module")
    module = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = module
    spec.loader.exec_module(module)
    return module


cm = load_covariance_module()


def covariance_matrix(
    rows: Sequence[Functional],
    cols: Sequence[Functional],
) -> mp.matrix:
    matrix = mp.matrix(len(rows), len(cols))
    for i, (point_a, alpha) in enumerate(rows):
        for j, (point_b, beta) in enumerate(cols):
            matrix[i, j] = cm.cov(point_a, alpha, point_b, beta, L)
    return matrix


def conditional_affine(
    targets: Sequence[Functional],
    conditions: Sequence[Functional],
) -> tuple[mp.matrix, mp.matrix]:
    """
    Return A, Sigma such that

        E[target | conditions=value] = A value,
        Cov(target | conditions) = Sigma.
    """
    gram = covariance_matrix(conditions, conditions)
    target_condition = covariance_matrix(targets, conditions)
    target_target = covariance_matrix(targets, targets)
    inverse = mp.inverse(gram)
    affine = target_condition * inverse
    covariance = target_target - affine * target_condition.T
    covariance = (covariance + covariance.T) / 2
    return affine, covariance


def mp_vector(values: Sequence[mp.mpf | float | int]) -> mp.matrix:
    return mp.matrix([mp.mpf(value) for value in values])


def affine_mean(affine: mp.matrix, values: Sequence[mp.mpf | float | int]) -> mp.matrix:
    return affine * mp_vector(values)


def gaussian_log_density(
    mean: mp.matrix,
    covariance: mp.matrix,
    value: Sequence[mp.mpf | float | int],
) -> mp.mpf:
    delta = mp_vector(value) - mean
    dimension = len(value)
    determinant = mp.det(covariance)
    if determinant <= 0:
        raise ValueError(f"nonpositive density covariance determinant: {determinant}")
    quadratic = (delta.T * mp.inverse(covariance) * delta)[0]
    return -mp.mpf("0.5") * (
        dimension * mp.log(2 * mp.pi)
        + mp.log(determinant)
        + quadratic
    )


def mp_eigen_factor(covariance: mp.matrix) -> tuple[np.ndarray, np.ndarray, dict]:
    eigenvalues_mp, eigenvectors_mp = mp.eigsy(covariance)
    eigenvalues = np.array(
        [float(eigenvalues_mp[i]) for i in range(eigenvalues_mp.rows)],
        dtype=float,
    )
    eigenvectors = np.array(
        [
            [float(eigenvectors_mp[i, j]) for j in range(eigenvectors_mp.cols)]
            for i in range(eigenvectors_mp.rows)
        ],
        dtype=float,
    )
    if np.any(eigenvalues < -1e-30):
        raise ValueError(f"materially negative covariance eigenvalue: {eigenvalues.min()}")
    eigenvalues = np.maximum(eigenvalues, 0.0)
    factor = eigenvectors * np.sqrt(eigenvalues)
    diagnostics = {
        "minimum_eigenvalue": mp.nstr(eigenvalues_mp[0], 35),
        "maximum_eigenvalue": mp.nstr(
            eigenvalues_mp[eigenvalues_mp.rows - 1], 35
        ),
        "condition_ratio_min_over_max": mp.nstr(
            eigenvalues_mp[0] / eigenvalues_mp[eigenvalues_mp.rows - 1],
            35,
        ),
    }
    return factor, eigenvalues, diagnostics


def hessian_determinants(samples: np.ndarray) -> tuple[np.ndarray, ...]:
    if samples.shape[1] not in (6, 9):
        raise ValueError("expected two or three 2x2 Hessians")
    determinants = []
    for offset in range(0, samples.shape[1], 3):
        hxx = samples[:, offset]
        hxy = samples[:, offset + 1]
        hyy = samples[:, offset + 2]
        determinants.append(hxx * hyy - hxy * hxy)
    return tuple(determinants)


def pair_weight(samples: np.ndarray) -> np.ndarray:
    det_m, det_s = hessian_determinants(samples)
    typed = (det_m > 0) & (samples[:, 0] < 0) & (det_s < 0)
    return np.abs(det_m * det_s) * typed


def triple_max_weight(samples: np.ndarray) -> np.ndarray:
    det_m, det_s, det_x = hessian_determinants(samples)
    typed = (
        (det_m > 0)
        & (samples[:, 0] < 0)
        & (det_s < 0)
        & (det_x > 0)
        & (samples[:, 6] < 0)
    )
    return np.abs(det_m * det_s * det_x) * typed


def monte_carlo_weight(
    mean: mp.matrix,
    factor: np.ndarray,
    standard_normals: np.ndarray,
    *,
    triple: bool,
) -> dict:
    mean_float = np.array([float(mean[i]) for i in range(mean.rows)])
    samples = mean_float + standard_normals @ factor.T
    weights = triple_max_weight(samples) if triple else pair_weight(samples)
    positive = weights > 0
    average = float(np.mean(weights))
    standard_error = float(np.std(weights, ddof=1) / math.sqrt(weights.size))
    return {
        "mean": average,
        "standard_error": standard_error,
        "relative_standard_error": (
            None if average == 0 else standard_error / average
        ),
        "positive_weight_fraction": float(np.mean(positive)),
    }


def pair_geometry(r: mp.mpf) -> tuple[Point, Point, mp.mpf]:
    ell = r**3 / 6
    maximum = (-r / 2, mp.mpf("0"))
    saddle = (r / 2, mp.mpf("0"))
    return maximum, saddle, ell


def six_pin_design(r: mp.mpf) -> tuple[list[Functional], list[mp.mpf], list[Functional]]:
    maximum, saddle, ell = pair_geometry(r)
    pins: list[Functional] = []
    hessians: list[Functional] = []
    for point in (maximum, saddle):
        pins.extend(
            [
                (point, (0, 0)),
                (point, (1, 0)),
                (point, (0, 1)),
            ]
        )
        hessians.extend(
            [
                (point, (2, 0)),
                (point, (1, 1)),
                (point, (0, 2)),
            ]
        )
    values = [B, mp.mpf("0"), mp.mpf("0"), B - ell, mp.mpf("0"), mp.mpf("0")]
    return pins, values, hessians


def station_atlas(r: mp.mpf) -> list[dict]:
    maximum, saddle, _ = pair_geometry(r)
    root_two = mp.sqrt(2)

    def record(name: str, point: Point, family: str) -> dict:
        d_m = mp.sqrt((point[0] - maximum[0]) ** 2 + (point[1] - maximum[1]) ** 2)
        d_s = mp.sqrt((point[0] - saddle[0]) ** 2 + (point[1] - saddle[1]) ** 2)
        return {
            "name": name,
            "family": family,
            "point": point,
            "distance_to_M_over_r": d_m / r,
            "distance_to_S_over_r": d_s / r,
            "outside_two_r_collars": bool(d_m > 2 * r and d_s > 2 * r),
        }

    stations = [
        record("east_of_S_2p25r", (saddle[0] + mp.mpf("2.25") * r, 0), "pair-scaled"),
        record("north_of_S_2p25r", (saddle[0], mp.mpf("2.25") * r), "pair-scaled"),
        record(
            "midpoint_diag_3r",
            (3 * r / root_two, 3 * r / root_two),
            "pair-scaled",
        ),
        record("midpoint_north_8r", (0, 8 * r), "pair-scaled"),
        record("north_0p25", (0, mp.mpf("0.25")), "fixed-near"),
        record("north_0p5", (0, mp.mpf("0.5")), "fixed-near"),
        record("north_1", (0, mp.mpf("1")), "fixed-near"),
        record("north_2", (0, mp.mpf("2")), "fixed-near"),
        record("east_1", (mp.mpf("1"), 0), "fixed-near"),
        record("east_2", (mp.mpf("2"), 0), "fixed-near"),
        record("east_5", (mp.mpf("5"), 0), "exterior"),
        record("north_5", (0, mp.mpf("5")), "exterior"),
        record("diag_5_5", (mp.mpf("5"), mp.mpf("5")), "exterior"),
        record("east_8", (mp.mpf("8"), 0), "exterior"),
    ]
    return stations


def evaluate_rung(r_text: str, rung_index: int) -> dict:
    r = mp.mpf(r_text)
    maximum, saddle, ell = pair_geometry(r)
    pins, pin_values, pair_hessians = six_pin_design(r)

    pair_affine, pair_covariance = conditional_affine(pair_hessians, pins)
    pair_mean = affine_mean(pair_affine, pin_values)
    pair_factor, _, pair_spectrum = mp_eigen_factor(pair_covariance)

    rng_den = np.random.default_rng(SEED_DEN + rung_index)
    z_den = rng_den.standard_normal((N_DEN, 6))
    denominator = monte_carlo_weight(
        pair_mean,
        pair_factor,
        z_den,
        triple=False,
    )

    source_anchor = {
        "0.05": 0.00808,
        "0.025": 0.00202046,
    }.get(r_text)
    if source_anchor is not None:
        denominator["source_anchor"] = source_anchor
        denominator["relative_difference_from_source_anchor"] = (
            denominator["mean"] / source_anchor - 1
        )

    quadrature_nodes = [
        -mp.sqrt(mp.mpf("3") / 5),
        mp.mpf("0"),
        mp.sqrt(mp.mpf("3") / 5),
    ]
    quadrature_weights = [
        mp.mpf("5") / 9,
        mp.mpf("8") / 9,
        mp.mpf("5") / 9,
    ]

    rows = []
    for station_index, station in enumerate(station_atlas(r)):
        x = station["point"]
        observed = [
            (x, (0, 0)),
            (x, (1, 0)),
            (x, (0, 1)),
        ]
        observed_affine, observed_covariance = conditional_affine(observed, pins)
        observed_mean = affine_mean(observed_affine, pin_values)

        all_conditions = pins + observed
        triple_hessians = pair_hessians + [
            (x, (2, 0)),
            (x, (1, 1)),
            (x, (0, 2)),
        ]
        triple_affine, triple_covariance = conditional_affine(
            triple_hessians, all_conditions
        )
        triple_factor, _, triple_spectrum = mp_eigen_factor(triple_covariance)

        rng = np.random.default_rng(
            SEED_NUM_BASE + 100 * rung_index + station_index
        )
        z = rng.standard_normal((N_NUM, 9))

        node_rows = []
        integrated_intensity = mp.mpf("0")
        integrated_mc_error = mp.mpf("0")

        # Map t in [-1,1] to u in [b-ell,b]:
        # u = b - ell/2 + (ell/2)t, du=(ell/2)dt.
        for t, weight in zip(quadrature_nodes, quadrature_weights):
            u = B - ell / 2 + (ell / 2) * t
            log_density = gaussian_log_density(
                observed_mean,
                observed_covariance,
                [u, 0, 0],
            )
            density = mp.exp(log_density)

            condition_values = pin_values + [u, mp.mpf("0"), mp.mpf("0")]
            triple_mean = affine_mean(triple_affine, condition_values)
            numerator = monte_carlo_weight(
                triple_mean,
                triple_factor,
                z,
                triple=True,
            )
            lambda_u = density * mp.mpf(numerator["mean"]) / mp.mpf(
                denominator["mean"]
            )
            lambda_u_se = density * mp.mpf(
                numerator["standard_error"]
            ) / mp.mpf(denominator["mean"])

            integrated_intensity += (ell / 2) * weight * lambda_u
            # Conservative linear aggregation; common random numbers make this
            # an overstatement rather than an independence assumption.
            integrated_mc_error += (ell / 2) * abs(weight) * lambda_u_se

            node_rows.append(
                {
                    "t": mp.nstr(t, 25),
                    "u": mp.nstr(u, 35),
                    "log_density_f_grad": mp.nstr(log_density, 35),
                    "density_f_grad": mp.nstr(density, 35),
                    "weighted_hessian_moment": numerator,
                    "lambda_u": mp.nstr(lambda_u, 35),
                    "lambda_u_mc_se_numerator_only": mp.nstr(
                        lambda_u_se, 25
                    ),
                }
            )

        coefficient_per_area = integrated_intensity / (r**3)
        coefficient_error = integrated_mc_error / (r**3)
        stationary_per_area = RHO_MAX_B / 6

        rows.append(
            {
                "name": station["name"],
                "family": station["family"],
                "point": [
                    mp.nstr(x[0], 30),
                    mp.nstr(x[1], 30),
                ],
                "distance_to_M_over_r": mp.nstr(
                    station["distance_to_M_over_r"], 25
                ),
                "distance_to_S_over_r": mp.nstr(
                    station["distance_to_S_over_r"], 25
                ),
                "outside_two_r_collars": station["outside_two_r_collars"],
                "observed_conditional_mean": [
                    mp.nstr(observed_mean[i], 35)
                    for i in range(observed_mean.rows)
                ],
                "observed_covariance_determinant": mp.nstr(
                    mp.det(observed_covariance), 35
                ),
                "triple_hessian_spectrum": triple_spectrum,
                "quadrature": node_rows,
                "window_intensity": mp.nstr(integrated_intensity, 35),
                "coefficient_per_area_r3": mp.nstr(
                    coefficient_per_area, 35
                ),
                "coefficient_per_area_mc_se_numerator_only": mp.nstr(
                    coefficient_error, 25
                ),
                "ratio_to_stationary_maximum_benchmark": mp.nstr(
                    coefficient_per_area / stationary_per_area, 30
                ),
            }
        )

    exterior_rows = [row for row in rows if row["family"] == "exterior"]
    outside_collar_rows = [
        row for row in rows if row["outside_two_r_collars"]
    ]

    return {
        "r": r_text,
        "ell": mp.nstr(ell, 35),
        "pair_points": {
            "M": [mp.nstr(maximum[0], 30), mp.nstr(maximum[1], 30)],
            "S": [mp.nstr(saddle[0], 30), mp.nstr(saddle[1], 30)],
        },
        "denominator_pair_weight": denominator,
        "pair_hessian_spectrum": pair_spectrum,
        "station_rows": rows,
        "summary": {
            "exterior_ratio_min": min(
                float(row["ratio_to_stationary_maximum_benchmark"])
                for row in exterior_rows
            ),
            "exterior_ratio_max": max(
                float(row["ratio_to_stationary_maximum_benchmark"])
                for row in exterior_rows
            ),
            "outside_collar_coefficient_per_area_min": min(
                float(row["coefficient_per_area_r3"])
                for row in outside_collar_rows
            ),
            "outside_collar_coefficient_per_area_max": max(
                float(row["coefficient_per_area_r3"])
                for row in outside_collar_rows
            ),
            "largest_station": max(
                outside_collar_rows,
                key=lambda row: float(row["coefficient_per_area_r3"]),
            )["name"],
        },
    }
