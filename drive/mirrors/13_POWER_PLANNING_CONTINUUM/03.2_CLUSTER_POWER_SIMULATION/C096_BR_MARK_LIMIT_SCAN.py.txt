#!/usr/bin/env python3
"""
C096 BR-MARK limiting Gaussian mark-law reduction and diagnostic scan.

This instrument studies the Gaussian (unweighted) part of the six-pin
critical-value mark law for two additional candidate critical points whose
separation d tends to zero.

Let m be their midpoint, e their separation direction, n=e^\perp, and define

    A_d = (f(m-de/2)+f(m+de/2))/2,
    Z_d = (f(m+de/2)-f(m-de/2))/d^3.

Conditioning the two candidate gradients to vanish forces, in the d->0 limit,

    D_e f(m) = 0,
    D_e^2 f(m) = 0,
    D_n f(m) = 0,
    D_e D_n f(m) = 0,

and Taylor expansion gives

    A_d -> f(m),
    Z_d -> -(1/12) D_e^3 f(m).

The script computes the exact limiting Gaussian covariance of (A,Z) under the
original typed-pair *linear six-pin residual covariance* for the exact
periodized BF field. It scans fixed and pair-scaled midpoint families to
identify where a uniform Gaussian density bound is plausible or degenerates.

This does NOT include:
  * saddle Hessian typing of the candidate pair,
  * determinant Palm reweighting,
  * factorial-point-process repulsion,
  * a certified continuum interval enclosure.

Accordingly the output grade is DERIVED-EXACT-REDUCTION plus
MEASURED-DIAGNOSTIC-SCAN, not BR-MARK closure.
"""

from __future__ import annotations

from dataclasses import dataclass
import json
import math
from pathlib import Path
from typing import Dict, Iterable, Tuple

import mpmath as mp

BASE = Path(__file__).resolve().parent
mp.mp.dps = 90

Multi = Tuple[int, int]


def probabilists_hermite(n: int, x: mp.mpf) -> mp.mpf:
    if n == 0:
        return mp.mpf("1")
    if n == 1:
        return x
    hm2 = mp.mpf("1")
    hm1 = x
    for k in range(1, n):
        h = x * hm1 - k * hm2
        hm2, hm1 = hm1, h
    return hm1


def theta_derivative(
    t: mp.mpf,
    L: mp.mpf,
    order: int,
    shells: int = 5,
) -> mp.mpf:
    return mp.fsum(
        (-1) ** order
        * probabilists_hermite(order, t + j * L)
        * mp.e ** (-(t + j * L) ** 2 / 2)
        for j in range(-shells, shells + 1)
    )


def k1(t: mp.mpf, L: mp.mpf, order: int = 0) -> mp.mpf:
    return theta_derivative(t, L, order) / theta_derivative(
        mp.mpf("0"), L, 0
    )


@dataclass(frozen=True)
class Functional:
    point: Tuple[mp.mpf, mp.mpf]
    coeffs: Tuple[Tuple[Multi, mp.mpf], ...]
    name: str

    @staticmethod
    def partial(
        point: Tuple[mp.mpf, mp.mpf],
        multi: Multi,
        name: str,
        scale: mp.mpf = mp.mpf("1"),
    ) -> "Functional":
        return Functional(point, ((multi, scale),), name)


def directional(
    point: Tuple[mp.mpf, mp.mpf],
    e: Tuple[mp.mpf, mp.mpf],
    n: Tuple[mp.mpf, mp.mpf],
    order_e: int,
    order_n: int,
    name: str,
    scale: mp.mpf = mp.mpf("1"),
) -> Functional:
    coeffs: Dict[Multi, mp.mpf] = {}
    for i in range(order_e + 1):
        ce = (
            mp.binomial(order_e, i)
            * e[0] ** i
            * e[1] ** (order_e - i)
        )
        for j in range(order_n + 1):
            cn = (
                mp.binomial(order_n, j)
                * n[0] ** j
                * n[1] ** (order_n - j)
            )
            multi = (i + j, order_e + order_n - i - j)
            coeffs[multi] = coeffs.get(multi, mp.mpf("0")) + scale * ce * cn
    return Functional(
        point,
        tuple(sorted(coeffs.items())),
        name,
    )


def covariance(a: Functional, b: Functional, L: mp.mpf) -> mp.mpf:
    dx = a.point[0] - b.point[0]
    dy = a.point[1] - b.point[1]
    total = mp.mpf("0")
    for alpha, ca in a.coeffs:
        for beta, cb in b.coeffs:
            order_x = alpha[0] + beta[0]
            order_y = alpha[1] + beta[1]
            total += (
                ca
                * cb
                * (-1) ** (beta[0] + beta[1])
                * k1(dx, L, order_x)
                * k1(dy, L, order_y)
            )
    return total


def covariance_matrix(
    rows: Iterable[Functional],
    cols: Iterable[Functional],
    L: mp.mpf,
) -> mp.matrix:
    rows = tuple(rows)
    cols = tuple(cols)
    out = mp.matrix(len(rows), len(cols))
    for i, left in enumerate(rows):
        for j, right in enumerate(cols):
            out[i, j] = covariance(left, right, L)
    return out


def conditional_covariance(
    variables: Tuple[Functional, ...],
    constraints: Tuple[Functional, ...],
    L: mp.mpf,
) -> Tuple[mp.matrix, mp.matrix]:
    vv = covariance_matrix(variables, variables, L)
    vc = covariance_matrix(variables, constraints, L)
    cc = covariance_matrix(constraints, constraints, L)
    result = vv - vc * mp.inverse(cc) * vc.T
    return result, cc


def six_pin_constraints(r: mp.mpf) -> Tuple[Functional, ...]:
    M = (-r / 2, mp.mpf("0"))
    S = (r / 2, mp.mpf("0"))
    rows = []
    for label, point in (("M", M), ("S", S)):
        rows.extend(
            [
                Functional.partial(point, (0, 0), f"f_{label}"),
                Functional.partial(point, (1, 0), f"fx_{label}"),
                Functional.partial(point, (0, 1), f"fy_{label}"),
            ]
        )
    return tuple(rows)


def limit_mark_covariance(
    r: mp.mpf,
    midpoint: Tuple[mp.mpf, mp.mpf],
    angle: mp.mpf,
    L: mp.mpf,
) -> dict:
    e = (mp.cos(angle), mp.sin(angle))
    n = (-mp.sin(angle), mp.cos(angle))

    A = Functional.partial(midpoint, (0, 0), "A_limit")
    Z = directional(
        midpoint, e, n, 3, 0, "Z_limit", scale=-mp.mpf(1) / 12
    )
    local_constraints = (
        directional(midpoint, e, n, 1, 0, "D_e"),
        directional(midpoint, e, n, 2, 0, "D_e2"),
        directional(midpoint, e, n, 0, 1, "D_n"),
        directional(midpoint, e, n, 1, 1, "D_eD_n"),
    )
    constraints = six_pin_constraints(r) + local_constraints
    cov, gram = conditional_covariance((A, Z), constraints, L)

    var_a = cov[0, 0]
    var_z = cov[1, 1]
    cov_az = cov[0, 1]
    determinant = mp.det(cov)
    correlation = cov_az / mp.sqrt(var_a * var_z)
    density_max = (
        mp.inf
        if determinant <= 0
        else 1 / (2 * mp.pi * mp.sqrt(determinant))
    )
    gram_eigenvalues = mp.eigsy(gram, eigvals_only=True)
    cov_eigenvalues = mp.eigsy(cov, eigvals_only=True)

    M = (-r / 2, mp.mpf("0"))
    S = (r / 2, mp.mpf("0"))
    dM = mp.sqrt((midpoint[0] - M[0]) ** 2 + midpoint[1] ** 2)
    dS = mp.sqrt((midpoint[0] - S[0]) ** 2 + midpoint[1] ** 2)

    return {
        "r": mp.nstr(r, 30),
        "midpoint": [mp.nstr(midpoint[0], 30), mp.nstr(midpoint[1], 30)],
        "angle": mp.nstr(angle, 30),
        "distance_to_M_over_r": mp.nstr(dM / r, 30),
        "distance_to_S_over_r": mp.nstr(dS / r, 30),
        "var_A": mp.nstr(var_a, 40),
        "var_Z": mp.nstr(var_z, 40),
        "cov_AZ": mp.nstr(cov_az, 40),
        "correlation": mp.nstr(correlation, 40),
        "det_cov_AZ": mp.nstr(determinant, 40),
        "density_max": mp.nstr(density_max, 40),
        "min_cov_eigenvalue": mp.nstr(cov_eigenvalues[0], 40),
        "min_constraint_gram_eigenvalue": mp.nstr(
            gram_eigenvalues[0], 40
        ),
    }


def main() -> None:
    L = mp.mpf("24")
    rungs = [
        mp.mpf("0.05"),
        mp.mpf("0.025"),
        mp.mpf("0.0125"),
        mp.mpf("0.00625"),
    ]
    angles = [
        mp.mpf("0"),
        mp.pi / 4,
        mp.pi / 2,
        3 * mp.pi / 4,
    ]

    # Pair-scaled midpoints outside conservative radius-2 collars.
    scaled_midpoints = [
        (mp.mpf("-3"), mp.mpf("0")),
        (mp.mpf("3"), mp.mpf("0")),
        (mp.mpf("0"), mp.mpf("2.25")),
        (mp.mpf("0"), mp.mpf("-2.25")),
        (mp.mpf("-2"), mp.mpf("2")),
        (mp.mpf("2"), mp.mpf("2")),
        (mp.mpf("-3"), mp.mpf("1")),
        (mp.mpf("3"), mp.mpf("1")),
    ]
    # Fixed physical midpoints probing the non-collapsing exterior.
    fixed_midpoints = [
        (mp.mpf("0.25"), mp.mpf("0")),
        (mp.mpf("0"), mp.mpf("0.25")),
        (mp.mpf("0.5"), mp.mpf("0.5")),
        (mp.mpf("1"), mp.mpf("0")),
        (mp.mpf("0"), mp.mpf("1")),
        (mp.mpf("2"), mp.mpf("1")),
    ]

    rows = []
    failures = []
    for r in rungs:
        for u, v in scaled_midpoints:
            midpoint = (r * u, r * v)
            for angle in angles:
                try:
                    row = limit_mark_covariance(r, midpoint, angle, L)
                    row["family"] = "pair-scaled"
                    row["scaled_midpoint"] = [str(u), str(v)]
                    rows.append(row)
                except Exception as exc:
                    failures.append(
                        {
                            "family": "pair-scaled",
                            "r": str(r),
                            "scaled_midpoint": [str(u), str(v)],
                            "angle": mp.nstr(angle, 30),
                            "exception": repr(exc),
                        }
                    )
        for midpoint in fixed_midpoints:
            for angle in angles:
                try:
                    row = limit_mark_covariance(r, midpoint, angle, L)
                    row["family"] = "fixed-physical"
                    rows.append(row)
                except Exception as exc:
                    failures.append(
                        {
                            "family": "fixed-physical",
                            "r": str(r),
                            "midpoint": [str(midpoint[0]), str(midpoint[1])],
                            "angle": mp.nstr(angle, 30),
                            "exception": repr(exc),
                        }
                    )

    finite_rows = [
        row for row in rows if row["density_max"] not in {"inf", "+inf"}
    ]
    by_family = {}
    for family in ("pair-scaled", "fixed-physical"):
        subset = [row for row in finite_rows if row["family"] == family]
        by_family[family] = {
            "count": len(subset),
            "max_density": max(
                subset, key=lambda row: mp.mpf(row["density_max"])
            ) if subset else None,
            "min_det": min(
                subset, key=lambda row: mp.mpf(row["det_cov_AZ"])
            ) if subset else None,
            "min_var_A": min(
                subset, key=lambda row: mp.mpf(row["var_A"])
            ) if subset else None,
            "min_var_Z": min(
                subset, key=lambda row: mp.mpf(row["var_Z"])
            ) if subset else None,
        }

    report = {
        "cycle": "C096",
        "result_id": "BR_MARK_GAUSSIAN_LIMIT_REDUCTION",
        "grades": {
            "limit_reduction": "DERIVED-EXACT",
            "finite_scan": "MEASURED-DIAGNOSTIC",
        },
        "theorem": {
            "constraints": [
                "D_e f(m)=0",
                "D_e^2 f(m)=0",
                "D_n f(m)=0",
                "D_e D_n f(m)=0",
            ],
            "limit_variables": {
                "A": "f(m)",
                "Z": "-D_e^3 f(m)/12",
            },
            "conditioning": (
                "original six linear pair pins plus the four limiting "
                "candidate-gradient jet constraints"
            ),
            "density_formula": (
                "sup density of centered Gaussian (A,Z) is "
                "1/(2*pi*sqrt(det Sigma_AZ))"
            ),
        },
        "scan_design": {
            "rungs": [str(r) for r in rungs],
            "angles": [mp.nstr(angle, 20) for angle in angles],
            "scaled_midpoints": [
                [str(x), str(y)] for x, y in scaled_midpoints
            ],
            "fixed_midpoints": [
                [str(x), str(y)] for x, y in fixed_midpoints
            ],
            "model": "exact normalized periodized BF, L=24",
        },
        "summary_by_family": by_family,
        "rows": rows,
        "failures": failures,
        "adjudication": {
            "Gaussian_six_pin_limit_law": "DERIVED-EXACT",
            "uniform_compact_density_candidate": (
                "SUPPORTED on the scanned finite families only"
            ),
            "BR_MARK": "OPEN",
            "remaining": [
                "validated continuum interval enclosure over midpoint/orientation/r",
                "finite-d remainder from the d->0 jet reduction",
                "candidate saddle Hessian typing",
                "two-saddle determinant Palm weight",
                "factorial repulsion constant BR-REP",
                "source-to-target mark-transfer grade",
            ],
        },
    }

    output = BASE / "C096_BR_MARK_LIMIT_SCAN.json"
    output.write_text(json.dumps(report, indent=2), encoding="utf-8")
    print(
        json.dumps(
            {
                "result_id": report["result_id"],
                "summary_by_family": by_family,
                "failure_count": len(failures),
                "row_count": len(rows),
                "adjudication": report["adjudication"],
            },
            indent=2,
        )
    )


if __name__ == "__main__":
    main()
