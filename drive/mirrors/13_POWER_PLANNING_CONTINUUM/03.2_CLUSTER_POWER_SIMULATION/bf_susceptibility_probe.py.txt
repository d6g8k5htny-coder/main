#!/usr/bin/env python3
"""
bf_susceptibility_probe.py

Stage-0 finite-volume probe for the Q0-X program.

It samples a unit-variance periodized Bargmann--Fock field on a square torus,
conditions only on f(center) = b + delta, and measures the periodic
four-neighbor excursion cluster {f >= b} containing the pin.

This is NOT the canonical pair-Palm q_L experiment. It does not impose
gradient constraints, Hessian typing, a nearby saddle pin, determinant Palm
weights, or persistence pairing. Its purpose is to test the prerequisite
hypothesis that a basic conditioned excursion susceptibility stabilizes with
volume away from critical height.
"""

from __future__ import annotations

import argparse
from collections import deque
import csv
import json
import math
from pathlib import Path
from typing import Dict, Iterable, List, Sequence, Tuple

import numpy as np


def periodized_bf_spectrum(side_length: float, grid_size: int) -> np.ndarray:
    """Normalized discrete Fourier coefficients of the periodized BF kernel."""
    dx = side_length / grid_size
    frequencies = 2.0 * np.pi * np.fft.fftfreq(grid_size, d=dx)
    xi, eta = np.meshgrid(frequencies, frequencies, indexing="ij")
    coefficients = np.exp(-0.5 * (xi * xi + eta * eta))
    coefficients /= coefficients.sum()
    return coefficients


def covariance_grid(side_length: float, grid_size: int) -> np.ndarray:
    coefficients = periodized_bf_spectrum(side_length, grid_size)
    return np.fft.ifft2((grid_size * grid_size) * coefficients).real


def sample_field(
    side_length: float,
    grid_size: int,
    rng: np.random.Generator,
    sqrt_spectrum: np.ndarray,
) -> np.ndarray:
    white = rng.normal(size=(grid_size, grid_size))
    white_hat = np.fft.fft2(white)
    return np.fft.ifft2(white_hat * (grid_size * sqrt_spectrum)).real


def condition_on_center_value(
    field: np.ndarray,
    shifted_covariance: np.ndarray,
    center: Tuple[int, int],
    target: float,
) -> np.ndarray:
    """Exact Gaussian regression for the single linear constraint f(center)=target."""
    return field + shifted_covariance * (target - field[center])


def periodic_cluster_stats(
    active: np.ndarray,
    start: Tuple[int, int],
) -> Dict[str, float]:
    """Cluster size, torus radius, and winding detection via lifted BFS coordinates."""
    n, m = active.shape
    if n != m:
        raise ValueError("the probe expects a square grid")
    if not active[start]:
        return {"size": 0.0, "radius_grid": 0.0, "wrap": 0.0}

    lifts: Dict[Tuple[int, int], Tuple[int, int]] = {start: (0, 0)}
    queue = deque([start])
    wrap_x = False
    wrap_y = False
    steps = ((1, 0), (-1, 0), (0, 1), (0, -1))

    while queue:
        i, j = queue.popleft()
        li, lj = lifts[(i, j)]
        for di, dj in steps:
            ni = (i + di) % n
            nj = (j + dj) % n
            if not active[ni, nj]:
                continue
            proposed = (li + di, lj + dj)
            old = lifts.get((ni, nj))
            if old is None:
                lifts[(ni, nj)] = proposed
                queue.append((ni, nj))
            else:
                difference = (proposed[0] - old[0], proposed[1] - old[1])
                wrap_x = wrap_x or difference[0] != 0
                wrap_y = wrap_y or difference[1] != 0

    si, sj = start
    max_radius_squared = 0
    for i, j in lifts:
        di = min((i - si) % n, (si - i) % n)
        dj = min((j - sj) % n, (sj - j) % n)
        max_radius_squared = max(max_radius_squared, di * di + dj * dj)

    return {
        "size": float(len(lifts)),
        "radius_grid": math.sqrt(max_radius_squared),
        "wrap": float(wrap_x or wrap_y),
    }


def run_probe(
    side_lengths: Sequence[float],
    levels: Sequence[float],
    *,
    dx: float,
    samples: int,
    delta: float,
    seed: int,
) -> Tuple[List[Dict[str, float]], List[Dict[str, float]]]:
    rng = np.random.default_rng(seed)
    summary: List[Dict[str, float]] = []
    raw: List[Dict[str, float]] = []

    for side_length in side_lengths:
        grid_size = int(round(side_length / dx))
        if grid_size % 2:
            grid_size += 1
        actual_dx = side_length / grid_size
        spectrum = periodized_bf_spectrum(side_length, grid_size)
        sqrt_spectrum = np.sqrt(spectrum)
        covariance = covariance_grid(side_length, grid_size)
        center = (grid_size // 2, grid_size // 2)
        shifted_covariance = np.roll(
            np.roll(covariance, center[0], axis=0), center[1], axis=1
        )
        if abs(float(shifted_covariance[center]) - 1.0) > 1e-10:
            raise RuntimeError("covariance normalization failed")

        for level in levels:
            areas: List[float] = []
            radii: List[float] = []
            wraps: List[float] = []
            for sample_index in range(samples):
                field = sample_field(
                    side_length, grid_size, rng, sqrt_spectrum
                )
                field = condition_on_center_value(
                    field, shifted_covariance, center, level + delta
                )
                stats = periodic_cluster_stats(field >= level, center)
                area = stats["size"] * actual_dx * actual_dx
                radius = stats["radius_grid"] * actual_dx
                wrap = stats["wrap"]
                areas.append(area)
                radii.append(radius)
                wraps.append(wrap)
                raw.append(
                    {
                        "L": side_length,
                        "N": grid_size,
                        "dx": actual_dx,
                        "b": level,
                        "sample": sample_index,
                        "delta": delta,
                        "pin_height": level + delta,
                        "area": area,
                        "radius": radius,
                        "wrap": wrap,
                    }
                )

            area_array = np.asarray(areas)
            radius_array = np.asarray(radii)
            wrap_array = np.asarray(wraps)
            standard_error = float(area_array.std(ddof=1) / math.sqrt(samples))
            mean_area = float(area_array.mean())
            summary.append(
                {
                    "L": side_length,
                    "N": grid_size,
                    "dx": actual_dx,
                    "b": level,
                    "samples": samples,
                    "delta": delta,
                    "pin_height": level + delta,
                    "mean_area": mean_area,
                    "se_area": standard_error,
                    "ci95_low": mean_area - 1.96 * standard_error,
                    "ci95_high": mean_area + 1.96 * standard_error,
                    "median_area": float(np.median(area_array)),
                    "p90_area": float(np.quantile(area_array, 0.9)),
                    "mean_radius": float(radius_array.mean()),
                    "wrap_rate": float(wrap_array.mean()),
                    "area_over_L2": mean_area / (side_length * side_length),
                }
            )

    return summary, raw


def write_csv(path: Path, rows: Iterable[Dict[str, float]]) -> None:
    rows = list(rows)
    if not rows:
        raise ValueError("cannot write an empty result")
    with path.open("w", newline="", encoding="utf-8") as handle:
        writer = csv.DictWriter(handle, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)


def parse_float_list(value: str) -> List[float]:
    return [float(item.strip()) for item in value.split(",") if item.strip()]


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--Ls", default="12,24,48")
    parser.add_argument("--levels", default="1.2,0.8,0.4,0.2")
    parser.add_argument("--dx", type=float, default=0.25)
    parser.add_argument("--samples", type=int, default=300)
    parser.add_argument("--delta", type=float, default=0.25)
    parser.add_argument("--seed", type=int, default=20260714)
    parser.add_argument(
        "--output-prefix",
        default="/mnt/data/bf_susceptibility_probe",
    )
    args = parser.parse_args()

    if args.dx <= 0 or args.samples < 2:
        raise ValueError("dx must be positive and samples must be at least 2")

    summary, raw = run_probe(
        parse_float_list(args.Ls),
        parse_float_list(args.levels),
        dx=args.dx,
        samples=args.samples,
        delta=args.delta,
        seed=args.seed,
    )

    prefix = Path(args.output_prefix)
    write_csv(prefix.with_name(prefix.name + "_results.csv"), summary)
    write_csv(prefix.with_name(prefix.name + "_raw.csv"), raw)
    prefix.with_name(prefix.name + "_metadata.json").write_text(
        json.dumps(
            {
                "model": "unit-variance discrete periodized Bargmann--Fock",
                "conditioning": "f(center)=b+delta only",
                "connectivity": "periodic four-neighbor",
                "Ls": parse_float_list(args.Ls),
                "levels": parse_float_list(args.levels),
                "dx": args.dx,
                "samples": args.samples,
                "delta": args.delta,
                "seed": args.seed,
                "canonical_q_estimate": False,
            },
            indent=2,
        ),
        encoding="utf-8",
    )
    print(json.dumps(summary, indent=2))


if __name__ == "__main__":
    main()
