#!/usr/bin/env python3
"""
bf_two_critical_value_gap.py

Exact two-critical-value gap law for the planar Bargmann--Fock field.

Take two points at separation d on the x-axis and condition on both gradients
being zero. For the two field values V_- and V_+, define

    A = (V_- + V_+)/2,
    Z = (V_+ - V_-)/d^3.

The script derives the exact conditional covariance and its small-d expansion.
This is the mark-side input for a corrected near-diagonal Bonferroni argument:
the average critical value has O(1) variance while the value gap has O(d^3)
standard deviation.

This is an exact planar reference calculation. Extending it uniformly to the
six-pin pair-conditioned field requires a Schur/GCJA transfer certificate.

Output:
    /mnt/data/bf_two_critical_value_gap_report.json
"""

from __future__ import annotations

import json
import math
from pathlib import Path

import mpmath as mp
import sympy as sp


def symbolic_formulas():
    d = sp.symbols("d", positive=True)
    k = sp.exp(-d**2 / 2)
    c = (1 - d**2) * k

    value_cov = sp.Matrix([[1, k], [k, 1]])
    value_gradient_cov = sp.Matrix([[0, -d * k], [d * k, 0]])
    gradient_cov = sp.Matrix([[1, c], [c, 1]])

    conditional = sp.simplify(
        value_cov
        - value_gradient_cov
        * gradient_cov.inv()
        * value_gradient_cov.T
    )

    diagonal = sp.simplify(conditional[0, 0])
    off_diagonal = sp.simplify(conditional[0, 1])
    var_average = sp.simplify((diagonal + off_diagonal) / 2)
    var_gap = sp.simplify(2 * (diagonal - off_diagonal))
    var_scaled_gap = sp.simplify(var_gap / d**6)

    return d, conditional, var_average, var_gap, var_scaled_gap


def numeric_variances(d_value: float):
    d = mp.mpf(str(d_value))
    e = mp.e ** (d * d)
    denominator = d**4 - 2 * d**2 - e + 1
    diagonal = (d**4 - d**2 - e + 1) / denominator
    off = (
        (-d**2 - e + 1)
        * mp.e ** (-d**2 / 2)
        / denominator
    )
    var_average = (diagonal + off) / 2
    var_gap = 2 * (diagonal - off)
    var_scaled_gap = var_gap / d**6
    joint_density_max = 1 / (
        2 * mp.pi * mp.sqrt(var_average * var_scaled_gap)
    )
    average_density_max = 1 / mp.sqrt(
        2 * mp.pi * var_average
    )
    return {
        "d": float(d),
        "var_average": float(var_average),
        "var_gap_over_d6": float(var_scaled_gap),
        "max_joint_density_A_Z": float(joint_density_max),
        "max_average_density": float(average_density_max),
    }


def main():
    mp.mp.dps = 100
    d, conditional, var_average, var_gap, var_scaled_gap = symbolic_formulas()

    points = [
        numeric_variances(value)
        for value in [0.1, 0.05, 0.025, 0.01, 0.001]
    ]

    report = {
        "cycle": "C091",
        "field": "planar Bargmann--Fock, K(x)=exp(-|x|^2/2)",
        "conditioning": (
            "gradients vanish at two points separated by d on the x-axis"
        ),
        "definitions": {
            "A": "(f(-d/2,0)+f(d/2,0))/2",
            "Z": "(f(d/2,0)-f(-d/2,0))/d^3",
        },
        "conditional_value_covariance": str(conditional),
        "exact_variances": {
            "Var_A": str(var_average),
            "Var_value_gap": str(var_gap),
            "Var_Z": str(var_scaled_gap),
            "Cov_A_Z": "0 exactly by reflection symmetry",
        },
        "small_d_series": {
            "Var_A": str(sp.series(var_average, d, 0, 10)),
            "Var_value_gap": str(sp.series(var_gap, d, 0, 14)),
            "Var_Z": str(sp.series(var_scaled_gap, d, 0, 8)),
        },
        "limits": {
            "Var_A": 2 / 3,
            "Var_Z": 1 / 24,
            "max_joint_density_A_Z": 3 / math.pi,
            "max_average_density": math.sqrt(3) / (2 * math.sqrt(math.pi)),
        },
        "numeric_diagnostics": points,
        "window_geometry": {
            "statement": (
                "For a height interval W of width ell, the set of "
                "(A,Z) for which both values lie in W has area ell^2/d^3. "
                "Also its A-projection has length ell."
            ),
            "probability_bound": (
                "P(V_-,V_+ in W | two gradients zero) <= "
                "min(B_A*ell, B_AZ*ell^2/d^3), where B_A and B_AZ "
                "are density suprema."
            ),
        },
        "adjudication": {
            "exact_planar_mark_law": "CLOSED",
            "six_pin_uniform_transfer": "OPEN",
            "importance": (
                "The critical-value difference scale is d^3, so the "
                "near-diagonal Bonferroni proof must split at "
                "d approximately ell^(1/3), rather than factor two "
                "height-window lengths uniformly down to d=0."
            ),
        },
    }

    output = Path("/mnt/data/bf_two_critical_value_gap_report.json")
    output.write_text(json.dumps(report, indent=2), encoding="utf-8")
    print(json.dumps(report, indent=2))


if __name__ == "__main__":
    main()
