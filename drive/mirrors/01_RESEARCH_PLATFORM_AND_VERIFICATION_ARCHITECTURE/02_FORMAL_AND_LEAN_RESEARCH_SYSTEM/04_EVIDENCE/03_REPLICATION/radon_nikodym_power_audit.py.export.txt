#!/usr/bin/env python3
"""Symbolic power audit for the LS-DER-023 boundary mechanism."""
from sympy import symbols, Matrix, factor


r, kappa, q, a, w = symbols("r kappa q a w", nonzero=True)
s = q / r


HM = r * Matrix([[-kappa, -a/2], [-a/2, s-w/2]])
HS = r * Matrix([[ kappa,  a/2], [ a/2, s+w/2]])


detM = factor(HM.det())
detS = factor(HS.det())
generic_product = factor(detM * detS)


# Boundary layer q=r*s0 with s0=O(1)
s0 = symbols("s0")
boundary_product = factor(generic_product.subs(q, r*s0))


print("det(H_M) =", detM)
print("det(H_S) =", detS)
print("generic determinant product =", generic_product)
print("boundary q=r*s0 product =", boundary_product)
print("Generic q=O(1): product has leading r^2.")
print("Boundary q=O(r): product has leading r^4.")
print("With Z=Theta(r^2), dP/dQ is O(1) generically and O(r^2) on the boundary.")
print("Boundary width dq=r ds contributes one more r, yielding O(r^3) Palm mass.")
