#!/usr/bin/env python3
"""Fresh Start 2.0 mathematical integrity gate.


This tool performs deterministic metadata and symbolic-exponent checks.
It may return PASS, HOLD, or NO. It never proves a theorem, fabricates
independence, edits Drive, or overrides an exact mathematical review.


Input: one JSON package card. See REQUIRED_KEYS and example at bottom.
"""
from __future__ import annotations


import argparse
import json
import re
from dataclasses import dataclass
from fractions import Fraction
from pathlib import Path
from typing import Any


REQUIRED_KEYS = {
    "object_id", "version", "statement_sha256", "law", "coordinates",
    "domain", "dependencies", "power_ledger", "review_paths",
    "vulnerability_adjudication", "reopening_rule", "status_surfaces",
}


VULNERABILITY_PATTERNS = {
    "VUL-001": [
        r"/\s*[A-Za-z_][A-Za-z0-9_]*",
        r"divide\s+by",
        r"quotient",
    ],
    "VUL-002": [
        r"sup[^\n]{0,80}\+[^\n]{0,80}sup",
        r"supremum[^\n]{0,80}sum",
    ],
    "VUL-003": [
        r"finite\s+grid",
        r"sampled\s+points?[^\n]{0,80}(all|every|continuum)",
        r"\d+\s+points?[^\n]{0,80}(box|continuum)",
    ],
    "VUL-004": [
        r"fixed[- ]r[^\n]{0,100}uniform",
        r"fixed[- ]q[^\n]{0,100}fixed[- ]s",
        r"asymptotic[^\n]{0,100}finite[- ]r",
    ],
    "VUL-005": [
        r"planar[^\n]{0,100}exact[- ]torus",
        r"finite[- ]Q4[^\n]{0,100}full[- ]field",
        r"conditioned[^\n]{0,100}Palm",
    ],
    "VUL-006": [
        r"restricted\s+normalizer",
        r"good[- ]event\s+denominator",
        r"adjacency[- ]conditioned\s+denominator",
    ],
    "VUL-007": [
        r"superseded",
        r"historical\s+hash",
        r"mutable\s+header",
    ],
    "VUL-008": [
        r"monotone",
        r"hardest\s+endpoint",
        r"worst[- ]case\s+endpoint",
    ],
    "VUL-009": [
        r"physical\s+coordinates?[^\n]{0,100}normalized",
        r"volume[^\n]{0,100}Jacobian",
        r"change\s+of\s+variables",
    ],
    "VUL-010": [
        r"independent[^\n]{0,100}same[- ]family",
        r"copied\s+(code|implementation)",
        r"multiple\s+sessions?[^\n]{0,100}independent",
    ],
    "VUL-011": [
        r"(width|radius|slope|witness)[^\n]{0,100}/",
        r"axial\s+gate",
        r"denominator[^\n]{0,100}(zero|vanish|positive)",
    ],
    "VUL-012": [
        r"typical\s+samples?",
        r"empirical[^\n]{0,100}(uniform|every|all)",
        r"finite[^\n]{0,100}(uniform\s+theorem|for\s+all)",
    ],
}


@dataclass
class Finding:
    gate: str
    verdict: str
    detail: str




def as_fraction(value: Any) -> Fraction:
    if isinstance(value, int):
        return Fraction(value, 1)
    if isinstance(value, str):
        return Fraction(value)
    if isinstance(value, float):
        return Fraction(str(value))
    raise TypeError(f"unsupported exponent {value!r}")




def check_required(card: dict[str, Any]) -> Finding:
    missing = sorted(REQUIRED_KEYS - card.keys())
    if missing:
        return Finding("IDENTITY", "HOLD", f"missing keys: {missing}")
    if not re.fullmatch(r"[0-9a-fA-F]{64}", str(card["statement_sha256"])):
        return Finding("IDENTITY", "NO", "statement_sha256 is not 64 hex characters")
    return Finding("IDENTITY", "PASS", "required identity fields present")




def check_domain(card: dict[str, Any]) -> Finding:
    domain = card.get("domain", {})
    required = {"fixed", "varying", "quantifiers", "hardest_endpoint", "bridge_required"}
    missing = sorted(required - domain.keys())
    if missing:
        return Finding("DOMAIN", "HOLD", f"missing domain fields: {missing}")
    if domain.get("uniform_claim") and not domain.get("monotonicity_proof"):
        return Finding("DOMAIN", "HOLD", "uniform claim lacks monotonicity proof or full interval certificate")
    cited = set(card.get("dependencies", []))
    bridges = set(domain.get("bridges", []))
    incompatible = set(domain.get("incompatible_dependencies", [])) & cited
    if incompatible and not bridges:
        return Finding("DOMAIN", "NO", f"incompatible dependencies without bridge: {sorted(incompatible)}")
    return Finding("DOMAIN", "PASS", "domain and endpoint metadata complete")




def check_power_ledger(card: dict[str, Any]) -> Finding:
    ledger = card.get("power_ledger", {})
    terms = ledger.get("terms", [])
    if not terms:
        return Finding("POWER", "HOLD", "no symbolic power terms")
    total = Fraction(0)
    for term in terms:
        if not {"name", "exponent", "operation", "coordinate_type"} <= term.keys():
            return Finding("POWER", "HOLD", f"incomplete term: {term}")
        exponent = as_fraction(term["exponent"])
        if term["operation"] == "multiply":
            total += exponent
        elif term["operation"] == "divide":
            total -= exponent
        else:
            return Finding("POWER", "NO", f"unknown operation {term['operation']}")
    claimed = as_fraction(ledger.get("claimed_total"))
    if total != claimed:
        return Finding("POWER", "NO", f"computed r^{total}, claimed r^{claimed}")
    if any(not term.get("coordinate_type") for term in terms):
        return Finding("POWER", "HOLD", "untyped coordinate contribution")
    return Finding("POWER", "PASS", f"symbolic total reproduces r^{total}")




def scan_vulnerabilities(card: dict[str, Any]) -> tuple[Finding, list[dict[str, str]]]:
    text = "\n".join([
        str(card.get("statement", "")),
        str(card.get("proof_excerpt", "")),
        json.dumps(card.get("domain", {}), sort_keys=True),
        json.dumps(card.get("power_ledger", {}), sort_keys=True),
    ])
    warnings: list[dict[str, str]] = []
    for signature, patterns in VULNERABILITY_PATTERNS.items():
        for pattern in patterns:
            match = re.search(pattern, text, flags=re.IGNORECASE)
            if match:
                warnings.append({"signature": signature, "match": match.group(0)[:160]})
                break
    adjudication = card.get("vulnerability_adjudication", {})
    unresolved = sorted({w["signature"] for w in warnings if adjudication.get(w["signature"]) != "RESOLVED"})
    if unresolved:
        return Finding("VULNERABILITY", "HOLD", f"unresolved warnings: {unresolved}"), warnings
    return Finding("VULNERABILITY", "PASS", "all triggered signatures adjudicated"), warnings




def check_interfaces(card: dict[str, Any]) -> Finding:
    states = card.get("dependency_states", {})
    unresolved = [dep for dep in card.get("dependencies", []) if states.get(dep) != "TERMINAL"]
    if unresolved:
        if card.get("conclusion_status") == "CONDITIONAL":
            return Finding("INTERFACES", "PASS", f"conditional scope preserves open dependencies: {unresolved}")
        return Finding("INTERFACES", "HOLD", f"open dependencies: {unresolved}")
    return Finding("INTERFACES", "PASS", "all consumed dependencies terminal")




def review_path_class(path: dict[str, Any], author_line: str) -> str:
    if path.get("reviewer_line") == author_line:
        return "CORROBORATION_ONLY"
    if not path.get("identity_reproduced") or not path.get("controls_executed"):
        return "INELIGIBLE"
    if path.get("copied_implementation") or path.get("post_result_exposure"):
        return "ELIGIBLE_WITH_COMPENSATION"
    if path.get("same_signature_conflict"):
        return "ELIGIBLE_WITH_COMPENSATION"
    if not path.get("independent_method"):
        return "CORROBORATION_ONLY"
    return "ELIGIBLE_SOLE"




def check_reviews(card: dict[str, Any]) -> Finding:
    author = str(card.get("author_line", ""))
    classes = [review_path_class(p, author) for p in card.get("review_paths", [])]
    sole = classes.count("ELIGIBLE_SOLE")
    compensated = classes.count("ELIGIBLE_WITH_COMPENSATION")
    formal = bool(card.get("clean_formal_verifier"))
    if sole >= 2 or (sole >= 1 and formal):
        return Finding("INDEPENDENCE", "PASS", f"review classes={classes}; formal={formal}")
    if sole >= 1 and compensated >= 1:
        return Finding("INDEPENDENCE", "PASS", f"one sole path plus compensated distinct path: {classes}")
    return Finding("INDEPENDENCE", "HOLD", f"insufficient independent paths: {classes}; formal={formal}")




def check_state(card: dict[str, Any]) -> Finding:
    surfaces = card.get("status_surfaces", {})
    values = {str(v) for v in surfaces.values() if v not in (None, "")}
    if len(values) > 1:
        return Finding("STATE", "HOLD", f"status surfaces disagree: {surfaces}")
    if card.get("active_contradiction"):
        return Finding("STATE", "NO", "active contradictory evidence")
    if not card.get("reopening_rule"):
        return Finding("STATE", "HOLD", "missing reopening rule")
    return Finding("STATE", "PASS", "status surfaces agree and reopening rule exists")




def evaluate(card: dict[str, Any]) -> dict[str, Any]:
    findings = [
        check_required(card),
        check_domain(card),
        check_power_ledger(card),
    ]
    vuln_finding, warnings = scan_vulnerabilities(card)
    findings.extend([
        vuln_finding,
        check_interfaces(card),
        check_reviews(card),
        check_state(card),
    ])
    verdicts = {f.verdict for f in findings}
    if "NO" in verdicts:
        overall = "NO"
    elif "HOLD" in verdicts:
        overall = "HOLD"
    else:
        overall = "PASS"
    return {
        "overall": overall,
        "findings": [f.__dict__ for f in findings],
        "vulnerability_warnings": warnings,
        "authority": "NONE_BY_SCRIPT_ALONE",
        "note": "PASS establishes metadata/gate eligibility only; mathematical proof status follows exact evidence and review.",
    }




def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("card", type=Path)
    parser.add_argument("--output", type=Path)
    args = parser.parse_args()
    card = json.loads(args.card.read_text(encoding="utf-8"))
    result = evaluate(card)
    payload = json.dumps(result, indent=2, sort_keys=True)
    print(payload)
    if args.output:
        args.output.write_text(payload + "\n", encoding="utf-8")
    return {"PASS": 0, "HOLD": 2, "NO": 3}[result["overall"]]




if __name__ == "__main__":
    raise SystemExit(main())
