from sympy import symbols, Matrix, simplify
r=symbols("r", nonzero=True)
T=Matrix([
[1/2,1/2,0,0,0,0],
[-r**-3,r**-3,-1/(2*r**2),0,-1/(2*r**2),0],
[0,0,1/2,0,1/2,0],
[0,0,-r**-1,0,r**-1,0],
[0,0,0,1/2,0,1/2],
[0,0,0,-r**-1,0,r**-1],
])
assert simplify(T.det()+r**-5)==0
assert simplify((r**3/6)*r**-5*r**2-1/6)==0
print("PASS")
