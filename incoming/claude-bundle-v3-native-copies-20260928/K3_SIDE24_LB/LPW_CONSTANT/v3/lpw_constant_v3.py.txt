def sha256_of(path):
    h_ = hashlib.sha256()
    with open(path, "rb") as fh:
        for chunk in iter(lambda: fh.read(1 << 20), b""):
            h_.update(chunk)
    return h_.hexdigest()

    ck(fr > 0, "expansion argument positive")
    e = 0
    while fr >= Fr(10) ** e:
        e += 1
    while fr < Fr(10) ** (e - 1):
        e -= 1
    k = e - sig
    scaled = fr / Fr(10) ** k
    ck(10 ** (sig - 1) <= scaled < 10 ** sig, "expansion scaling aligned")
    mant = scaled.numerator // scaled.denominator
    ck(Fr(mant) <= scaled < Fr(mant + 1), "exact floor verified")

    return "%s.%se%s%d" % (s[0], s[1:], "-" if e - 1 < 0 else "+", abs(e - 1)), mant, e - 1
