def k1m(n, d):
    key = (n, mpmath.nstr(d, 30))

    return _K1CACHE[key]


def covm(a, b, dx, dy):
    v = k1m(a[0] + b[0], dx) * k1m(a[1] + b[1], dy)
    if (b[0] + b[1]) % 2 == 1:
        v = -v
    return v

def abs_moment(n):
    h = mp.pi / 12
    J = int(mp.ceil(30 / h)) + 1
    tot = mpf(0)
    z1 = mpf(0)
    for j in range(-J, J + 1):
        k = j * h
        w = mpmath.exp(-k**2 / 2)
        tot += (abs(k)**n) * w
        z1 += w
    tail = ce.spectral_tail_1d(n) if n <= 8 else _tail9(n)
    return tot / z1 + tail


def _tail9(n):
    # same geometric-series bound as ce.spectral_tail_1d, for n > 8
    h = mp.pi / 12
    J = int(mp.ceil(30 / h)) + 1
    J0 = J + 1
    k0 = J0 * h
    rho = mpmath.exp(-(2 * J0 + 1) * h * h / 2) * (1 + mpf(1) / J0)**n
    ck(rho < 1, "spectral tail ratio rho >= 1 for n=%d" % n)
    t0 = mpmath.exp(-k0**2 / 2) * k0**n
    return 2 * t0 / (1 - rho)


# ---------------------------------------------------------------------------
# Hermite weights along the corridor (in xi; d/dxi = -r d/dx)
# ---------------------------------------------------------------------------
def hwts(xi):
    t = xi
    return (2*t**3 - 3*t**2 + 1, t**3 - 2*t**2 + t,
            -2*t**3 + 3*t**2, t**3 - t**2)


def hwts_d(xi):
    t = xi
    return (6*t**2 - 6*t, 3*t**2 - 4*t + 1, -6*t**2 + 6*t, 3*t**2 - 2*t)


def hwts_dd(xi):
    t = xi
    return (12*t - 6, 6*t - 4, -12*t + 6, 6*t - 2)


class Kit6:
    """The 6-pin law Q_r (axial pins); pointwise certified moments."""
    def __init__(self, r):
        self.r = r = mpf(r)
        self.ell = r**3 / 6
        self.s = B - self.ell
        self.M = (-r / 2, mpf(0))
        self.S = (r / 2, mpf(0))
        A = mp.zeros(6, 6)
        for a_, pa in enumerate([self.M, self.S]):
            for b_, pb in enumerate([self.M, self.S]):
                for ia in range(3):
                    for ib in range(3):
                        A[3*a_+ia, 3*b_+ib] = covm(IDX[JET6[ia]], IDX[JET6[ib]],
                                                   pa[0]-pb[0], pa[1]-pb[1])
        self.A = A
        self.Ainv = A ** -1
        Ires = A * self.Ainv - mp.eye(6)
        self.pinres = max(abs(Ires[i, j]) for i in range(6) for j in range(6))
        ck(self.pinres < mpf('1e-40'),
           "pin inversion residual %s" % ns(self.pinres, 3))
        self.pv = mp.matrix([B, 0, 0, self.s, 0, 0])
        self.Lp2 = (self.pv.T * self.Ainv * self.pv)[0, 0]
        self.Lp = mpmath.sqrt(self.Lp2)

    def q(self, xi):
        return (self.r / 2 - self.r * xi, mpf(0))

    def cross(self, x, names):
        Bm = mp.zeros(6, len(names))
        for j, P in enumerate([self.M, self.S]):
            for k in range(3):
                for i, dn in enumerate(names):
                    Bm[3*j+k, i] = covm(IDX[JET6[k]], IDX[dn],
                                        P[0]-x[0], P[1]-x[1])
        return Bm

    def law(self, x, names):
        Bm = self.cross(x, names)
        C = mp.zeros(len(names), len(names))
        for i, di in enumerate(names):
            for j, dj in enumerate(names):
                C[i, j] = covm(IDX[di], IDX[dj], mpf(0), mpf(0))
        BmA = Bm.T * self.Ainv
        return BmA * self.pv, C - BmA * Bm

    # -- explicit (unconditional) Hermite-error functionals on the corridor --
    def spec(self, xi, name):
        """Term list (point, jet, weight) of the explicit functional:
        E_k = d^k/dxi^k [(f-H)(q(xi))]  (H the cubic pin-Hermite);
        C_k = E_{k+2}/r^2               (curvature-zone scalings)."""
        r = self.r
        h = hwts(xi)
        hd = hwts_d(xi)
        hdd = hwts_dd(xi)
        hddd = (mpf(12), mpf(6), mpf(-12), mpf(6))
        if name == 'E0':
            return [(self.q(xi), 'f', mpf(1)), (self.S, 'f', -h[0]),
                    (self.S, 'fx', r * h[1]), (self.M, 'f', -h[2]),
                    (self.M, 'fx', r * h[3])]
        if name == 'E1':
            return [(self.q(xi), 'fx', -r), (self.S, 'f', -hd[0]),
                    (self.S, 'fx', r * hd[1]), (self.M, 'f', -hd[2]),
                    (self.M, 'fx', r * hd[3])]
        if name == 'E2':
            return [(self.q(xi), 'fxx', r**2), (self.S, 'f', -hdd[0]),
                    (self.S, 'fx', r * hdd[1]), (self.M, 'f', -hdd[2]),
                    (self.M, 'fx', r * hdd[3])]
        if name == 'E3':
            return [(self.q(xi), 'fxxx', -r**3), (self.S, 'f', -hddd[0]),
                    (self.S, 'fx', r * hddd[1]), (self.M, 'f', -hddd[2]),
                    (self.M, 'fx', r * hddd[3])]
        if name[0] == 'E' and 4 <= int(name[1:]) <= 11:
            k = int(name[1:])
            return [(self.q(xi), 'f' + 'x' * k, (-r)**k)]
        if name[0] == 'C':
            k = int(name[1])
            return [(p, d, w / r**2) for (p, d, w) in self.spec(xi, 'E%d' % (k + 2))]
        ck(False, "unknown spec %s" % name)

    def paircov(self, ta, tb):
        """Cov(spec A, spec B) -- unconditional, exact (total order <= 8)."""
        v = mpf(0)
        for (pa, da, wa) in ta:
            for (pb, db, wb) in tb:
                v += wa * wb * covm(IDX[da], IDX[db], pa[0]-pb[0], pa[1]-pb[1])
        return v

    def pcap(self, ta, tb):
        """|Cov(ta, tb)| bound: exact if within the order cap, else the
        crude spectral-product bound (always an upper bound)."""
        mo = max(IDX[da][0] + IDX[da][1] + IDX[db][0] + IDX[db][1]
                 for (pa, da, wa) in ta for (pb, db, wb) in tb)
        if mo <= 8:
            return self.paircov(ta, tb)
        return mpmath.sqrt(self.crudeW_terms(ta) * self.crudeW_terms(tb))

    def crudeW_terms(self, terms):
        tot = mpf(0)
        for (pa, da, wa) in terms:
            ia = IDX[da]
            for (pb, db, wb) in terms:
                ib = IDX[db]
                tot += abs(wa * wb) * ah_g[ia[0] + ib[0]] * ah_g[ia[1] + ib[1]]
        return tot

    def crudeW(self, xi, name, ah=None):
        """Crude global sup of Var(spec) via |weight| sups and absolute
        spectral moments (valid at every xi)."""
        return self.crudeW_terms(self.spec(xi, name))


def grid_sup4(vals, d1, d2, d3, G4, g):
    """Certified sup of a smooth function on a uniform grid: pointwise value
    vals, derivatives d1..d3 (mixed exact/crude-upper); global |4th
    derivative| bound G4 (crude); spacing g.
    sup <= max(vals + g|d1| + g^2/2 |d2| + g^3/6 |d3|) + g^4 G4/24."""
    return max(v + g * abs(a) + g**2 * abs(b2) / 2 + g**3 * abs(c) / 6
               for v, a, b2, c in zip(vals, d1, d2, d3)) + g**4 * G4 / 24


# derivative coefficients of Var(F): d^n/dxi^n Var(F) = sum_{i+j=n} c^{(n)}_{ij}
# Cov(F_i, F_j); the c coefficients from the Leibniz recursion (symmetric):
VDER = [
    [(0, 0, 1)],
    [(0, 1, 2)],
    [(1, 1, 2), (0, 2, 2)],
    [(0, 3, 2), (1, 2, 6)],
    [(0, 4, 2), (1, 3, 8), (2, 2, 6)],
    [(0, 5, 2), (1, 4, 10), (2, 3, 20)],
    [(0, 6, 2), (1, 5, 12), (2, 4, 30), (3, 3, 20)],

def varsup(kit, base, xis, g):
    """Certified sup of Var(F(xi)) over the continuum grid (spacing g),
    F = spec(base), F_k = spec(base+k).  Uses the pointwise-exact Taylor
    chain through order 8 (all covariances within the order-8 cap, with
    pcap's crude fallback for isolated over-cap pairs) and a crude global
    9th-derivative remainder bound at the g^9/9! level."""
    b0, k0 = base[0], int(base[1])
    terms = [[kit.spec(xi, '%s%d' % (b0, k0 + i)) for i in range(10)]
             for xi in xis]
    C = [[[kit.pcap(t[i], t[j]) if i <= j else None
           for j in range(9)] for i in range(9)] for t in terms]

    def cij(pt, i, j):
        a, b = (i, j) if i <= j else (j, i)
        return C[pt][a][b]

    ders = []
    for n in range(9):
        ders.append([sum(cf * cij(pt, i, j) for (i, j, cf) in VDER[n])
                     for pt in range(len(xis))])
    # crude global 9th-derivative remainder bound (coefficient sums of the
    # Leibniz chain are 2^n; every term bounded by sqrt(W_i W_j))
    W = [max(kit.crudeW_terms(t[i]) for t in terms) for i in range(10)]
    G9 = mpf(512) * max(mpmath.sqrt(W[i] * W[9 - i]) for i in range(10))
    sup = max(sum(g**n * abs(ders[n][pt]) / mpmath.factorial(n)
                  for n in range(9)) for pt in range(len(xis))) \
        + g**9 * G9 / mpmath.factorial(9)
    return sup


def rung(r, scales, gates=True):
    sc_m, sc_s, sc_d, sc_rn = scales
    kit = Kit6(r)
    emit("  r=%s: pin inversion residual %s; Lp^2 = p^T A^-1 p = %s (the"
         " pin-likelihood statistic)" % (ns(r, 6), ns(kit.pinres, 2),
         ns(kit.Lp2, 6)))
    a2, a4, a6, a8 = (ce.moment_a(1), ce.moment_a(2), ce.moment_a(3),
                      ce.moment_a(4))
    global ah_g
    ah_g = ah = {n: abs_moment(n) for n in range(23)}
    # global mean-derivative bounds via the pin-likelihood statistic
    U1 = kit.Lp * mpmath.sqrt(a2)      # |E[f_x|pins]|
    U3 = kit.Lp * mpmath.sqrt(a6)      # |E[f_xxx|pins]|
    U4 = kit.Lp * mpmath.sqrt(a8)      # |E[f_xxxx|pins]|

    # ---- Z1: curvature zone (0, XI0] -------------------------------------
    N1 = 49
    g1 = XI0 / (N1 - 1)
    h2g, h3g = [], []
    xis1 = []
    for i in range(N1):
        z = g1 * i
        xis1.append(z)
        mu, cov = kit.law(kit.q(z), ['fxx', 'fxxx'])
        h2g.append(mu[0])
        h3g.append(abs(mu[1]))
    # certified inf of the mean curvature over the zone (pointwise (h2,h3)
    # plus the global 4th-derivative bound via the pin-likelihood statistic)
    h_min = min(h2g[i] - g1 * r * h3g[i] for i in range(N1)) \
        - g1**2 * r**2 * U4 / 2
    sb1 = mpmath.sqrt(varsup(kit, 'C0', xis1, g1))
    L1 = mpmath.sqrt(varsup(kit, 'C1', xis1, g1))
    D1 = DUDLEY_C * L1 * XI0 * JCONST
    m1 = sc_m * h_min
    sb1s, D1s = sc_s * sb1, sc_d * D1
    kap1 = (m1 - D1s) / sb1s
    ck(kap1 > 0, "Z1 margin swallowed by entropy: kap1=%s" % ns(kap1, 4))
    Q1 = mpmath.exp(-kap1**2 / 2)
    emit("  Z1 curvature zone (0,%s]: h_min=%s (~%s*r), sd_bar=%s (~%s*r^2),"
         " L1=%s, D1=%s, kap1=%s -> Q1<=exp(-%s)"
         % (ns(XI0, 3), ns(h_min, 5), ns(h_min / r, 4), ns(sb1, 4),
            ns(sb1 / r**2, 4), ns(L1, 3), ns(D1, 3), ns(kap1, 5),
            ns(kap1**2 / 2, 5)))

    # ---- Z2: value corridor [XI0, 1] -------------------------------------
    N2 = 121
    g2 = (1 - XI0) / (N2 - 1)
    ug, upg = [], []
    xis2 = []
    for i in range(N2):
        x = XI0 + g2 * i
        xis2.append(x)
        mu, cov = kit.law(kit.q(x), ['f', 'fx'])
        ug.append(mu[0] - kit.s)
        upg.append(abs(-r * mu[1]))          # |d/dxi (mu-s)|, pointwise exact
    # certified continuum inf: pointwise (u, u') + global 2nd-derivative bound
    U2 = r**2 * kit.Lp * mpmath.sqrt(a4)     # |u''| = r^2|E[f_xx|pins]|
    u_star = min(ug[i] - g2 * upg[i] for i in range(N2)) - g2**2 * U2 / 2
    sb2 = mpmath.sqrt(varsup(kit, 'E0', xis2, g2))
    L2 = mpmath.sqrt(varsup(kit, 'E1', xis2, g2))
    D2 = DUDLEY_C * L2 * (1 - XI0) * JCONST
    m2 = sc_m * u_star
    sb2s, D2s = sc_s * sb2, sc_d * D2
    kap2 = (m2 - D2s) / sb2s
    ck(kap2 > 0, "Z2 margin swallowed by entropy: kap2=%s" % ns(kap2, 4))
    Q2 = mpmath.exp(-kap2**2 / 2)
    emit("  Z2 value corridor [%s,1]: u_*=%s (~%s*ell), sd_bar=%s (~%s*ell),"
         " L2=%s, D2=%s, kap2=%s -> Q2<=exp(-%s)"
         % (ns(XI0, 3), ns(u_star, 5), ns(u_star / kit.ell, 4), ns(sb2, 4),
            ns(sb2 / kit.ell, 4), ns(L2, 3), ns(D2, 3), ns(kap2, 5),
            ns(kap2**2 / 2, 5)))

    # ---- assembly ----------------------------------------------------------
    Q = Q1 + Q2
    crn = C_RN * sc_rn
    ck(crn <= C_RN, "consumed RN factor drifted above H4-RN's certified"
         " C_RN = 3.47 (mutation M3 fires)")
    Pb = crn * mpmath.sqrt(Q)
    emit("  assembly: Q1+Q2 = %s;  P_r(B4) <= C_RN*sqrt(Q1+Q2) = %s   vs"
         " r^3 = %s   (ratio %s)"
         % (ns(Q, 3), ns(Pb, 3), ns(r**3, 3), ns(Pb / r**3, 3)))
    if gates:
        ck(Pb <= r**3, "certificate bound fails the o(r^3) gate at r=%s"
           % ns(r, 4))
    return dict(r=r, kap1=kap1, kap2=kap2, Pb=Pb, h_min=h_min, u_star=u_star,
                sb1=sb1, sb2=sb2, D1=D1, D2=D2, Q1=Q1, Q2=Q2)
