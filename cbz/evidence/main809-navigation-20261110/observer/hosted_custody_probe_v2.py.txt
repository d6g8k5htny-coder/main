"""Run exact #812 v2 and its flag mutation with and without a cache prefix."""
import hashlib
import json
import os
from pathlib import Path
import re
import shutil
import subprocess
import sys
import tempfile

if len(sys.argv) != 3:
    raise SystemExit('usage: hosted_custody_probe_v2.py SOURCE OUTPUT')
source, output = (Path(value).resolve() for value in sys.argv[1:])
output.mkdir(parents=True, exist_ok=True)
EXPECTED = {
    'tests/test_periodic_pilot_custody.py': 'c4b449311b922f75c7977918cc92f604e8779ba8',
    'tests/test_periodic_pilot_custody_oserror.py': '95628e4b1cf895ca905fef32b46228ac2d76f6d8',
    'tools/periodic_pilot_custody.py': '024f860f6cec1e603a083bb2f09f32d7e32a2399',
}
for name, expected in EXPECTED.items():
    raw = (source / name).read_bytes()
    got = hashlib.sha1(b'blob ' + str(len(raw)).encode() + b'\0' + raw).hexdigest()
    if got != expected:
        raise ValueError('source blob mismatch: ' + name)
test_path = 'tests/test_periodic_pilot_custody_oserror.py'
original = (source / test_path).read_bytes()
anchor = b"subprocess.run([sys.executable, '-B', '-c', FAULT, mode, str(REPO)]"
replacement = b"subprocess.run([sys.executable, '-c', FAULT, mode, str(REPO)]"
if original.count(anchor) != 1:
    raise ValueError('mutation anchor mismatch')
mutant = original.replace(anchor, replacement, 1)
if hashlib.sha256(mutant).hexdigest() != '38fc25aec0f4b47b9fdb8eda454dd25c87155a9acde585f7211b88744b268955':
    raise ValueError('mutated test source mismatch')
ids = ['test_fault_child_does_not_write_bytecode_to_source_tree',
       'test_output_scandir_oserror_exits_74', 'test_upload_stage_mkdir_oserror_exits_74']
records = []
for optimized in (False, True):
    for prefix_set in (False, True):
        for mutated in (False, True):
            with tempfile.TemporaryDirectory(prefix='cbz-custody-hosted-v2-') as temporary:
                checkout = Path(temporary) / 'source'
                for name in EXPECTED:
                    dst = checkout / name
                    dst.parent.mkdir(parents=True, exist_ok=True)
                    shutil.copyfile(source / name, dst)
                if mutated:
                    (checkout / test_path).write_bytes(mutant)
                env = dict(os.environ)
                env.pop('PYTHONDONTWRITEBYTECODE', None)
                env.pop('PYTHONPYCACHEPREFIX', None)
                if prefix_set:
                    env['PYTHONPYCACHEPREFIX'] = str(Path(temporary) / 'external-cache')
                command = [sys.executable, '-B', '-S'] + (['-O'] if optimized else []) + [
                    '-m', 'unittest', 'discover', '-s', 'tests', '-p',
                    'test_periodic_pilot_custody_oserror.py', '-v']
                run = subprocess.run(command, cwd=checkout, env=env, capture_output=True, timeout=120)
                text = run.stderr.decode('utf-8')
                observed = []
                for line in text.splitlines():
                    match = re.fullmatch(r'(\w+) \(test_periodic_pilot_custody_oserror\.RawOSErrorExitTests\.\w+\) \.\.\. (ok|FAIL)', line)
                    if match:
                        observed.append(match.groups())
                wanted = [(name, 'FAIL' if mutated and i == 0 else 'ok') for i, name in enumerate(ids)]
                if (observed != wanted or run.returncode != (1 if mutated else 0)
                        or 'Ran 3 tests' not in text or 'skipped' in text
                        or not text.rstrip().endswith('FAILED (failures=1)' if mutated else 'OK')):
                    raise ValueError('unexpected test disposition ' + str((optimized, prefix_set, mutated)) + '\n' + text)
                if mutated and ('in test_fault_child_does_not_write_bytecode_to_source_tree' not in text
                                or "self.assertEqual(list(tools.rglob('*.pyc')), [])" not in text
                                or 'periodic_pilot_custody.' + sys.implementation.cache_tag + '.pyc' not in text):
                    raise ValueError('mutation did not trigger the bytecode assertion')
                case = ('optimized' if optimized else 'normal') + ('-prefix' if prefix_set else '-no-prefix') + ('-mutant' if mutated else '-candidate')
                (output / (case + '.stdout')).write_bytes(run.stdout)
                (output / (case + '.stderr')).write_bytes(run.stderr)
                records.append({'case': case, 'returncode': run.returncode, 'results': observed,
                                'stdout_sha256': hashlib.sha256(run.stdout).hexdigest(),
                                'stderr_sha256': hashlib.sha256(run.stderr).hexdigest()})
summary = {'python': sys.version, 'source_git_blobs': EXPECTED,
           'mutation': 'remove only the child -B flag', 'cases': records,
           'scope': 'three focused tests in temporary copies with and without an inherited cache prefix; not full discovery or G5'}
(output / 'observation.json').write_text(json.dumps(summary, indent=2) + '\n')
print(json.dumps(summary, indent=2))
