name: CBZ bounded architecture artifact observation
on:
  push:
    branches: [codex/cbz-architecture-observation-20261110]
permissions:
  actions: read
  contents: read
jobs:
  observe:
    runs-on: ubuntu-24.04
    timeout-minutes: 4
    steps:
      - name: Verify and inspect one retained preparation result
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          set -euo pipefail
          archive="$RUNNER_TEMP/architecture-preparation.zip"
          curl --fail --silent --show-error --location --max-time 60 \
            -H "Authorization: Bearer $GH_TOKEN" \
            -H 'Accept: application/vnd.github+json' \
            'https://api.github.com/repos/d6g8k5htny-coder/main/actions/artifacts/13734016279/zip' \
            --output "$archive"
          python3 -I -B -S - "$archive" <<'PY'
          import hashlib, json, pathlib, re, sys, zipfile
          data = pathlib.Path(sys.argv[1]).read_bytes()
          if len(data) > 2 * 1024 * 1024:
              raise ValueError('archive size bound')
          digest = hashlib.sha256(data).hexdigest()
          if digest != 'bbb826252a0385ae9427e05353c9fbffa55265d91e351536eef365a1678d3d1e':
              raise ValueError('original artifact digest mismatch')
          with zipfile.ZipFile(pathlib.Path(sys.argv[1])) as archive:
              entries = archive.infolist()
              names = [info.filename for info in entries]
              if len(entries) > 100 or len(names) != len(set(names)):
                  raise ValueError('archive member count or duplicate')
              matches = [info for info in entries if info.filename.split('/')[-1] == 'preparation-result.json']
              if len(matches) != 1 or matches[0].file_size > 256 * 1024:
                  raise ValueError('exact bounded result member missing or ambiguous')
              result = json.loads(archive.read(matches[0]).decode('utf-8'))
          context = result.get('context', {})
          if context.get('checked_commit') != 'b269dba8e17e082395a46a90ef60b3a609c64921' or str(context.get('run_id')) != '42751353078':
              raise ValueError('native result identity mismatch')
          def safe_error(value):
              if value is None:
                  return None
              if not isinstance(value, dict):
                  raise ValueError('unexpected error representation')
              kind, message = value.get('type'), value.get('message')
              if not isinstance(kind, str) or not re.fullmatch(r'[A-Za-z0-9_]{1,80}', kind):
                  raise ValueError('unexpected error type')
              if not isinstance(message, str) or not re.fullmatch(r'[A-Za-z0-9_ .:/+()\-]{0,512}', message):
                  message = 'REDACTED_UNEXPECTED_ERROR_TEXT'
              return {'type': kind, 'message': message}
          print(json.dumps({'archive_sha256': digest, 'archive_members': len(entries),
              'checked_commit': context['checked_commit'], 'run_id': str(context['run_id']),
              'status': result.get('status'), 'preparation_error': safe_error(result.get('preparation_error')),
              'recovery_error': safe_error(result.get('recovery_error')),
              'daemon_mutation_completed': result.get('daemon_mutation_completed'),
              'source_and_test_exposed': result.get('source_and_test_exposed')}, sort_keys=True))
          PY
