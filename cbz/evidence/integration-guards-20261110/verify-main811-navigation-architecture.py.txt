def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def check_zip(path, expected_size, expected_sha, count):
    raw = path.read_bytes()
    if len(raw) != expected_size or sha(raw) != expected_sha:
        raise ValueError('archive identity: ' + path.name)
    z = zipfile.ZipFile(path)
    names = z.namelist()
    if len(names) != count or len(set(names)) != count or z.testzip() is not None:
        raise ValueError('archive members or CRC: ' + path.name)
    return z
