name: CBZ sandbox control evidence observation
on:
  push:
    branches: ["cbz/main-sandbox-observation-20261029"]
  workflow_dispatch:
permissions:
  actions: read
  contents: read
jobs:
  inspect:
    runs-on: ubuntu-24.04-arm
    timeout-minutes: 10
    steps:
      - name: Read exact archived architecture test logs
        shell: bash
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          set -euo pipefail
          gh api "/repos/$GITHUB_REPOSITORY/actions/artifacts/12930593803/zip" > "$RUNNER_TEMP/architecture.zip"
          printf '%s  %s\n' 98bfe64ead2b0be836dd0852d714c5a4dde06bf135f4f77a71199ea122ea03c2 "$RUNNER_TEMP/architecture.zip" | sha256sum --check
          python3 - "$RUNNER_TEMP/architecture.zip" <<'PY'
          import sys, zipfile
          names = [
              "test_kernel_enforces_cpu_memory_and_process_limits",
              "test_network_namespace_has_no_external_interface_or_route",
              "test_nonroot_and_no_new_privileges",
              "test_output_quota_stops_disk_exhaustion_and_cleanup_recovers_space",
              "test_source_mount_is_read_only_and_environment_has_no_publishing_secret",
          ]
          with zipfile.ZipFile(sys.argv[1]) as z:
              for suffix in ("tests-normal.log", "tests-optimized.log"):
                  matches = [n for n in z.namelist() if n == suffix or n.endswith("/" + suffix)]
                  if len(matches) != 1:
                      raise SystemExit("Unexpected log membership: " + suffix)
                  raw = z.read(matches[0])
                  lines = raw.decode("utf-8", "strict").splitlines()
                  for name in names:
                      hits = [line for line in lines if line.startswith(name + " ") or line.startswith(name + "(")]
                      if len(hits) != 1 or not hits[0].endswith(" ... ok"):
                          raise SystemExit("Missing or non-passing sandbox method: " + name + " in " + suffix)
                      print(suffix + ": " + hits[0])
          PY
