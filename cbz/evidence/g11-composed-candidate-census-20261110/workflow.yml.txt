name: CBZ exact composed G11 census observation
on:
  push:
    branches: [codex/cbz-architecture-observation-20261110]
permissions:
  contents: read
jobs:
  census:
    runs-on: ubuntu-24.04
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
        with:
          ref: ef2903b4028ca369223ff2128cc5f81ca6749ee0
          persist-credentials: false
      - uses: actions/setup-python@a26af69be951a213d495a4c3e4e4022e16d87065
        with:
          python-version: '3.11'
      - name: Verify exact source
        shell: bash
        run: |
          test "$(git rev-parse HEAD)" = ef2903b4028ca369223ff2128cc5f81ca6749ee0
          test "$(git rev-parse 'HEAD^{tree}')" = 8a26d14e12f20ea82c6cec1f134b18de5d55505b
          test "$(git hash-object tools/cbz/g11_census.py)" = 164b04e0b9906010aff9cf2e9ff88ae6810174f9
          test "$(git hash-object tools/cbz/g11_mutation_selftest.py)" = 9cb24001c37cf90b06cb10ea37f8b4cc5466c9e3
      - name: Observe normal census
        id: normal
        continue-on-error: true
        shell: bash
        run: |
          env -u PYTHONPATH -u PYTHONHOME -u PYTHONPYCACHEPREFIX -u PYTHONSTARTUP -u PYTHONUSERBASE python3 -B -S tools/cbz/g11_census.py --rev ef2903b4028ca369223ff2128cc5f81ca6749ee0 --json "$RUNNER_TEMP/g11-census-normal.json"
      - name: Observe optimized census
        id: optimized
        continue-on-error: true
        shell: bash
        run: |
          env -u PYTHONPATH -u PYTHONHOME -u PYTHONPYCACHEPREFIX -u PYTHONSTARTUP -u PYTHONUSERBASE python3 -B -S -O tools/cbz/g11_census.py --rev ef2903b4028ca369223ff2128cc5f81ca6749ee0 --json "$RUNNER_TEMP/g11-census-optimized.json"
      - name: Describe original reports
        if: always()
        shell: bash
        run: |
          python3 - <<'PY'
          import hashlib, json, os, pathlib
          root = pathlib.Path(os.environ["RUNNER_TEMP"])
          for mode in ("normal", "optimized"):
              path = root / ("g11-census-" + mode + ".json")
              if not path.is_file():
                  print("CENSUS_REPORT_MISSING", mode)
                  continue
              raw = path.read_bytes()
              data = json.loads(raw)
              print("CENSUS_REPORT", json.dumps({"mode":mode,"sha256":hashlib.sha256(raw).hexdigest(),
                  "bytes":len(raw),"commit":data["commit"],"status":data["status"],
                  "files":data["files"],"tracked":data["tracked"],"skipped":data["skipped"],
                  "failing":data["failing"],"failing_tools":data["failing_tools"]}, sort_keys=True))
          PY
      - uses: actions/upload-artifact@cf430e030ddbb5b0abf93d22962f4752f3646cd9
        if: always()
        with:
          name: cbz-composed-g11-census-${{ github.run_id }}-${{ github.run_attempt }}
          path: |
            ${{ runner.temp }}/g11-census-normal.json
            ${{ runner.temp }}/g11-census-optimized.json
          if-no-files-found: error
          retention-days: 90
      - name: Report original census outcomes
        if: always()
        shell: bash
        env:
          NORMAL: ${{ steps.normal.outcome }}
          OPTIMIZED: ${{ steps.optimized.outcome }}
        run: |
          printf 'normal=%s optimized=%s\n' "$NORMAL" "$OPTIMIZED"
          test "$NORMAL" = success && test "$OPTIMIZED" = success
