name: CBZ G11 workflow-format regression
on:
  push:
    branches: [codex/cbz-architecture-observation-20261110]
permissions:
  contents: read
jobs:
  origin-regression:
    runs-on: ubuntu-24.04
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
        with:
          persist-credentials: false
      - uses: actions/setup-python@a26af69be951a213d495a4c3e4e4022e16d87065
        with:
          python-version: '3.11'
      - name: Run actual-source origin regressions
        id: normal
        continue-on-error: true
        shell: bash
        run: |
          set -o pipefail
          python -B -S -m unittest tests.test_g11_workflow_formats -v 2>&1 | tee "$RUNNER_TEMP/g11-workflow-formats-normal.log"
      - name: Run optimized actual-source origin regressions
        id: optimized
        continue-on-error: true
        shell: bash
        run: |
          set -o pipefail
          python -B -S -O -m unittest tests.test_g11_workflow_formats -v 2>&1 | tee "$RUNNER_TEMP/g11-workflow-formats-optimized.log"
      - uses: actions/upload-artifact@cf430e030ddbb5b0abf93d22962f4752f3646cd9
        if: always()
        with:
          name: cbz-g11-workflow-formats-${{ github.run_id }}-${{ github.run_attempt }}
          path: |
            ${{ runner.temp }}/g11-workflow-formats-normal.log
            ${{ runner.temp }}/g11-workflow-formats-optimized.log
          if-no-files-found: error
          retention-days: 90
      - name: Report actual outcomes
        if: always()
        shell: bash
        env:
          NORMAL: ${{ steps.normal.outcome }}
          OPTIMIZED: ${{ steps.optimized.outcome }}
        run: |
          printf 'normal=%s optimized=%s\n' "$NORMAL" "$OPTIMIZED"
          test "$NORMAL" = success && test "$OPTIMIZED" = success
