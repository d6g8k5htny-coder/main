name: CBZ G11 mutation inventory preflight
on:
  push:
    branches: [codex/cbz-architecture-observation-20261110]
permissions:
  contents: read
jobs:
  preflight:
    runs-on: ubuntu-24.04
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
        with:
          persist-credentials: false
      - uses: actions/setup-python@a26af69be951a213d495a4c3e4e4022e16d87065
        with:
          python-version: '3.11'
      - name: Check targets and adapted case normally
        id: normal
        continue-on-error: true
        run: python3 -B -S cbz/evidence/g11-adapted-target-20261110/preflight.py.txt
      - name: Check targets and adapted case optimized
        id: optimized
        continue-on-error: true
        run: python3 -B -S -O cbz/evidence/g11-adapted-target-20261110/preflight.py.txt
      - name: Report actual outcomes
        if: always()
        env:
          NORMAL: ${{ steps.normal.outcome }}
          OPTIMIZED: ${{ steps.optimized.outcome }}
        run: |
          printf 'normal=%s optimized=%s\n' "$NORMAL" "$OPTIMIZED"
          test "$NORMAL" = success && test "$OPTIMIZED" = success
