          python3 -I -B -S - "$RUNNER_TEMP/cbz-artifact-readback" <<'PY'
          import collections, copy, hashlib, json, os, pathlib, re, sys, urllib.parse, urllib.request, zipfile
          root = pathlib.Path(sys.argv[1])
          commit = 'd94372e50662eaaf693af540b1fa133f3394f8b2'
          files = {
            'publisher_source': 'tools/current_site_publication.py',
            'publisher_test': 'tests/test_current_site_publication.py',
            'recovery_source': 'tools/cbz_site_receipt_recovery.py',
            'recovery_test': 'tests/test_cbz_site_recovery.py',
            'recovery_workflow': '.github/workflows/cbz-site-recovery.yml',
            'recovery_workflow_test': 'tests/test_cbz_site_recovery_workflow.py',
            'recorder': 'tools/cbz/reference_crawler_mutation_check.py',
          }
          def require(condition, why):
              if not condition: raise ValueError(why)
          def source_digest(path):
              url = 'https://api.github.com/repos/d6g8k5htny-coder/main/contents/' + urllib.parse.quote(path, safe='/') + '?ref=' + commit
              request = urllib.request.Request(url, headers={'Authorization': 'Bearer ' + os.environ['GH_TOKEN'], 'Accept': 'application/vnd.github.raw+json', 'User-Agent': 'cbz-bounded-readback'})
              with urllib.request.urlopen(request, timeout=30) as response:
                  require(response.status == 200 and response.geturl() == url, 'source response identity')
                  raw = response.read(1024 * 1024 + 1)
              require(len(raw) <= 1024 * 1024, 'source size bound')
              text = raw.decode('utf-8').replace('\r\n', '\n').replace('\r', '\n')
              return hashlib.sha256(text.encode('utf-8')).hexdigest()
          source = {key: source_digest(path) for key, path in files.items()}
          def verify_origin(report, mode, selected, phase):
              require(report.get('schema') == 'reference-crawler-origin-v3', 'origin schema')
              require(report.get('optimize') == mode and report.get('tests_run') == 1, 'origin mode/population')
              require(report.get('names') == [selected], 'selected test identity')
              require(report.get('ids') == report.get('started') == report.get('stopped') and len(report.get('ids', [])) == 1, 'test lifecycle')
              require(all(report.get(k) == 0 for k in ('skip_count', 'expected_failure_count', 'unexpected_success_count')), 'skip/unexpected result')
              events = report.get('events')
              require(type(events) is list, 'event list')
              if phase == 'baseline':
                  require(report.get('failure_count') == 0 and report.get('error_count') == 0 and not events, 'baseline failure')
              elif phase == 'selected':
                  require(type(report.get('failure_count')) is int and report['failure_count'] >= 1 and report.get('error_count') == 0 and report.get('authorized_failures') == [selected] and report.get('unauthorized_failures') == [] and report.get('errors') == [], 'selected failure identity')
                  require(len(events) == report['failure_count'] and all(e.get('kind') == 'failure' and e.get('test') == selected and e.get('authorized') is True for e in events), 'selected event provenance')
              elif phase == 'unauthorized':
                  require(report.get('failure_count') == 1 and report.get('error_count') == 0 and report.get('unauthorized_failures') == [selected] and report.get('authorized_failures') == [], 'unauthorized origin')
              elif phase == 'error':
                  require(report.get('failure_count') == 0 and report.get('error_count') == 1 and report.get('authorized_failures') == [], 'runtime error origin')
              else: raise ValueError('phase')
          def validate_archive(name, digest, publisher):
              path = root / (name + '.zip')
              require(path.stat().st_size <= 2 * 1024 * 1024, 'ZIP byte bound')
              require(hashlib.sha256(path.read_bytes()).hexdigest() == digest, 'ZIP digest')
              with zipfile.ZipFile(path) as archive:
                  infos = archive.infolist(); names = [x.filename for x in infos]
                  require(len(infos) <= 10000 and len(names) == len(set(names)), 'ZIP member uniqueness/bound')
                  require(sum(x.file_size for x in infos) <= 64 * 1024 * 1024, 'ZIP expansion bound')
                  require(all(not n.startswith('/') and '..' not in n.split('/') and '\\' not in n for n in names), 'ZIP unsafe path')
                  require(archive.testzip() is None, 'ZIP CRC failure')
                  def get(suffix):
                      selected = [x for x in infos if x.filename == suffix or x.filename.endswith('/' + suffix)]
                      require(len(selected) == 1 and selected[0].file_size <= 1024 * 1024, 'missing/ambiguous bounded member: ' + suffix)
                      return archive.read(selected[0])
                  actual = {}
                  for mode, flag in [('normal', 0), ('optimized', 1)]:
                      log = get('unit-' + mode + '.log').decode('utf-8')
                      expected_count = 40 if publisher else 34
                      require(re.search(r'(?m)^Ran ' + str(expected_count) + r' tests in [0-9.]+s$', log) is not None and log.rstrip().endswith('\nOK'), 'unit footer')
                      require(len(re.findall(r'(?m)^test_.* \.\.\. ok$', log)) == expected_count and ' ... skipped ' not in log, 'unit outcomes')
                      summary = json.loads(get('mutations-' + mode + '/results.json'))
                      require(summary.get('optimize') == flag, 'summary mode')
                      if publisher:
                          binding = {'source_sha256': 'publisher_source', 'test_sha256': 'publisher_test', 'recorder_sha256': 'recorder'}
                          expected = {'authorized-selected-failure': 11, 'rejected-unauthorized-origin': 3, 'rejected-error-origin': 1}
                      else:
                          binding = {'subject_sha256': 'recovery_source', 'test_sha256': 'recovery_test', 'workflow_sha256': 'recovery_workflow', 'workflow_test_sha256': 'recovery_workflow_test', 'recorder_sha256': 'recorder'}
                          expected = {'authorized-selected-failure': 38, 'rejected-unauthorized-origin': 3}
                      require(all(summary.get(key) == source[value] for key, value in binding.items()), 'tested source hash binding')
                      rows = summary.get('results'); require(type(rows) is list, 'results list')
                      require(dict(collections.Counter(row.get('mutant') for row in rows)) == expected, 'outcome counts')
                      require(len({row.get('mutation') for row in rows}) == len(rows), 'distinct mutation labels')
                      if not publisher:
                          require(sum(row.get('mutation') == 'deployment-empty-status' and row.get('mutant') == 'authorized-selected-failure' for row in rows) == 1, 'empty-status mutation missing')
                      controls = 0; synthetic = 0
                      for row in rows:
                          label = row['mutation']; require(re.fullmatch(r'[a-z0-9-]+', label) is not None, 'mutation label')
                          prefix = 'mutations-' + mode + '/' + label + '/'
                          selected = row.get('selected')
                          if row['mutant'] == 'authorized-selected-failure':
                              require(row.get('baseline') == 'pass' and type(selected) is str, 'clean baseline')
                              base = json.loads(get(prefix + 'baseline/origin.json')); changed = json.loads(get(prefix + 'mutant/origin.json'))
                              verify_origin(base, flag, selected, 'baseline')
                              try:
                                  verify_origin(changed, flag, selected, 'selected')
                              except ValueError:
                                  print(json.dumps({'failed_label': label, 'mode': mode, 'selected': selected, 'origin_fields': {k: changed.get(k) for k in ('names', 'failure_count', 'error_count', 'authorized_failures', 'unauthorized_failures', 'errors', 'events')}}, sort_keys=True))
                                  raise
                              choices = [candidate for candidate in ('mutant.py', 'mutant.yml') if any(n == prefix + candidate or n.endswith('/' + prefix + candidate) for n in names)]
                              require(len(choices) == 1, 'missing or ambiguous mutant source')
                              mutant_path = choices[0]
                              claimed = row.get('sha256') if publisher else row.get('mutant_sha256')
                              require(hashlib.sha256(get(prefix + mutant_path)).hexdigest() == claimed, 'mutant bytes')
                              if controls == 0:
                                  for key, value in [('optimize', 1 - flag), ('skip_count', 1), ('authorized_failures', [])]:
                                      bad = copy.deepcopy(changed); bad[key] = value
                                      try: verify_origin(bad, flag, selected, 'selected')
                                      except ValueError: synthetic += 1
                                      else: raise ValueError('synthetic adverse origin accepted')
                              controls += 1
                          else:
                              if publisher:
                                  origin = json.loads(get(prefix + 'origin.json'))
                                  if row['mutant'] == 'rejected-error-origin':
                                      selected = origin['names'][0]; verify_origin(origin, flag, selected, 'error')
                                  else: verify_origin(origin, flag, selected, 'unauthorized')
                              else:
                                  origin = json.loads(get(prefix + 'child/origin.json')); verify_origin(origin, flag, selected, 'unauthorized')
                      require(synthetic == 3, 'synthetic adverse checks incomplete')
                      actual[mode] = {'unit_tests': expected_count, 'selected_mutation_pairs': controls, 'other_outcomes': expected, 'synthetic_origin_refusals': synthetic}
                  print(json.dumps({'artifact': name, 'sha256': digest, 'entries': len(infos), 'crc': 'PASS', 'checked_source_commit': commit, 'modes': actual}, sort_keys=True))
