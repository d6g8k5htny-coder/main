def interval(v):

class KernelControls(unittest.TestCase):

        # A missing integer guard would accept V1's false real-M claim.
        for m in [0, -1, 0.9, True, "2", Q(9, 10)]:
            with self.subTest(m=m):
                with self.assertRaises(KernelHold) as e:
                    certify(request(cutoff=m))

    def test_positive_input_required(self):
        for lo, hi in [("0/1", "1/1"), ("2/1", "1/1"), ("-1/1", "1/1")]:

    def test_zero_mode_and_independent_geometric_normalizer_control(self):
        c = certify(request())
        lo, hi = interval(c["bounds"]["z"])
        self.assertGreater(lo, Q(1))
        # pi>3 => alpha>18; e^3>20 by its positive Taylor terms.
        # Sum exp(-alpha*m^2) < sum (20^-6)^m, so z<1+1e-6.
        self.assertLess(hi, Q(1000001, 1000000))
        self.assertGreater(interval(c["bounds"]["mu2"])[0], 0)

    def test_physical_frequency_not_normalized_integer_frequency(self):
        c = certify(request())
        a, b, d = [interval(c["bounds"][k]) for k in ("mu2", "mu4", "mu6")]
        # Every nonzero q has q^2=h^2*m^2>36 at L=1, since pi>3.
        self.assertGreater(b[0], 36*a[1])
        self.assertGreater(d[0], 36*b[1])

    def test_pi_interval_has_independent_broad_analytic_guard(self):
        lo, hi = interval(certify(request())["pi"]["interval"])
        self.assertGreater(lo, 3)
        self.assertLess(hi, 4)

        self.assertGreater(lo, Q(1, 3))
        self.assertLess(hi, Q(3, 8))
