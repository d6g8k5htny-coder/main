theorem lifetime_torusDist_eq_infDist
    (d : ℕ) (L : ℝ) (hL : 0 < L) (x y : Lifetime.E d) :
    Lifetime.torusDist d L x y =
      Metric.infDist (y - x)
        ((torus_periodLattice d L hL).toAddSubgroup : Set (Lifetime.E d)) := by
  classical
  let S : Set (Lifetime.E d) :=
    ((torus_periodLattice d L hL).toAddSubgroup : Set (Lifetime.E d))
  have hS : S.Nonempty :=
    ⟨0, (torus_periodLattice d L hL).zero_mem⟩
  have hlattice (n : Fin d → ℤ) :
      Lifetime.latticeVec d L n = torus_latticeVec d L n := rfl
  have hdist (n : Fin d → ℤ) :
      dist (y - x) (torus_latticeVec d L n) =
        ‖x - y + Lifetime.latticeVec d L n‖ := by
    rw [dist_eq_norm', hlattice n]
    congr 1
    abel
  have hmem (n : Fin d → ℤ) : torus_latticeVec d L n ∈ S := by
    change torus_latticeVec d L n ∈ torus_periodLattice d L hL
    exact (torus_lattice_mem_iff d L hL _).2 ⟨n, rfl⟩
  have hbounded : BddBelow
      (Set.range (fun n : Fin d → ℤ => ‖x - y + Lifetime.latticeVec d L n‖)) := by
    refine ⟨0, ?_⟩
    rintro _ ⟨n, rfl⟩
    exact norm_nonneg _
  change (⨅ n : Fin d → ℤ, ‖x - y + Lifetime.latticeVec d L n‖) =
    Metric.infDist (y - x) S
  apply le_antisymm
  · apply (Metric.le_infDist hS).2
    intro q hq

    obtain ⟨n, hn⟩ := (torus_lattice_mem_iff d L hL q).1 hq'

          ‖x - y + Lifetime.latticeVec d L n‖ := ciInf_le hbounded n

/-- Distinct points of the existing quotient have positive literal Lifetime
distance. The dimension is arbitrary; positive period and quotient separation

theorem lifetime_torusDist_pos_of_torus_proj_ne
    (d : ℕ) (L : ℝ) (hL : 0 < L) (x y : Lifetime.E d)
    (hxy : torus_proj d L hL x ≠ torus_proj d L hL y) :
    0 < Lifetime.torusDist d L x y := by
  let S : Set (Lifetime.E d) :=
    ((torus_periodLattice d L hL).toAddSubgroup : Set (Lifetime.E d))
  have hS : S.Nonempty :=
    ⟨0, (torus_periodLattice d L hL).zero_mem⟩
  have hclosed : IsClosed S := by
    change IsClosed
      ((torus_periodLattice d L hL).toAddSubgroup : Set (Lifetime.E d))
    exact AddSubgroup.isClosed_of_discreteTopology
  have hnotmem : y - x ∉ S := by
    intro hmem
    have hmem' : y - x ∈ torus_periodLattice d L hL := hmem
    obtain ⟨n, hn⟩ := (torus_lattice_mem_iff d L hL (y - x)).1 hmem'
    apply hxy
    apply (torus_proj_eq_iff d L hL x y).2
    refine ⟨n, ?_⟩
    rw [hn]
    abel
  rw [lifetime_torusDist_eq_infDist d L hL x y]
  exact (hclosed.notMem_iff_infDist_pos hS).1 hnotmem
