def eye(n):
    return [[F(i == j) for j in range(n)] for i in range(n)]


def tr(a):
    return [list(row) for row in zip(*a)]


def mm(a, b):
    return [[sum((x*y for x, y in zip(row, col)), F(0))
             for col in zip(*b)] for row in a]


def add(a, b):
    return [[x+y for x, y in zip(ar, br)] for ar, br in zip(a, b)]


def scale(c, a):
    return [[c*x for x in row] for row in a]


def mv(a, x):
    return [sum((u*v for u, v in zip(row, x)), F(0)) for row in a]


def dot(x, y):
    return sum((a*b for a, b in zip(x, y)), F(0))


def quad(x, a):
    return dot(x, mv(a, x))


def det(a):
    b = [row[:] for row in a]
    d = F(1)
    for k in range(len(a)):
        pivot = next((j for j in range(k, len(a)) if b[j][k]), None)
        if pivot is None:
            return F(0)
        if pivot != k:
            b[k], b[pivot] = b[pivot], b[k]
            d = -d
        p = b[k][k]
        d *= p
        for j in range(k+1, len(a)):
            c = b[j][k]/p
            for i in range(k+1, len(a)):
                b[j][i] -= c*b[k][i]
    return d


def det_permutation(a):
    total = F(0)
    n = len(a)
    for p in permutations(range(n)):
        inversions = sum(p[i] > p[j] for i in range(n) for j in range(i+1, n))
        v = F((-1)**inversions)
        for i in range(n):
            v *= a[i][p[i]]
        total += v
    return total


def inv(a):
    n = len(a)
    b = [a[i][:] + eye(n)[i] for i in range(n)]
    for k in range(n):
        pivot = next((j for j in range(k, n) if b[j][k]), None)
        if pivot is None:
            raise ValueError('singular')
        b[k], b[pivot] = b[pivot], b[k]
        p = b[k][k]
        b[k] = [x/p for x in b[k]]
        for j in range(n):
            if j != k:
                c = b[j][k]
                b[j] = [x-c*y for x, y in zip(b[j], b[k])]
    return [row[n:] for row in b]


def spd(a):
    return (a == tr(a) and all(det([row[:k] for row in a[:k]]) > 0
                               for k in range(1, len(a)+1)))


def all_principal_nonnegative(a):
    n = len(a)
    return all(det([[a[i][j] for j in range(n) if mask >> j & 1]
                    for i in range(n) if mask >> i & 1]) >= 0
               for mask in range(1, 1 << n))


def closed_parameters(s0, s, delta):
    """Return rational (squared prefactor, exponent) of 1+chi^2(P||Q)."""
    k = add(scale(F(2), s0), scale(F(-1), s))
    if not spd(s0) or not spd(s) or not spd(k):
        raise ValueError('requires S0>0, S>0, and 2*S0-S>0')
    return det(s0)**2/(det(s)*det(k)), quad(delta, inv(k))


def rho_bound(n, eta, b2):
    if type(n) is not int or n < 1 or not (0 <= eta < 1) or b2 < 0:
        raise ValueError('invalid envelope')
    r = F(n, 2)*eta**2/(1-eta**2) + b2/(1-eta)
    if not (0 <= r < 1):
        raise ValueError('rational shortcut requires rho<1; exact route may exist')
    return r, r/(1-r)


def sqrt_upper(x, digits=12):
    if x < 0:
        raise ValueError('negative square')
    base = 10**digits
    y = isqrt(x.numerator*base*base//x.denominator)
    if F(y*y, base*base) < x:
        y += 1
    return F(y, base)


def polynomial_add(a, b):
    c = a.copy()
    for k, v in b.items():
        c[k] = c.get(k, F(0)) + v
    return {k: v for k, v in c.items() if v}


def polynomial_mul(a, b):
    c = {}
    for ka, va in a.items():
        for kb, vb in b.items():
            k = tuple(x+y for x, y in zip(ka, kb))
            c[k] = c.get(k, F(0)) + va*vb
    return {k: v for k, v in c.items() if v}


def standard_moment(powers):
    ans = 1
    for k in powers:
        if k % 2:
            return F(0)
        ans *= factorial(k)//(2**(k//2)*factorial(k//2))
    return F(ans)


def latent_expectation(poly):
    return sum((v*standard_moment(k) for k, v in poly.items()), F(0))


def moment_function(mu, cov):
    @lru_cache(None)
    def moment(powers):
        if not any(powers):
            return F(1)
        i = next(i for i, x in enumerate(powers) if x)
        b = list(powers)
        b[i] -= 1
        answer = mu[i]*moment(tuple(b))
        for j, v in enumerate(b):
            if v:
                c = b[:]
                c[j] -= 1
                answer += v*cov[i][j]*moment(tuple(c))
        return answer
    return moment
