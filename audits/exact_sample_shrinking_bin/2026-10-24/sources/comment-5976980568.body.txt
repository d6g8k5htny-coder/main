     exactly when c A^2 > 10/3 (checked on both sides of the threshold).
Mutants (--mutant X) must make the run exit 1: HALF_STRIPS (strips cut to [a-d,a) u [b-d,b)), A_GE_D (allow a = d),
WRONG_GEOM (h^2/4 instead of h^2/2), DET_POWER1 (the false bound |det M| <= ||M||_op), RATE_103 (threshold 3 instead of

from fractions import Fraction as F

MUT = None
if len(sys.argv) == 3 and sys.argv[1] == '--mutant':
    MUT = sys.argv[2]
    if MUT not in ('HALF_STRIPS', 'A_GE_D', 'WRONG_GEOM', 'DET_POWER1', 'RATE_103'):

elif len(sys.argv) != 1:
    print('usage: esb_check.py [--mutant LABEL]'); sys.exit(2)

def check(cond, msg):
    if not cond:
        fails.append(msg)

def in_bin(l, a, b):
    return a <= l < b

def strips(a, b, d):
    if MUT == 'HALF_STRIPS':
        return [(a - d, a), (b - d, b)]
    return [(a - d, a + d), (b - d, b + d)]

def count(ls, lo, hi):
    return sum(1 for l in ls if lo <= l < hi)

def count_union(ls, ivs):
    return sum(1 for l in ls if any(lo <= l < hi for lo, hi in ivs))

# ---------------------------------------------------------------- S: the sharp-bin injection
rng = random.Random(20261004)
nS = 0
for trial in range(4000):
    den = rng.choice([4, 8, 12, 24])
    eps = F(rng.randint(1, 6), den)
    d = 2 * eps
    P = []
    for _ in range(rng.randint(0, 12)):
        birth = F(rng.randint(0, 200), den)
        life = F(rng.randint(1, 80), den)
        P.append((birth, birth - life))           # superlevel bar: birth > death
    # matched images: each endpoint moves by at most eps (exact multiples of 1/den to hit endpoints)
    Q, pairs = [], []
    for (bP, sP) in P:
        while True:

            ds = F(rng.randint(-int(eps * den), int(eps * den)), den)
            bQ, sQ = bP + db, sP + ds
            if bQ - sQ > 0:
                break

        Q.append((bQ, sQ)); pairs.append(((bP, sP), (bQ, sQ)))

    for _ in range(rng.randint(0, 4)):

        Q.append((bQ, bQ - F(rng.randint(1, int(d * den)), den)))
    # P bars matched to the diagonal would have lifetime <= d: drop matched P pairs that are short with prob 1/3
    lP = [b - s for b, s in P]
    lQ = [b - s for b, s in Q]

        a = F(rng.randint(1, 60), den)
        b = a + F(rng.randint(1, 60), den)

        nS += 1

d = F(1, 2)

P2, Q2 = [], [d]

gmax = F(0)

            else:                                    # diagonal (1,0)-(0,1)

        check(r1 < prev[0] and r2 < prev[1] and r3 < prev[2], 'R: ratios must decrease along the schedule')
    prev = (r1, r2, r3)
